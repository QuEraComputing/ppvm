4//! Hot-path Pauli product / commutator on raw word chunks.
5//!
6//! Phase encoding: Pauli product `P·Q = ω · R` where `ω = i^phase` and
7//! `phase ∈ {0,1,2,3}` ↔ `ω ∈ {1, i, -1, -i}`. The per-byte XOR/AND
8//! formulas are the same ones used by
9//! [`ppvm_pauli_word::phase::PhasedPauliWord`]'s `MulAssign`. This module
10//! keeps a copy that returns the unpacked `(word, phase)` pair without
11//! constructing a phased wrapper.
12
13use crate::word::{Chunk, W_CHUNKS, Word};
14use fxhash::FxHashMap;
15use num::Complex;
16use ppvm_traits::PauliWordTrait;
17
18/// Magnitude below which an expanded Pauli coefficient is treated as
19/// cancellation noise and dropped.
20pub(crate) const COEFF_DROP_TOL: f64 = 1e-14;
21
22/// One Pauli term in a complex linear combination (a single summand of
23/// `L = Σ_a λ_a P_a`, or of a precomputed product such as `L†L`).
24#[derive(Clone)]
25pub(crate) struct PauliTerm<const C: usize = W_CHUNKS> {
26    pub(crate) word: Word<C>,
27    pub(crate) coeff: Complex<f64>,
28}
29
30/// Expand `A†B = (Σ_a λ_a P_a)† (Σ_b μ_b P_b) = Σ_{a,b} λ_a* μ_b P_a P_b`
31/// as a Pauli linear combination, dropping FP-noise zeros. For `A = B`
32/// (the jump-operator `L†L`) the coefficients are real; in general they
33/// are complex.
34pub(crate) fn precompute_adag_b<const C: usize>(
35    a_terms: &[PauliTerm<C>],
36    b_terms: &[PauliTerm<C>],
37) -> Vec<PauliTerm<C>> {
38    let zero = Complex::new(0.0, 0.0);
39    let mut acc: FxHashMap<Word<C>, Complex<f64>> = FxHashMap::default();
40    for a in a_terms {
41        for b in b_terms {
42            let (word, phase) = pauli_mul(&a.word, &b.word);
43            let coeff = a.coeff.conj() * b.coeff * phase_factor(phase);
44            *acc.entry(word).or_insert(zero) += coeff;
45        }
46    }
47    acc.into_iter()
48        .filter(|(_, c)| c.norm() > COEFF_DROP_TOL)
49        .map(|(word, coeff)| PauliTerm { word, coeff })
50        .collect()
51}

53/// Union of the supports (`xbits | zbits`) of every term, as raw chunks.
54pub(crate) fn support_mask<const C: usize>(terms: &[PauliTerm<C>]) -> [Chunk; C] {
55    let mut mask = [0 as Chunk; C];
56    for t in terms {
57        for (i, slot) in mask.iter_mut().enumerate() {
58            *slot |= t.word.xbits.data[i] | t.word.zbits.data[i];
59        }
60    }
61    mask
62}

64#[inline(always)]
65pub(crate) fn phase_factor(phase: u8) -> Complex<f64> {
66    match phase & 3 {
67        0 => Complex::new(1.0, 0.0),
68        1 => Complex::new(0.0, 1.0),
69        2 => Complex::new(-1.0, 0.0),
70        _ => Complex::new(0.0, -1.0),
71    }
72}

74/// `true` if Pauli words `a` and `b` anti-commute.
75///
76/// Two Pauli strings anti-commute iff
77/// `popcount(a.x & b.z) + popcount(a.z & b.x)` is odd.
78#[inline(always)]
79pub(crate) fn anti_commutes<const C: usize>(a: &Word<C>, b: &Word<C>) -> bool {
80    let mut bits: u32 = 0;
81    for i in 0..C {
82        bits += (a.xbits.data[i] & b.zbits.data[i]).count_ones();
83        bits += (a.zbits.data[i] & b.xbits.data[i]).count_ones();
84    }
85    bits & 1 == 1
86}

88/// Commutator product `h · p`: returns `(out, eps)` where `out = h ⊕ p` and
89///
90/// - `eps =  0` if `h` and `p` commute (caller should skip — `[h,p] = 0`),
91/// - `eps = -2.0` if `h·p` has phase `+i` (so `i·[h,p] = -2·out`),
92/// - `eps = +2.0` if `h·p` has phase `-i` (so `i·[h,p] = +2·out`).
93#[inline(always)]
94pub(crate) fn comm_product<const C: usize>(h: &Word<C>, p: &Word<C>) -> (Word<C>, f64) {
95    let (out, phase) = pauli_mul(h, p);
96    let eps = match phase {
97        1 => -2.0,
98        3 => 2.0,
99        _ => 0.0,
100    };
101    (out, eps)
102}

104/// Full Pauli product `p · q`: returns `(out, phase)` where the product
105/// is `ω · out` with `ω = i^phase`.
106#[inline(always)]
107pub(crate) fn pauli_mul<const C: usize>(p: &Word<C>, q: &Word<C>) -> (Word<C>, u8) {
108    let mut out = Word::<C>::new(p.n_qubits());
109    let mut sign_count: u32 = 0;
110    let mut imag_count: u32 = 0;
111    for i in 0..C {
112        let a = p.xbits.data[i];
113        let b = p.zbits.data[i];
114        let c = q.xbits.data[i];
115        let d = q.zbits.data[i];
116        let sign = (a & b & c & !d) | (a & !b & !c & d) | (!a & b & c & d);
117        let imag = (a & !b & d) | (a & !c & d) | (!a & b & c) | (b & c & !d);
118        sign_count += sign.count_ones();
119        imag_count += imag.count_ones();
120        out.xbits.data[i] = a ^ c;
121        out.zbits.data[i] = b ^ d;
122    }
123    out.rehash();
124    (out, ((2 * sign_count + imag_count) & 3) as u8)
125}
