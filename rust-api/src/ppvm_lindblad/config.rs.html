4//! Configuration objects for the predictor-corrector stepper.
5
6/// Truncation and execution policy for a single predictor-corrector step
7/// ([`crate::LindbladSpec::pc_step`], [`crate::LindbladSpec::pc_step_timed`]).
8///
9/// These are the per-run *tuning knobs*, kept separate from the per-call data
10/// (`basis`, `coeffs`, `dt`, `protected`).
11///
12/// `max_basis` is the primary accuracy/cost dial; `admit_basis` selects the
13/// displacement scheme; `drop_tol` is the churn valve of the admission-bound
14/// scheme; `tau_add` is a wall optimization at most.
15#[derive(Debug, Clone, Copy)]
16pub struct PcStepConfig {
17    /// Hard rank cap on the retained basis: after the corrector, only the
18    /// top-`max_basis` strings by `|coeff|` are kept (protected words always
19    /// survive). The primary convergence dial — verify by re-running at 2×.
20    pub max_basis: usize,
21    /// Working-set (admission) bound. When `Some(a)` with `a > max_basis`,
22    /// enrichment may grow the live basis to `a` and the final cap performs a
23    /// genuine top-`max_basis`-of-union rank displacement (the analog of
24    /// two-site TDVP truncation at `χ_max`); `drop_tol` is then not needed
25    /// for membership turnover. `None` bounds admission by `max_basis`
26    /// itself — the valve scheme, which requires `drop_tol > 0` to keep the
27    /// basis adapting once it fills.
28    pub admit_basis: Option<usize>,
29    /// Magnitude prune applied after the corrector: basis entries whose
30    /// `|coeff|` is below `drop_tol` are discarded (protected words are
31    /// always kept). `<= 0.0` disables pruning — valid only with
32    /// `admit_basis` set, otherwise the basis freezes once it fills the cap.
33    pub drop_tol: f64,
34    /// Optional absolute rate threshold on leakage admission: a candidate is
35    /// admitted only if its inflow rate exceeds `tau_add`. This is the
36    /// natural (dt- and drop_tol-independent) parameterization — the
37    /// admission accuracy cliff sits at a fixed `tau_add`. `None` = no
38    /// filter, the recommended default with cap-based truncation.
39    pub tau_add: Option<f64>,
40    /// When `Some(n)`, run the entire step inside a freshly built rayon
41    /// thread pool of `n` threads (useful for benchmarking parallel
42    /// scaling). When `None`, the global rayon pool is used.
43    pub num_threads: Option<usize>,
44}
45
46impl Default for PcStepConfig {
47    /// Uncapped, unfiltered, no pruning: the near-exact reference
48    /// configuration. Production runs should set `max_basis` (and usually
49    /// `admit_basis ≈ 2-3×` it).
50    fn default() -> Self {
51        Self {
52            max_basis: usize::MAX,
53            admit_basis: None,
54            drop_tol: 0.0,
55            tau_add: None,
56            num_threads: None,
57        }
58    }
59}
