4//! Matrix-free `exp(dt · L*) · b`, driven by the external `quspin-expm`
5//! crate — for both the real (`f64`) adaptive path and the complex,
6//! phase-aware orbit-rep path.
7//!
8//! The in-basis-restricted generator's per-column action is computed ONCE
9//! per expm call (via
10//! [`build_mf_cols`] / [`build_orbit_rep_cols`]), stored row-major, and
11//! reused across every Krylov/Taylor matvec
12//! by [`CsrOp`] (a [`quspin_types::LinearOperator`]) fed to
13//! [`quspin_expm::ExpmOp::from_parts`]. Each matvec is then a cheap,
14//! deterministic CSR product; the Pauli-commutator action is never
15//! recomputed per matvec.
16//! `from_parts` (rather than `ExpmOp::new`) supplies the diagonal shift `μ`,
17//! the partition count `s`, and the truncation order `m*` directly, bypassing
18//! quspin's adaptive parameter selection — so the 1-norm *estimator* and
19//! `dot_transpose` are never invoked on the single-vector `apply` path; only
20//! [`LinearOperator::dot`] runs.
21//!
22//! `μ`, the trace, and the column 1-norm of `A − μ·I` are computed in the
23//! same single action pass as the cache, and turned into an `apply` by the
24//! shared [`expm_apply_cached`] tail. The `(m, s)` Taylor partition is
25//! picked with the tolerance-matched tables in [`crate::expm`]: a relaxed
26//! `tol=1e-6` table when the PC prunes coarsely (`drop_tol ≥ 1e-4`), else the
27//! double-precision table (keeping the exact-reference test paths bit-exact).
28

39/// Per-column `(raw, diag)` for the `μ`/1-norm selection: `raw` bounds
40/// `Σ_r |M[r,c]|` from above and `diag = M[c,c]`.
41type PerCol<T> = Vec<(f64, T)>;
42
43/// Scratch buffers for [`LindbladSpec::compute_action_terms`].
44type ActionScratch<const C: usize> = (Vec<u32>, Vec<u32>, FxHashMap<Word<C>, Complex<f64>>);
45
46/// One `(row, col, val)` entry of the generator, staged between the
47/// column-wise action pass and the row-major layout.
48type Entry<T> = (u32, u32, T);
49
50/// Staged entries indexed by row range (one bucket per [`CsrRange`]).
51type Buckets<T> = Vec<Vec<Entry<T>>>;
52
53/// Consecutive CSR rows stored flat: local row `i` holds
54/// `cols[offsets[i]..offsets[i + 1]]` and the matching `vals`, in
55/// ascending column order.
56struct CsrRange<T> {
57    offsets: Vec<u32>,
58    cols: Vec<u32>,
59    vals: Vec<T>,
60}
61
62/// Cached in-basis action in row-major (CSR) form, stored as ranges of
63/// `1 << range_shift` consecutive rows.
64///
65/// Row-major storage makes the matvec deterministic: each output entry is
66/// summed by one thread over its row in ascending column order, so `M·x`
67/// is bit-identical for every rayon thread count, with no per-thread
68/// partial vectors and no serial reduction. `range_shift` depends on
69/// `dim` only.
70///
71/// One exactly-sized allocation triple per range replaces one `Vec` per
72/// row: at `|basis| ~ 10^6` a per-row layout would leave ~10^6 small
73/// allocations for the system allocator to retain after the expm call.
74pub(crate) struct BlockCsr<T> {
75    ranges: Vec<CsrRange<T>>,
76    range_shift: u32,
77    dim: usize,
78}
79
80/// `log2` of the rows per [`CsrRange`]: the smallest power of two that
81/// gives at most 256 ranges, and at least 64 rows per range. A function of
82/// `dim` alone, so the layout (and every row's summation order) is
83/// independent of the thread count. A power of two turns the per-entry
84/// range lookup into a shift.
85fn range_shift(dim: usize) -> u32 {
86    dim.div_ceil(256)
87        .max(64)
88        .next_power_of_two()
89        .trailing_zeros()
90}

92/// Build the [`BlockCsr`] cache and the per-column `(raw, diag)` data for
93/// a `dim`-column generator. `col(c, scratch, rows, vals)` appends the
94/// in-basis entries of column `c` to `rows`/`vals` and returns its
95/// `(raw, diag)`.
96///
97/// The action is naturally column-wise, so the build runs in two parallel
98/// passes. (1) Over column blocks: compute the action and stage each entry
99/// in an exactly-sized bucket of its row range, in column order. (2) Over
100/// row ranges: a stable counting sort of the range's buckets (taken in
101/// block order) into CSR, dropping each bucket once consumed. The cache is
102/// never held in both layouts at once, but staging stores the row as well:
103/// 16 B/entry for real `T` and 24 B/entry for complex `T`, against
104/// 12 B/entry and 20 B/entry for the final CSR.
105fn build_block_csr<const C: usize, T, F>(
106    spec: &LindbladSpec<C>,
107    dim: usize,
108    col: F,
109) -> (BlockCsr<T>, PerCol<T>)
110where
111    T: Coeff,
112    F: Fn(usize, &mut ActionScratch<C>, &mut Vec<u32>, &mut Vec<T>) -> (f64, T) + Sync,
113{
114    // ~16 blocks per thread for load balance, but never so small that the
115    // per-block allocations matter. Block boundaries do not affect the
116    // result: pass 2 restores ascending column order within every row.
117    let block = dim
118        .div_ceil(16 * rayon::current_num_threads().max(1))
119        .clamp(64, 4096);
120    let shift = range_shift(dim);
121    let n_ranges = dim.div_ceil(1 << shift);
122
123    // Pass 1: `staged[b][k]` holds block `b`'s entries in row range `k`.
124    let (staged, per_col): (Vec<Buckets<T>>, Vec<PerCol<T>>) = (0..dim.div_ceil(block))
125        .into_par_iter()
126        .map_init(
127            || {
128                let scratch: ActionScratch<C> = (
129                    Vec::with_capacity(spec.n_qubits()),
130                    Vec::with_capacity(128),
131                    FxHashMap::with_capacity_and_hasher(128, FxBuildHasher::default()),
132                );
133                (
134                    scratch,
135                    Vec::<u32>::new(),
136                    Vec::<T>::new(),
137                    Vec::<usize>::new(),
138                    vec![0usize; n_ranges],
139                )
140            },
141            |(scratch, rows, vals, ends, counts), b| {
142                let cols = (b * block)..dim.min((b + 1) * block);
143                rows.clear();
144                vals.clear();
145                ends.clear();
146                let mut per_col = Vec::with_capacity(cols.len());
147                for c in cols.clone() {
148                    per_col.push(col(c, scratch, rows, vals));
149                    ends.push(rows.len());
150                }
151                counts.fill(0);
152                for &r in rows.iter() {
153                    counts[(r >> shift) as usize] += 1;
154                }
155                let mut buckets: Buckets<T> =
156                    counts.iter().map(|&k| Vec::with_capacity(k)).collect();
157                let mut lo = 0;
158                for (c, &hi) in cols.zip(ends.iter()) {
159                    for e in lo..hi {
160                        let r = rows[e];
161                        buckets[(r >> shift) as usize].push((r, c as u32, vals[e]));
162                    }
163                    lo = hi;
164                }
165                (buckets, per_col)
166            },
167        )
168        .unzip();
169    let per_col = per_col.into_iter().flatten().collect();
170
171    // Regroup the bucket handles by row range (moves `Vec` headers only).
172    let mut by_range: Vec<Buckets<T>> = (0..n_ranges)
173        .map(|_| Vec::with_capacity(staged.len()))
174        .collect();
175    for blk in staged {
176        for (k, bucket) in blk.into_iter().enumerate() {
177            by_range[k].push(bucket);
178        }
179    }
180
181    // Pass 2: stable counting sort by row. Buckets are visited in block
182    // order and each is in column order, so every row ends up in
183    // ascending column order.
184    let ranges = by_range
185        .into_par_iter()
186        .enumerate()
187        .map(|(k, buckets)| {
188            let row0 = k << shift;
189            let nrows = (1usize << shift).min(dim - row0);
190            let nnz: usize = buckets.iter().map(Vec::len).sum();
191            // The per-range offsets are u32; check once so the counts
192            // below cannot wrap.
193            u32::try_from(nnz).expect("CSR row range exceeds u32 entries");
194            let mut offsets = vec![0u32; nrows + 1];
195            for &(r, _, _) in buckets.iter().flatten() {
196                offsets[r as usize - row0 + 1] += 1;
197            }
198            for i in 0..nrows {
199                offsets[i + 1] += offsets[i];
200            }
201            let mut cursor = offsets[..nrows].to_vec();
202            let mut cols = vec![0u32; nnz];
203            let mut vals = vec![T::zero(); nnz];
204            for bucket in buckets {
205                for (r, c, v) in bucket {
206                    let i = r as usize - row0;
207                    let p = cursor[i] as usize;
208                    cols[p] = c;
209                    vals[p] = v;
210                    cursor[i] += 1;
211                }
212            }
213            CsrRange {
214                offsets,
215                cols,
216                vals,
217            }
218        })
219        .collect();
220    (
221        BlockCsr {
222            ranges,
223            range_shift: shift,
224            dim,
225        },
226        per_col,
227    )
228}

230/// Per-column in-basis action of the real generator `M`, plus the data the
231/// `(m, s)`/`μ` selection needs — all from ONE action pass over the basis.
232///
233/// Returns `(mat, per_col)`. Column `c` of the row-major cache `mat` holds
234/// `M[row, c] = coeff` for every action output of `L*(basis[c])` that lands
235/// back in `basis` at index `row`. `per_col[c] = (raw, diag)`, with
236/// `raw = Σ|coeff|` over ALL action outputs (in- and out-of-basis, an upper
237/// bound on the column 1-norm) and `diag` the coefficient of the output
238/// Word equal to the input Word. Both sums run in the canonical term order
239/// of [`LindbladSpec::compute_action_terms`]. The cache is reused by
240/// [`CsrOp`] across every Krylov/Taylor matvec.
241fn build_mf_cols<const C: usize>(
242    spec: &LindbladSpec<C>,
243    basis: &[Word<C>],
244    index: &FxHashMap<Word<C>, u32>,
245) -> (BlockCsr<f64>, PerCol<f64>) {
246    build_block_csr(spec, basis.len(), |c, (s1, s2, lm), rows, vals| {
247        let p = &basis[c];
248        let terms = spec.compute_action_terms(p, s1, s2, lm);
249        let mut raw = 0.0;
250        let mut diag = 0.0;
251        for (w, v) in terms.iter() {
252            raw += v.abs();
253            if w == p {
254                diag = *v;
255            }
256            if let Some(&row) = index.get(w) {
257                rows.push(row);
258                vals.push(*v);
259            }
260        }
261        (raw, diag)
262    })
263}

265/// Per-column **phase-aware** action of the in-basis-restricted orbit-rep
266/// generator `M` at momentum `sector`, plus the `(m, s)`/`μ` selection data
267/// — from ONE action pass over the basis.
268///
269/// Column `c` of the row-major cache receives
270/// `χ_k(g_{cnt_q}) · v_q · |orbit_c| / |orbit_row|` at row `row` for every
271/// action output Pauli `q` of `L*(basis[c])` whose orbit rep `r_q` is in
272/// `basis` at index `row`; outputs whose rep is out of basis are dropped.
273/// Several outputs can share a rep, so a row can hold the same column more
274/// than once; those duplicates are emitted, and later summed, in the
275/// canonical term order of [`LindbladSpec::compute_action_terms`]. This is the expensive part of the orbit-rep dynamics
276/// (`compute_action_terms`, [`Sector::canonicalize_phase`]).
277///
278/// The character-weighted sum runs over the *output* orbit's distinct
279/// members, which makes it the generator in the **summing** convention
280/// `ĉ_r = |orbit_r| · c_r`. Coefficients here are in the *averaged*
281/// convention (`c_r` = the plain coefficient of the rep word, what
282/// `canonicalize_pauli_sum_complex` produces), so each entry carries the
283/// similarity factor `|orbit_c| / |orbit_row|` that converts between
284/// them. It is 1 exactly when both orbits are free — hence the factor is
285/// invisible until an orbit has a non-trivial stabilizer, and cannot be
286/// hoisted out as a global `|G|`.
287///
288/// Unlike [`build_mf_cols`], `per_col[c].0` sums only the retained
289/// in-basis entries — the exact column 1-norm of the restricted `M`, not an
290/// upper bound: several distinct outputs `q` can share one rep, so the
291/// out-of-basis magnitudes are not attributable to a column of `M`. `diag`
292/// accumulates for the same reason.
293fn build_orbit_rep_cols<const C: usize>(
294    spec: &LindbladSpec<C>,
295    basis: &[Word<C>],
296    index: &FxHashMap<Word<C>, u32>,
297    sector: &Sector<'_>,
298) -> (BlockCsr<Complex<f64>>, PerCol<Complex<f64>>) {
299    build_block_csr(spec, basis.len(), |c, (s1, s2, lm), rows, vals| {
300        let r = &basis[c];
301        // A rep that cannot carry the sector has coefficient zero
302        // identically, so its column is empty.
303        let Some(orbit_in) = sector.orbit_size(r) else {
304            return (0.0, Complex::new(0.0, 0.0));
305        };
306        let terms = spec.compute_action_terms(r, s1, s2, lm);
307        let mut raw = 0.0;
308        let mut diag = Complex::new(0.0, 0.0);
309        for (q, v) in terms.iter() {
310            let Some((r_q, phase, orbit_out)) = sector.canonicalize_phase(q) else {
311                continue;
312            };
313            if let Some(&row) = index.get(&r_q) {
314                let val = phase * *v * (orbit_in as f64 / orbit_out as f64);
315                raw += val.norm();
316                if row as usize == c {
317                    diag += val;
318                }
319                rows.push(row);
320                vals.push(val);
321            }
322        }
323        (raw, diag)
324    })
325}

327/// Borrowed CSR view of an in-basis-restricted generator `M`, backed by a
328/// cached action computed once per expm call ([`build_mf_cols`]). `dot`
329/// performs the row-parallel matvec `y = M·x` against the cache; the
330/// remaining `LinearOperator` entry points are unused on the `from_parts` +
331/// single-vector `apply` path.
332///
333/// Borrowed, not owned: `quspin-types` provides a blanket `LinearOperator`
334/// impl for `&T`, so `ExpmOp::from_parts(op, ...)` accepts a `CsrOp` by
335/// value while it keeps borrowing `mat`.
336pub(crate) struct CsrOp<'a, T> {
337    pub(crate) mat: &'a BlockCsr<T>,
338}
339
340impl<T> LinearOperator<T> for CsrOp<'_, T>
341where
342    T: ExpmComputation
343        + Copy
344        + PartialEq
345        + num::Zero
346        + std::ops::AddAssign
347        + std::ops::Mul<Output = T>
348        + Send
349        + Sync,
350{
351    fn dim(&self) -> usize {
352        self.mat.dim
353    }
354
355    fn parallel_hint(&self) -> bool {
356        // `dot` parallelises internally over row ranges, and we drive the
357        // sequential single-vector `apply` path; never let quspin run its
358        // persistent-thread pool on top of our rayon parallelism.
359        false
360    }
361
362    fn dot(&self, overwrite: bool, input: &[T], output: &mut [T]) -> Result<(), QuSpinError> {
363        // Rows in parallel, one range per task; each output entry is summed
364        // by one thread in ascending column order, so the result does not
365        // depend on the thread count.
366        let m = self.mat;
367        output
368            .par_chunks_mut(1 << m.range_shift)
369            .zip(m.ranges.par_iter())
370            .for_each(|(y, r)| {
371                for (i, yi) in y.iter_mut().enumerate() {
372                    let (lo, hi) = (r.offsets[i] as usize, r.offsets[i + 1] as usize);
373                    let mut acc = T::zero();
374                    for (&c, &v) in r.cols[lo..hi].iter().zip(&r.vals[lo..hi]) {
375                        // Skip zero inputs, as the column-wise product did:
376                        // saves the multiply-add for sparse `input`, and
377                        // keeps a non-finite entry of an unused column out
378                        // of `y`.
379                        let xc = input[c as usize];
380                        if xc != T::zero() {
381                            acc += v * xc;
382                        }
383                    }
384                    if overwrite {
385                        *yi = acc;
386                    } else {
387                        *yi += acc;
388                    }
389                }
390            });
391        Ok(())
392    }
393
394    fn trace(&self) -> T {
395        // Computed eagerly by the callers; never reached on the
396        // `from_parts` + single-vector `apply` path.
397        unreachable!("CsrOp::trace not used on the from_parts apply path")
398    }
399
400    fn onenorm(&self, _shift: T) -> <T as ExpmComputation>::Real {
401        unreachable!("CsrOp::onenorm not used on the from_parts apply path")
402    }
403
404    fn dot_transpose(
405        &self,
406        _overwrite: bool,
407        _input: &[T],
408        _output: &mut [T],
409    ) -> Result<(), QuSpinError> {
410        Err(QuSpinError::RuntimeError(
411            "CsrOp: dot_transpose not used on the from_parts apply path".into(),
412        ))
413    }
414
415    fn dot_many(
416        &self,
417        _overwrite: bool,
418        _input: ndarray::ArrayView2<'_, T>,
419        _output: ndarray::ArrayViewMut2<'_, T>,
420    ) -> Result<(), QuSpinError> {
421        Err(QuSpinError::RuntimeError(
422            "CsrOp: dot_many not used on the from_parts apply path".into(),
423        ))
424    }
425
426    fn dot_chunk(
427        &self,
428        _overwrite: bool,
429        _input: &[T],
430        _output_chunk: &mut [T],
431        _row_start: usize,
432    ) -> Result<(), QuSpinError> {
433        Err(QuSpinError::RuntimeError(
434            "CsrOp: dot_chunk not used on the from_parts apply path".into(),
435        ))
436    }
437
438    fn dot_transpose_chunk(
439        &self,
440        _input: &[T],
441        _output: &[<T as ExpmComputation>::Atomic],
442        _rows: std::ops::Range<usize>,
443    ) -> Result<(), QuSpinError> {
444        Err(QuSpinError::RuntimeError(
445            "CsrOp: dot_transpose_chunk not used on the from_parts apply path".into(),
446        ))
447    }
448}

450/// Shared tail of every matrix-free expm: from the cached action `mat` and
451/// its per-column `(raw, diag)` data derive the diagonal shift `μ = tr(M)/n` and a bound on the column 1-norm
452/// of `M − μ·I` (`raw − |diag| + |diag − μ|` per column), pick the Taylor
453/// partition via `select` from `‖dt·(M−μI)‖₁`, and hand everything to
454/// [`quspin_expm::ExpmOp::from_parts`]. Returns `exp(dt · M) · coeffs`.
455///
456/// `select` maps `‖dt·(M−μI)‖₁` to `(m*, s, backward-error tol)`; the two
457/// call sites differ only in that choice.
458fn expm_apply_cached<T>(
459    mat: &BlockCsr<T>,
460    per_col: &PerCol<T>,
461    dt: f64,
462    coeffs: &[T],
463    select: impl FnOnce(f64) -> (u32, u32, f64),
464) -> Vec<T>
465where
466    T: ExpmComputation<Real = f64>
467        + Coeff
468        + PartialEq
469        + num::Zero
470        + AddAssign
471        + Mul<Output = T>
472        + Sub<Output = T>
473        + Div<f64, Output = T>
474        + From<f64>
475        + Sum,
476{
477    let n = mat.dim;
478    let trace: T = per_col.iter().map(|(_, d)| *d).sum();
479    let mu = trace / n as f64;
480    let onenorm = per_col
481        .iter()
482        .map(|&(raw, diag)| raw - diag.mag() + (diag - mu).mag())
483        .fold(0.0_f64, f64::max);
484    let (m_star, s, expm_tol) = select(dt.abs() * onenorm);
485
486    let mut v = coeffs.to_vec();
487    let op = CsrOp { mat };
488    let expm =
489        quspin_expm::ExpmOp::from_parts(op, T::from(dt), mu, s as usize, m_star as usize, expm_tol);
490    expm.apply(ndarray::ArrayViewMut1::from(v.as_mut_slice()))
491        .expect("expm apply");
492    v
493}

495/// Compute `exp(dt · M) · coeffs` for the in-basis-restricted generator
496/// `M`, matrix-free, via `quspin-expm`. Returns a fresh `Vec<f64>` of length
497/// `basis.len()`.
498///
499/// ONE action pass builds the CSR cache `mat` (reused across every matvec)
500/// and, in the same pass, the `(raw, diag)` data the `μ`/1-norm selection
501/// needs; [`expm_apply_cached`] does the rest.
502pub(crate) fn expm_apply_mf<const C: usize>(
503    spec: &LindbladSpec<C>,
504    basis: &[Word<C>],
505    dt: f64,
506    coeffs: &[f64],
507    drop_tol: f64,
508) -> Vec<f64> {
509    if basis.is_empty() {
510        return Vec::new();
511    }
512    let index = build_basis_index(basis);
513    let (mat, per_col) = build_mf_cols(spec, basis, &index);
514
515    // Pick the Taylor backward-error tolerance to match the basis truncation:
516    // when the PC prunes coarsely (drop_tol >= 1e-4) a double-precision exp is
517    // ~10 orders more accurate than the state it acts on, so the relaxed
518    // (tol=1e-6, still >=100x tighter than the cut) table is used — it admits a
519    // lower-degree Taylor polynomial and cuts the SpMV count with no effect on
520    // the truncated result. At tight/zero drop_tol we keep double precision so
521    // the exact-reference paths (orbit-rep / merged) still agree bit-for-bit.
522    expm_apply_cached(&mat, &per_col, dt, coeffs, |t_norm| {
523        if drop_tol >= 1e-4 {
524            let (m, s) = expm::select_ms_loose(t_norm);
525            (m, s, 1e-6)
526        } else {
527            let (m, s) = expm::select_ms(t_norm);
528            (m, s, 1e-12)
529        }
530    })
531}

533/// Compute `exp(dt · M) · coeffs` for the in-basis-restricted **orbit-rep**
534/// generator `M` at momentum `sector`, via `quspin-expm`. Returns a fresh
535/// `Vec<Complex<f64>>` of length `basis.len()`.
536///
537/// The expensive phase-aware action is computed ONCE here (via
538/// [`build_orbit_rep_cols`]) and reused, row-major, across every
539/// Krylov–Taylor matvec, exactly as on the real path.
540pub(crate) fn expm_apply_orbit_rep<const C: usize>(
541    spec: &LindbladSpec<C>,
542    basis: &[Word<C>],
543    sector: &Sector<'_>,
544    dt: f64,
545    coeffs: &[Complex<f64>],
546) -> Vec<Complex<f64>> {
547    if basis.is_empty() {
548        return Vec::new();
549    }
550    let index = build_basis_index(basis);
551    let (mat, per_col) = build_orbit_rep_cols(spec, basis, &index, sector);
552
553    expm_apply_cached(&mat, &per_col, dt, coeffs, |t_norm| {
554        let (m, s) = expm::select_ms(t_norm);
555        (m, s, 1e-12)
556    })
557}

559/// `exp(dt · M) · b` where `M` is the REAL in-basis-restricted generator but
560/// the input vector `b` is complex. Because `M` is real,
561/// `exp(dt·M)·(re + i·im) = exp(dt·M)·re + i·exp(dt·M)·im`, so we split the
562/// complex vector into its real and imaginary parts, run two real
563/// matrix-free applies, and recombine. Used by the test-only full-space
564/// complex reference step.
565#[cfg(test)]
566pub(crate) fn expm_apply_mf_cxvec(
567    spec: &LindbladSpec,
568    basis: &[Word],
569    dt: f64,
570    b: &[Complex<f64>],
571    drop_tol: f64,
572) -> Vec<Complex<f64>> {
573    let n = basis.len();
574    if n == 0 {
575        return Vec::new();
576    }
577    let re: Vec<f64> = b.iter().map(|z| z.re).collect();
578    let im: Vec<f64> = b.iter().map(|z| z.im).collect();
579    let re_out = expm_apply_mf(spec, basis, dt, &re, drop_tol);
580    let im_out = expm_apply_mf(spec, basis, dt, &im, drop_tol);
581    re_out
582        .into_iter()
583        .zip(im_out)
584        .map(|(r, i)| Complex::new(r, i))
585        .collect()
586}
