4//! Momentum sectors of a translation group, and the phase-aware
5//! canonicalization that drives orbit-representative evolution.
6//!
7//! On the orbit-rep path the state lives entirely in **orbit-rep form**
8//! throughout: the basis contains only canonical translation-orbit
9//! representatives and the coefficients are complex (one per rep). The
10//! dynamics `L*` is computed with **phase-aware action** — for each
11//! output Pauli `q`, we canonicalize `q` to its orbit rep `r_q` with
12//! shift counter `cnt_q`, and accumulate
13//! `χ_k(g_{cnt_q}) · v · c_r · |orbit_r| / |orbit_{r_q}|` (where `v` is
14//! the matrix element of `L*` between input rep `r` and output `q`).
15//! [`Sector::canonicalize_phase`] is that step.
16//!
17//! Coefficients are in the **averaged** convention: `c_r` is the plain
18//! coefficient of the rep word, as produced by
19//! `canonicalize_pauli_sum_complex`. The character-weighted action is
20//! naturally the generator in the *summing* convention
21//! `ĉ_r = |orbit_r| · c_r`, which is where the orbit-size ratio comes
22//! from; it is 1 whenever both orbits are free.
23//!
24//! The orbit-rep basis is ~`|G|`× smaller than the full-basis
25//! representation, throughout the entire evolution.
26//!
27//! ## Limitations
28//!
29//! - Callers are responsible for ensuring the input basis is in
30//!   orbit-rep form (i.e. each entry is the canonical representative of
31//!   its translation orbit). Use [`canonicalize_basis_to_rep`] if
32//!   needed.
33//! - A [`Sector`] is fixed for the duration of one
34//!   [`LindbladSpec::pc_step_orbit_rep`](crate::LindbladSpec::pc_step_orbit_rep)
35//!   call. To compute a full site-resolved profile, call it once per
36//!   momentum mode and inverse-Fourier the results.
37
38use crate::Word;
39use num::Complex;
40use ppvm_pauli_sum::symmetry::{CharacterTable, TranslationGroup};
41
42/// A momentum sector of a translation group: the group `G` together with
43/// one integer mode index per generator. The wavenumber along generator
44/// `g` is `2π · k_modes[g] / group.generator_order(g)`; `k_modes = [0, …]`
45/// is the trivial sector.
46///
47/// The two halves are meaningless apart — every phase-aware routine
48/// needs both — so they travel as one value. Construction precomputes the
49/// sector's [`CharacterTable`] (`|G|` entries), so the per-term work in the
50/// hot loops is one canonicalization and a table lookup; build a `Sector`
51/// once per step and pass it by reference.
52#[derive(Clone)]
53pub struct Sector<'a> {
54    group: &'a TranslationGroup,
55    k_modes: &'a [i32],
56    characters: CharacterTable,
57}
58
59impl<'a> Sector<'a> {
60    pub fn new(group: &'a TranslationGroup, k_modes: &'a [i32]) -> Self {
61        let characters = group.character_table(k_modes);
62        Self {
63            group,
64            k_modes,
65            characters,
66        }
67    }
68
69    /// The translation group.
70    pub fn group(&self) -> &'a TranslationGroup {
71        self.group
72    }
73
74    /// The integer momentum mode per generator.
75    pub fn k_modes(&self) -> &'a [i32] {
76        self.k_modes
77    }
78
79    /// Canonicalize `q` to its orbit representative `r_q` and return it
80    /// alongside the character phase `χ_k(g_{cnt_q})` of the group
81    /// element that maps `q` to `r_q`, and the number of **distinct**
82    /// members of that orbit. The phase weights the matrix element of
83    /// `L*` when it is accumulated onto `r_q`; the orbit size converts
84    /// between the two coefficient conventions (see
85    /// [`Self::orbit_size`]).
86    ///
87    /// `None` when `q`'s orbit cannot carry this sector (its stabilizer
88    /// is incompatible with `k`): the coefficient of such a rep is
89    /// identically zero, so the term is dropped.
90    #[inline]
91    pub fn canonicalize_phase<const C: usize>(
92        &self,
93        q: &Word<C>,
94    ) -> Option<(Word<C>, Complex<f64>, usize)> {
95        let (rep, idx, orbit_size) = self
96            .group
97            .canonicalize_in_sector_indexed(q, &self.characters)?;
98        Some((rep, self.characters.value(idx), orbit_size))
99    }
100
101    /// Number of **distinct** members of `w`'s translation orbit, or
102    /// `None` if the orbit cannot carry this sector.
103    ///
104    /// This is the factor between the two orbit-rep coefficient
105    /// conventions: the *averaged* one, in which `c_r` is the plain
106    /// coefficient of the rep word (what `canonicalize_pauli_sum_complex`
107    /// and this crate's public orbit-rep API use), and the *summing* one
108    /// `ĉ_r = |orbit_r| · c_r` (what `momentum_merge_pauli_sum_pair`
109    /// uses). It is `|G|` only for free orbits.
110    #[inline]
111    pub fn orbit_size<const C: usize>(&self, w: &Word<C>) -> Option<usize> {
112        self.group
113            .canonicalize_in_sector_indexed(w, &self.characters)
114            .map(|(_, _, orbit_size)| orbit_size)
115    }
116}
117
118/// Replace each entry of `basis` with its canonical orbit
119/// representative under `group`. Pure rewrite; coefficients are
120/// untouched. Useful to enforce the orbit-rep invariant before calling
121/// [`LindbladSpec::pc_step_orbit_rep`](crate::LindbladSpec::pc_step_orbit_rep).
122///
123/// Does NOT deduplicate — if multiple input entries collapse to the
124/// same rep, both are kept (caller should run a merge afterwards).
125pub fn canonicalize_basis_to_rep<const C: usize>(basis: &mut [Word<C>], group: &TranslationGroup) {
126    for w in basis.iter_mut() {
127        *w = group.canonicalize(w);
128    }
129}
