4//! Precompiled Lindbladian: construction and the single-Pauli `L*` kernel.
5
6use crate::Error;
7use crate::algebra::{
8    PauliTerm, anti_commutes, comm_product, pauli_mul, phase_factor, precompute_adag_b,
9};
10use crate::kossakowski;
11use crate::word::{W_CHUNKS, Word, check_width, parse_pauli_string, word_hash, word_support};
12use fxhash::FxHashMap;
13use num::Complex;

15/// Parsed Hamiltonian term.
16#[derive(Clone)]
17struct HTerm<const C: usize = W_CHUNKS> {
18    word: Word<C>,
19    coeff: f64,
20}
21
22/// One entry of the dissipator. `HermitianPauli` and `General` are the
23/// jump-operator form (`K` diagonal); `Kossakowski` is one compiled pair of
24/// the general form. See [`crate::kossakowski`].
25enum JumpKind<const C: usize> {
26    HermitianPauli {
27        word: Word<C>,
28        rate: f64,
29    },
30    General {
31        terms: Vec<PauliTerm<C>>,         // L = Σ_a λ_a P_a
32        dagger_dagger: Vec<PauliTerm<C>>, // L†L = Σ_c μ_c P_c  (μ_c ∈ ℝ)
33        rate: f64,
34    },
35    Kossakowski(kossakowski::Pair<C>),
36}
37
38/// Union of `index[q]` for each `q ∈ p_support`, deduped.
39#[inline]
40fn candidate_terms(p_support: &[u32], index: &[Vec<u32>], scratch: &mut Vec<u32>) {
41    scratch.clear();
42    for &q in p_support {
43        scratch.extend_from_slice(&index[q as usize]);
44    }
45    scratch.sort_unstable();
46    scratch.dedup();
47}

49/// Precompiled Lindbladian. Constructed once from string-form Hamiltonian
50/// terms + jump operators; reused across many calls to [`Self::action`],
51/// [`Self::leakage`], [`Self::generator`]. `L*(p)` is recomputed on every
52/// call rather than cached: for sparse-local Hamiltonians a per-word cache
53/// costs more than the recompute (hash lookup ≳ recompute) and its several
54/// KB per cached word dominate memory at large basis sizes.
55pub struct LindbladSpec<const C: usize = W_CHUNKS> {
56    n_qubits: usize,
57    h_terms: Vec<HTerm<C>>,
58    j_kinds: Vec<JumpKind<C>>,
59    /// `h_support[q]` = indices of Hamiltonian terms acting on qubit `q`.
60    h_support: Vec<Vec<u32>>,
61    /// `j_support[q]` = indices of jumps whose support contains qubit `q`.
62    j_support: Vec<Vec<u32>>,
63}
64
65/// User-facing description of one jump operator: a complex Pauli linear
66/// combination together with its rate.
67#[derive(Clone, Debug)]
68pub struct JumpInput {
69    /// `(pauli_string, λ)` pairs forming `L_k = Σ_a λ_a P_a`.
70    pub lincomb: Vec<(String, Complex<f64>)>,
71    /// Non-negative GKSL rate `γ_k`.
72    pub rate: f64,
73}
74
75impl<const C: usize> LindbladSpec<C> {
76    /// Construct a Lindbladian spec from Hamiltonian terms and jump operators.
77    ///
78    /// `h_terms` are `(pauli_string, coefficient)` pairs forming the Hermitian
79    /// Hamiltonian. Each jump operator is a complex Pauli linear combination;
80    /// a length-1 jump with imaginary part `0` is routed to the Hermitian-Pauli
81    /// fast path (with rate scaled by the squared real coefficient).
82    pub fn new(
83        n_qubits: usize,
84        h_terms: &[(String, f64)],
85        jumps: &[JumpInput],
86    ) -> Result<Self, Error> {
87        check_width::<C>(n_qubits)?;
88
89        let mut h_parsed: Vec<HTerm<C>> = Vec::with_capacity(h_terms.len());
90        let mut h_support_idx: Vec<Vec<u32>> = vec![Vec::new(); n_qubits];
91        for (i, (s, c)) in h_terms.iter().enumerate() {
92            let (word, support) = parse_pauli_string(s, n_qubits)?;
93            for q in support {
94                h_support_idx[q as usize].push(i as u32);
95            }
96            h_parsed.push(HTerm { word, coeff: *c });
97        }
98
99        let mut j_kinds: Vec<JumpKind<C>> = Vec::with_capacity(jumps.len());
100        let mut j_support_idx: Vec<Vec<u32>> = vec![Vec::new(); n_qubits];
101        for (k, jump) in jumps.iter().enumerate() {
102            if jump.rate < 0.0 {
103                return Err(Error::NegativeRate {
104                    index: k,
105                    rate: jump.rate,
106                });
107            }
108            if jump.lincomb.is_empty() {
109                return Err(Error::EmptyLincomb { index: k });
110            }
111
112            // Fast path: single-term, purely real → Hermitian Pauli.
113            if jump.lincomb.len() == 1 && jump.lincomb[0].1.im == 0.0 {
114                let (s, c) = &jump.lincomb[0];
115                let (word, support) = parse_pauli_string(s, n_qubits)?;
116                for q in support {
117                    j_support_idx[q as usize].push(k as u32);
118                }
119                j_kinds.push(JumpKind::HermitianPauli {
120                    word,
121                    rate: jump.rate * c.re * c.re,
122                });
123                continue;
124            }
125
126            // General path: parse all terms, precompute L†L, record union support.
127            let mut terms: Vec<PauliTerm<C>> = Vec::with_capacity(jump.lincomb.len());
128            let mut union_support: std::collections::BTreeSet<u32> =
129                std::collections::BTreeSet::new();
130            for (s, c) in &jump.lincomb {
131                let (word, support) = parse_pauli_string(s, n_qubits)?;
132                for q in &support {
133                    union_support.insert(*q);
134                }
135                terms.push(PauliTerm { word, coeff: *c });
136            }
137            for q in union_support {
138                j_support_idx[q as usize].push(k as u32);
139            }
140            let dagger_dagger = precompute_adag_b(&terms, &terms);
141            j_kinds.push(JumpKind::General {
142                terms,
143                dagger_dagger,
144                rate: jump.rate,
145            });
146        }
147
148        Ok(Self {
149            n_qubits,
150            h_terms: h_parsed,
151            j_kinds,
152            h_support: h_support_idx,
153            j_support: j_support_idx,
154        })
155    }
156
157    /// Add a Kossakowski-form dissipator
158    /// `D*(O) = Σ_{n,m} K_nm ( A_n† O A_m − ½ {A_n† A_m, O} )`.
159    ///
160    /// `ops` lists the operators `A_n` as complex Pauli linear combinations;
161    /// `k` is the `n_ops × n_ops` Hermitian pair matrix. Contributions are
162    /// added to any jumps already present. See [`crate::kossakowski`] for
163    /// how each `(n, m)` entry is compiled.
164    pub fn add_kossakowski(
165        &mut self,
166        ops: &[Vec<(String, Complex<f64>)>],
167        k: &[Vec<Complex<f64>>],
168    ) -> Result<(), Error> {
169        for (pair, support) in kossakowski::compile(ops, k, self.n_qubits)? {
170            let idx = self.j_kinds.len() as u32;
171            self.j_kinds.push(JumpKind::Kossakowski(pair));
172            for q in support {
173                self.j_support[q as usize].push(idx);
174            }
175        }
176        Ok(())
177    }
178
179    pub fn n_qubits(&self) -> usize {
180        self.n_qubits
181    }
182
183    pub fn num_h_terms(&self) -> usize {
184        self.h_terms.len()
185    }
186
187    pub fn num_jump_terms(&self) -> usize {
188        self.j_kinds.len()
189    }

191    /// Apply `L*` to a single Pauli string `p`. Returns the output Pauli
192    /// strings and their real coefficients (zero entries omitted).
193    pub fn action(&self, p: &Word<C>) -> Vec<(Word<C>, f64)> {
194        let mut out: FxHashMap<Word<C>, f64> = FxHashMap::default();
195        let mut s1 = Vec::new();
196        let mut s2 = Vec::new();
197        self.accumulate_action(p, 1.0, &mut out, &mut s1, &mut s2);
198        out.into_iter().filter(|(_, c)| *c != 0.0).collect()
199    }
200
201    /// Compute the unscaled list of `(output, coefficient)` pairs that
202    /// `L*(p)` contributes (without the input coefficient), in a canonical
203    /// order: ascending [`word_hash`], ties by [`Word`] order.
204    ///
205    /// The order matters because every caller sums or inserts in it. The
206    /// drain order of `scratch_local` would depend on its capacity, which
207    /// carries over from earlier calls on the same rayon job and hence
208    /// depends on the thread count once one string has enough outputs to
209    /// grow the map.
210    pub(crate) fn compute_action_terms(
211        &self,
212        p: &Word<C>,
213        scratch_support: &mut Vec<u32>,
214        scratch_cands: &mut Vec<u32>,
215        scratch_local: &mut FxHashMap<Word<C>, Complex<f64>>,
216    ) -> Vec<(Word<C>, f64)> {
217        word_support(p, scratch_support);
218        let zero = Complex::new(0.0, 0.0);
219        scratch_local.clear();
220        let local = scratch_local;
221
222        // ── i [H, p] ─────────────────────────────────────────────────
223        candidate_terms(scratch_support, &self.h_support, scratch_cands);
224        for &i in scratch_cands.iter() {
225            let h = &self.h_terms[i as usize];
226            let (r, eps) = comm_product(&h.word, p);
227            if eps != 0.0 {
228                *local.entry(r).or_insert(zero) += Complex::new(h.coeff * eps, 0.0);
229            }
230        }
231
232        // ── dissipator ───────────────────────────────────────────────
233        candidate_terms(scratch_support, &self.j_support, scratch_cands);
234        for &k in scratch_cands.iter() {
235            match &self.j_kinds[k as usize] {
236                JumpKind::HermitianPauli { word, rate } => {
237                    if anti_commutes(word, p) {
238                        *local.entry(*p).or_insert(zero) += Complex::new(-2.0 * *rate, 0.0);
239                    }
240                }
241                JumpKind::General {
242                    terms,
243                    dagger_dagger,
244                    rate,
245                } => {
246                    let rate_c = Complex::new(*rate, 0.0);
247                    // Sandwich: γ Σ_{a,b} λ_a* λ_b P_a p P_b.
248                    for a in terms {
249                        let (r_ap, phi1) = pauli_mul(&a.word, p);
250                        for b in terms {
251                            let (s, phi2) = pauli_mul(&r_ap, &b.word);
252                            let coeff =
253                                a.coeff.conj() * b.coeff * phase_factor(phi1 + phi2) * rate_c;
254                            *local.entry(s).or_insert(zero) += coeff;
255                        }
256                    }
257                    // -1/2 γ {L†L, p}. For Hermitian Pauli P_c and Pauli p,
258                    // {P_c, p} = 2·sign·R if they commute (P_c·p = sign·R),
259                    //         = 0          if they anti-commute.
260                    for c_term in dagger_dagger {
261                        let (r, phase) = pauli_mul(&c_term.word, p);
262                        if phase & 1 == 0 {
263                            let sign = if phase == 0 { 1.0 } else { -1.0 };
264                            let coeff = -c_term.coeff * rate_c * Complex::new(sign, 0.0);
265                            *local.entry(r).or_insert(zero) += coeff;
266                        }
267                    }
268                }
269                JumpKind::Kossakowski(pair) => pair.accumulate(p, local),
270            }
271        }
272
273        // L* preserves Hermiticity; imaginary parts must cancel to FP noise.
274        // `drain()` empties `scratch_local` so its allocation can be reused
275        // by the next call on the same thread (`Vec` keeps capacity).
276        let mut terms: Vec<(Word<C>, f64)> = local
277            .drain()
278            .filter_map(|(w, c)| {
279                debug_assert!(
280                    c.im.abs() < 1e-9,
281                    "L*(p) produced non-real coefficient {c}; bug in dissipator"
282                );
283                if c.re == 0.0 { None } else { Some((w, c.re)) }
284            })
285            .collect();
286        // Output words are distinct, so any sort gives the same order.
287        terms.sort_unstable_by(|(a, _), (b, _)| {
288            word_hash(a).cmp(&word_hash(b)).then_with(|| a.cmp(b))
289        });
290        terms
291    }
292
293    /// Accumulate `scale · L*(p)` into `out`.
294    fn accumulate_action(
295        &self,
296        p: &Word<C>,
297        scale: f64,
298        out: &mut FxHashMap<Word<C>, f64>,
299        scratch_support: &mut Vec<u32>,
300        scratch_cands: &mut Vec<u32>,
301    ) {
302        let mut scratch_local = FxHashMap::default();
303        let terms =
304            self.compute_action_terms(p, scratch_support, scratch_cands, &mut scratch_local);
305        for (w, c) in terms.iter() {
306            *out.entry(*w).or_insert(0.0) += scale * c;
307        }
308    }
