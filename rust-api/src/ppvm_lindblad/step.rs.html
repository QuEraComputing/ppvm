14/// Per-phase timing breakdown (microseconds) returned by
15/// [`LindbladSpec::pc_step_timed`].
16#[derive(Default, Clone, Copy, Debug)]
17pub struct PcStepTimings {
18    pub leakage1_us: u64,
19    pub expand1_us: u64,
20    pub expm1_us: u64,
21    pub leakage2_us: u64,
22    pub expand2_us: u64,
23    pub expm2_us: u64,
24}
25
26impl PcStepTimings {
27    pub fn total_us(&self) -> u64 {
28        self.leakage1_us
29            + self.expand1_us
30            + self.expm1_us
31            + self.leakage2_us
32            + self.expand2_us
33            + self.expm2_us
34    }
35}
36
37/// Clock for one `pc_step` phase. Disarmed (`None`) on the untimed path, so
38/// [`LindbladSpec::pc_step`] pays no `Instant` syscalls.
39struct Phase(Option<Instant>);
40
41impl Phase {
42    fn start(timed: bool) -> Self {
43        Self(timed.then(Instant::now))
44    }
45
46    fn stop(self, slot: &mut u64) {
47        if let Some(t0) = self.0 {
48            *slot = t0.elapsed().as_micros() as u64;
49        }
50    }
51}
52
53impl<const C: usize> LindbladSpec<C> {
54    /// One predictor-corrector step `O ← exp(dt·L*) O` in the adaptive
55    /// real-coefficient Pauli basis: first-hop leakage admission, predictor
56    /// exponential, second-hop admission from the predicted state, corrector
57    /// exponential from the saved pre-step state, then truncation (prune +
58    /// rank cap) per [`PcStepConfig`]. Exact in `dt` within the working
59    /// basis — the only error is basis truncation.
60    ///
61    /// When the second hop admits no string (in particular when the first
62    /// hop already filled the admission room `admit_basis − |basis|`, the
63    /// usual case once the basis has reached `max_basis`), the corrector
64    /// would repeat the predictor exactly; the second leakage pass (if
65    /// `room = 0`) and the corrector exponential are then skipped, with
66    /// bit-identical results. The step's timings report 0 for skipped
67    /// phases.
68    ///
69    /// `protected` words are never dropped. All tuning knobs live in `cfg`.
70    pub fn pc_step(
71        &self,
72        basis: &mut Vec<Word<C>>,
73        coeffs: &mut Vec<f64>,
74        dt: f64,
75        protected: &[Word<C>],
76        cfg: &PcStepConfig,
77    ) -> Result<(), Error> {
78        self.run_in_pool(cfg, |this| {
79            this.pc_step_inner(basis, coeffs, dt, protected, cfg, false)
80                .map(|_| ())
81        })
82    }
83
84    /// Same as [`Self::pc_step`] but also returns a per-phase timing
85    /// breakdown (microseconds), for profiling parallel scaling and hot
86    /// spots.
87    pub fn pc_step_timed(
88        &self,
89        basis: &mut Vec<Word<C>>,
90        coeffs: &mut Vec<f64>,
91        dt: f64,
92        protected: &[Word<C>],
93        cfg: &PcStepConfig,
94    ) -> Result<PcStepTimings, Error> {
95        self.run_in_pool(cfg, |this| {
96            this.pc_step_inner(basis, coeffs, dt, protected, cfg, true)
97        })
98    }
99
100    fn run_in_pool<R: Send>(
101        &self,
102        cfg: &PcStepConfig,
103        f: impl FnOnce(&Self) -> Result<R, Error> + Send,
104    ) -> Result<R, Error> {
105        if let Some(n) = cfg.num_threads {
106            let pool = rayon::ThreadPoolBuilder::new()
107                .num_threads(n)
108                .build()
109                .map_err(|e| Error::Internal(format!("rayon pool build: {e}")))?;
110            pool.install(|| f(self))
111        } else {
112            f(self)
113        }
114    }
115
116    fn pc_step_inner(
117        &self,
118        basis: &mut Vec<Word<C>>,
119        coeffs: &mut Vec<f64>,
120        dt: f64,
121        protected: &[Word<C>],
122        cfg: &PcStepConfig,
123        timed: bool,
124    ) -> Result<PcStepTimings, Error> {
125        let PcStepConfig {
126            max_basis,
127            admit_basis,
128            drop_tol,
129            tau_add,
130            ..
131        } = *cfg;
132        // Admission bound: enrichment may grow the live basis to `admit`
133        // >= `max_basis`; the final `cap_basis` then keeps the top-
134        // `max_basis` strings by evolved |coeff| over the whole union
135        // (retained + admitted) — rank displacement. With `admit_basis =
136        // None` admission is bounded by `max_basis` itself, `cap_basis` is
137        // a no-op, and membership turnover requires `drop_tol > 0`.
138        let admit = admit_basis.unwrap_or(max_basis).max(max_basis);
139        let tau_add = tau_add.unwrap_or(0.0);
140        let mut t = PcStepTimings::default();
141
142        // 1. First-hop expansion. After this, `coeffs` contains the pre-step
143        // coefficients followed by zeros for the newly-added leakage strings.
144        // We rely on `coeffs` itself as the pre-step buffer for the corrector
145        // — no `.clone()` is needed because `expm_step` only borrows it.
146        let p = Phase::start(timed);
147        let leak = self.leakage_with_prune(basis, coeffs, protected, admit, tau_add)?;
148        p.stop(&mut t.leakage1_us);
149
150        let p = Phase::start(timed);
151        add_leakage_capped(basis, coeffs, leak, admit);
152        p.stop(&mut t.expand1_us);
153
154        // 2. Predictor: `expm_step` reads `coeffs` immutably and returns a
155        // new owned vector with the predicted state.
156        let p = Phase::start(timed);
157        let coeffs_predict = self.expm_step(basis, dt, coeffs, drop_tol);
158        p.stop(&mut t.expm1_us);
159
160        // 3. Second-hop expansion from the predicted state. Extend `coeffs`
161        // with zeros for any newly-added second-hop strings so it remains a
162        // valid input (pre-step state) for the corrector. Once the basis is
163        // full, the first hop usually fills the whole admission room; with
164        // `room = 0` the second hop can admit nothing, so its leakage pass
165        // is skipped.
166        let n_predict = basis.len();
167        if admit > n_predict {
168            let p = Phase::start(timed);
169            let leak2 =
170                self.leakage_with_prune(basis, &coeffs_predict, protected, admit, tau_add)?;
171            p.stop(&mut t.leakage2_us);
172
173            let p = Phase::start(timed);
174            add_leakage_capped(basis, coeffs, leak2, admit);
175            p.stop(&mut t.expand2_us);
176        }
177
178        // 4. Corrector: redo from pre-step state on the doubly-enlarged basis.
179        // If the second hop admitted nothing, the corrector would repeat the
180        // predictor's computation exactly, so the predicted state is kept.
181        if basis.len() == n_predict {
182            *coeffs = coeffs_predict;
183        } else {
184            drop(coeffs_predict);
185            let p = Phase::start(timed);
186            *coeffs = self.expm_step(basis, dt, coeffs, drop_tol);
187            p.stop(&mut t.expm2_us);
188        }
189
190        // 5. Prune basis entries below `drop_tol` (protected words never dropped).
191        prune_basis(basis, coeffs, drop_tol, protected);
192        cap_basis(basis, coeffs, max_basis, protected);
193        Ok(t)
194    }

196    /// Compute `exp(dt · M) · b` for the in-basis-restricted generator
197    /// `M`, matrix-free, via `quspin-expm` (see [`crate::mf_expm`]).
198    fn expm_step(&self, basis: &[Word<C>], dt: f64, b: &[f64], drop_tol: f64) -> Vec<f64> {
199        mf_expm::expm_apply_mf(self, basis, dt, b, drop_tol)
200    }
201
202    /// One predictor-corrector step in **orbit-rep form** at momentum
203    /// `sector`: the same five phases as [`Self::pc_step`], but the basis
204    /// holds only canonical translation-orbit representatives, the
205    /// coefficients are complex, and the `L*` action is phase-aware (see
206    /// [`crate::sector`]). The basis stays ~`|G|`× smaller than the
207    /// equivalent full-basis complex evolution, every step.
208    ///
209    /// `max_basis` is a hard rank cap on the live orbit-rep basis:
210    /// enrichment adds at most `admit − basis.len()` of the largest
211    /// leakage reps, the leakage map is capped to the same room, and the
212    /// post-step basis is trimmed to the top-`max_basis` reps by `|c|`.
213    /// Pass a large value (e.g. `usize::MAX`) for the near-exact,
214    /// uncapped case. `drop_tol` additionally prunes by magnitude.
215    /// `protected` reps are never dropped.
216    ///
217    /// `basis` is assumed to contain only canonical orbit
218    /// representatives. If not, call
219    /// [`canonicalize_basis_to_rep`](crate::canonicalize_basis_to_rep)
220    /// first.
221    ///
222    /// Honours `cfg.num_threads` the same way [`Self::pc_step`] does.
223    pub fn pc_step_orbit_rep(
224        &self,
225        basis: &mut Vec<Word<C>>,
226        coeffs: &mut Vec<Complex<f64>>,
227        dt: f64,
228        protected: &[Word<C>],
229        sector: &Sector<'_>,
230        cfg: &PcStepConfig,
231    ) -> Result<(), Error> {
232        self.run_in_pool(cfg, |this| {
233            this.pc_step_orbit_rep_inner(basis, coeffs, dt, protected, sector, cfg)
234        })
235    }
236
237    fn pc_step_orbit_rep_inner(
238        &self,
239        basis: &mut Vec<Word<C>>,
240        coeffs: &mut Vec<Complex<f64>>,
241        dt: f64,
242        protected: &[Word<C>],
243        sector: &Sector<'_>,
244        cfg: &PcStepConfig,
245    ) -> Result<(), Error> {
246        let PcStepConfig {
247            max_basis,
248            admit_basis,
249            drop_tol,
250            tau_add,
251            ..
252        } = *cfg;
253        // Admission bound, mirroring `pc_step_inner`: enrichment may grow
254        // the live basis to `admit` >= `max_basis`; the final `cap_basis`
255        // keeps the top-`max_basis` reps by evolved |coeff| over the whole
256        // union (rank displacement). With `admit_basis = None` admission is
257        // bounded by `max_basis` itself and membership turnover requires
258        // `drop_tol > 0`.
259        let admit = admit_basis.unwrap_or(max_basis).max(max_basis);
260        let tau_add = tau_add.unwrap_or(0.0);
261
262        // 1. First-hop phase-aware leakage.
263        let mut leak = self.leakage_orbit_rep(basis, coeffs, protected, sector, admit)?;
264        if tau_add > 0.0 {
265            leak.retain(|(_, c)| c.norm() > tau_add);
266        }
267        add_leakage_capped(basis, coeffs, leak, admit);
268
269        // 2. Predictor: the phase-aware action is built once and reused
270        //    across every matvec.
271        let coeffs_predict = mf_expm::expm_apply_orbit_rep(self, basis, sector, dt, coeffs);
272
273        // 3. Second-hop leakage from the predicted state, skipped when the
274        //    first hop left no admission room (see `pc_step_inner`).
275        let n_predict = basis.len();
276        if admit > n_predict {
277            let mut leak2 =
278                self.leakage_orbit_rep(basis, &coeffs_predict, protected, sector, admit)?;
279            if tau_add > 0.0 {
280                leak2.retain(|(_, c)| c.norm() > tau_add);
281            }
282            add_leakage_capped(basis, coeffs, leak2, admit);
283        }
284
285        // 4. Corrector: redo from the pre-step state if the basis grew;
286        //    otherwise it would reproduce the predictor exactly.
287        if basis.len() == n_predict {
288            *coeffs = coeffs_predict;
289        } else {
290            drop(coeffs_predict);
291            *coeffs = mf_expm::expm_apply_orbit_rep(self, basis, sector, dt, coeffs);
292        }
293
294        // 5. Prune by magnitude, then rank-cap to max_basis.
295        prune_basis(basis, coeffs, drop_tol, protected);
296        cap_basis(basis, coeffs, max_basis, protected);
297        Ok(())
298    }
