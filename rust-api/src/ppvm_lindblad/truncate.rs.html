4//! Basis truncation and enrichment: the magnitude prune, the rank cap,
5//! and the capped leakage admission shared by every `pc_step` variant.
6//!
7//! All three are generic over the coefficient scalar ([`Coeff`]), so the
8//! real adaptive path and the complex orbit-rep path run the same code.
9
10use crate::scalar::Coeff;
11use crate::word::Word;
12use fxhash::{FxHashMap, FxHashSet};
13
14/// Basis indices in descending coefficient magnitude. Leakage
15/// accumulation walks the basis in this order so the running room-cap
16/// keeps the entries most likely to be the true largest contributors.
17pub(crate) fn order_by_desc_mag<T: Coeff>(coeffs: &[T]) -> Vec<usize> {
18    let mut order: Vec<usize> = (0..coeffs.len()).collect();
19    order.sort_by(|&a, &b| desc_by_mag(coeffs[a], coeffs[b]));
20    order
21}

23/// Keep only the `room` largest-magnitude entries of a live leakage
24/// candidate map — `room` being the number of strings we could actually
25/// admit to the basis, so there is no point tracking more. Applied after
26/// each accumulation chunk.
27pub(crate) fn cap_map_to_room<const C: usize, T: Coeff>(
28    merged: &mut FxHashMap<Word<C>, T>,
29    room: usize,
30) {
31    if merged.len() <= room {
32        return;
33    }
34    if room == 0 {
35        merged.clear();
36        return;
37    }
38    let mut mags: Vec<f64> = merged.values().map(|v| v.mag()).collect();
39    let mut keep = top_k_keeper(&mut mags, room);
40    merged.retain(|_, v| keep(v.mag()));
41}

43/// Compact `basis` / `coeffs` in place: drop entries whose coefficient
44/// magnitude is below `drop_tol` unless the word appears in `protected`.
45/// No-op when `drop_tol ≤ 0`.
46pub(crate) fn prune_basis<const C: usize, T: Coeff>(
47    basis: &mut Vec<Word<C>>,
48    coeffs: &mut Vec<T>,
49    drop_tol: f64,
50    protected: &[Word<C>],
51) {
52    if drop_tol <= 0.0 {
53        return;
54    }
55    debug_assert_eq!(basis.len(), coeffs.len());
56    let protected_set: FxHashSet<&Word<C>> = protected.iter().collect();
57    retain_in_place(basis, coeffs, |w, c| {
58        c.mag() >= drop_tol || protected_set.contains(w)
59    });
60}

62/// Global max-basis cap (PauliStrings.jl-style top-M trim): keep only the
63/// `max_basis` largest-magnitude terms (protected strings always kept),
64/// dropping the rest. Rank-based total-basis bound; dual of `drop_tol`.
65/// A `max_basis` large enough to cover the whole basis is a no-op.
66pub(crate) fn cap_basis<const C: usize, T: Coeff>(
67    basis: &mut Vec<Word<C>>,
68    coeffs: &mut Vec<T>,
69    max_basis: usize,
70    protected: &[Word<C>],
71) {
72    if basis.len() <= max_basis {
73        return;
74    }
75    let protected_set: FxHashSet<&Word<C>> = protected.iter().collect();
76    let n_prot = basis.iter().filter(|w| protected_set.contains(w)).count();
77    let slots = max_basis.saturating_sub(n_prot);
78    let mut mags: Vec<f64> = basis
79        .iter()
80        .zip(coeffs.iter())
81        .filter(|(w, _)| !protected_set.contains(w))
82        .map(|(_, c)| c.mag())
83        .collect();
84    if slots >= mags.len() {
85        return;
86    }
87    let mut keep = top_k_keeper(&mut mags, slots);
88    retain_in_place(basis, coeffs, |w, c| {
89        protected_set.contains(w) || keep(c.mag())
90    });
91}

93/// Add the largest leakage strings to the basis, up to the available room
94/// `room = max_basis − basis.len()` — so the in-step basis (hence the
95/// expm/leakage peak memory) never exceeds `max_basis`. New strings get
96/// coefficient 0; the surrounding expm fills them. No magnitude filter: the
97/// top-`room` by `|leakage|` are added (a large `max_basis` adds them all).
98pub(crate) fn add_leakage_capped<const C: usize, T: Coeff>(
99    basis: &mut Vec<Word<C>>,
100    coeffs: &mut Vec<T>,
101    mut leak: Vec<(Word<C>, T)>,
102    max_basis: usize,
103) {
104    let room = max_basis.saturating_sub(basis.len());
105    if leak.len() > room {
106        if room > 0 {
107            leak.select_nth_unstable_by(room - 1, |a, b| desc_by_mag(a.1, b.1));
108        }
109        leak.truncate(room);
110    }
111    for (w, _) in leak {
112        basis.push(w);
113        coeffs.push(T::zero());
114    }
115}

117/// Keep the `basis`/`coeffs` entries satisfying `keep`, preserving order,
118/// by swapping survivors down and truncating.
119fn retain_in_place<const C: usize, T>(
120    basis: &mut Vec<Word<C>>,
121    coeffs: &mut Vec<T>,
122    mut keep: impl FnMut(&Word<C>, &T) -> bool,
123) {
124    let mut write = 0;
125    for read in 0..basis.len() {
126        if keep(&basis[read], &coeffs[read]) {
127            if write != read {
128                basis.swap(write, read);
129                coeffs.swap(write, read);
130            }
131            write += 1;
132        }
133    }
134    basis.truncate(write);
135    coeffs.truncate(write);
136}

138/// Predicate that, fed every magnitude of `mags` once, accepts exactly the
139/// `k` largest (`k < mags.len()`): all above the cutoff, then ties in visit
140/// order. Reorders `mags`.
141fn top_k_keeper(mags: &mut [f64], k: usize) -> impl FnMut(f64) -> bool {
142    let (cutoff, mut ties_left) = if k == 0 {
143        (f64::INFINITY, 0)
144    } else {
145        let cutoff = nth_largest(mags, k - 1);
146        let n_above = mags.iter().filter(|&&m| m > cutoff).count();
147        (cutoff, k.saturating_sub(n_above))
148    };
149    move |m| {
150        if m > cutoff {
151            return true;
152        }
153        let tie = m == cutoff && ties_left > 0;
154        ties_left -= tie as usize;
155        tie
156    }
157}

159/// The `k`-th largest element of `mags` (0-indexed), via a partial sort.
160/// Reorders `mags`. Panics if `k >= mags.len()`.
161fn nth_largest(mags: &mut [f64], k: usize) -> f64 {
162    mags.select_nth_unstable_by(k, |a, b| {
163        b.partial_cmp(a).unwrap_or(std::cmp::Ordering::Equal)
164    });
165    mags[k]
166}

168/// Descending comparison by magnitude, NaN-tolerant.
169fn desc_by_mag<T: Coeff>(a: T, b: T) -> std::cmp::Ordering {
170    b.mag()
171        .partial_cmp(&a.mag())
172        .unwrap_or(std::cmp::Ordering::Equal)
173}

175#[cfg(test)]
176mod tests {
177    use super::*;
178    use crate::word::word_from_codes;
179
180    /// `n` distinct 4-qubit words.
181    fn words(n: usize) -> Vec<Word> {
182        (0..n)
183            .map(|i| {
184                let codes: Vec<u8> = (0..4).map(|q| ((i >> (2 * q)) & 3) as u8).collect();
185                word_from_codes(&codes).unwrap()
186            })
187            .collect()
188    }
189
190    #[test]
191    fn cap_basis_is_a_hard_cap_under_ties() {
192        let mut basis = words(10);
193        // One large entry, nine tied zeros (fresh leakage admissions).
194        let mut coeffs = vec![0.0; 10];
195        coeffs[3] = 1.0;
196        cap_basis(&mut basis, &mut coeffs, 4, &[]);
197        assert_eq!(basis.len(), 4);
198        assert_eq!(coeffs, vec![0.0, 0.0, 0.0, 1.0]);
199    }
200
201    #[test]
202    fn cap_basis_keeps_protected_beyond_slots() {
203        let all = words(6);
204        let mut basis = all.clone();
205        let mut coeffs = vec![0.5; 6];
206        let protected = [all[5]];
207        cap_basis(&mut basis, &mut coeffs, 3, &protected);
208        assert_eq!(basis, vec![all[0], all[1], all[5]]);
209    }
210
211    #[test]
212    fn cap_map_to_room_keeps_exactly_room_under_ties() {
213        let mut merged: FxHashMap<Word, f64> = words(10).into_iter().map(|w| (w, 0.25)).collect();
214        cap_map_to_room(&mut merged, 3);
215        assert_eq!(merged.len(), 3);
216    }
217
218    #[test]
219    fn cap_map_to_room_keeps_the_largest() {
220        let ws = words(5);
221        let mut merged: FxHashMap<Word, f64> = ws
222            .iter()
223            .zip([1.0, -5.0, 3.0, 0.5, -4.0])
224            .map(|(w, c)| (*w, c))
225            .collect();
226        cap_map_to_room(&mut merged, 2);
227        let mut kept: Vec<f64> = merged.values().copied().collect();
228        kept.sort_by(f64::total_cmp);
229        assert_eq!(kept, vec![-5.0, -4.0]);
230    }
