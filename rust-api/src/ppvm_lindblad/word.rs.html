11/// Pauli-word storage chunk: `u64` on 64-bit targets, `u32` elsewhere
12/// (`bitvec` implements `BitStore` for `u64` only on 64-bit targets, so
13/// e.g. wasm32 builds use four 32-bit chunks instead of two 64-bit ones).
14#[cfg(target_pointer_width = "64")]
15pub(crate) type Chunk = u64;
16#[cfg(not(target_pointer_width = "64"))]
17pub(crate) type Chunk = u32;
18
19/// Bits per storage chunk.
20pub const CHUNK_BITS: usize = Chunk::BITS as usize;
21
22/// Chunks for a 128-qubit word: the default width, and the only one
23/// instantiated for registers of at most 128 qubits.
24pub const W_CHUNKS: usize = 128 / CHUNK_BITS;
25
26/// Chunk counts of the word widths the Python layer instantiates, narrowest
27/// first: 128, 256 and 512 qubits. [`chunks_for`] picks among them.
28pub const WIDTHS: [usize; 3] = [128 / CHUNK_BITS, 256 / CHUNK_BITS, 512 / CHUNK_BITS];
29
30/// Largest register any instantiated width supports.
31pub const MAX_SUPPORTED_QUBITS: usize = 512;
32
33/// Maximum number of qubits of the default-width [`Word`] (128).
34pub const MAX_QUBITS: usize = max_qubits::<W_CHUNKS>();
35
36/// Capacity of a `C`-chunk [`Word`], in qubits.
37pub const fn max_qubits<const C: usize>() -> usize {
38    C * CHUNK_BITS
39}
40
41/// The narrowest entry of [`WIDTHS`] that holds `n_qubits`, or `None` above
42/// [`MAX_SUPPORTED_QUBITS`].
43pub const fn chunks_for(n_qubits: usize) -> Option<usize> {
44    let mut i = 0;
45    while i < WIDTHS.len() {
46        if n_qubits <= WIDTHS[i] * CHUNK_BITS {
47            return Some(WIDTHS[i]);
48        }
49        i += 1;
50    }
51    None
52}

54/// The Pauli-word storage type used throughout this crate, `C` chunks wide.
55///
56/// The crate is const-generic in the chunk count so a register of any size
57/// up to [`MAX_SUPPORTED_QUBITS`] gets a fixed-size word; the default
58/// `C = W_CHUNKS` covers 128 qubits, byte-for-byte the historical layout.
59/// The `FxBuildHasher` matches the hash used by the `FxHashMap` keys we
60/// wrap with; `REHASH=true` means `set()` keeps the cached hash in sync.
61pub type Word<const C: usize = W_CHUNKS> = PauliWord<[Chunk; C], FxBuildHasher, true>;
62
63/// `Err(TooManyQubits)` unless a `C`-chunk word holds `n_qubits`.
64pub(crate) fn check_width<const C: usize>(n_qubits: usize) -> Result<(), Error> {
65    if n_qubits > max_qubits::<C>() {
66        return Err(Error::TooManyQubits {
67            got: n_qubits,
68            max: max_qubits::<C>(),
69        });
70    }
71    Ok(())
72}

74/// Build a [`Word`] from a length-`n_qubits` slice of Pauli labels
75/// (`0=I, 1=X, 2=Z, 3=Y` — the [`ppvm_traits::char::Pauli`] discriminants).
76/// Sets all bits and rehashes once.
77pub fn word_from_codes<const C: usize>(codes: &[u8]) -> Result<Word<C>, Error> {
78    let n_qubits = codes.len();
79    check_width::<C>(n_qubits)?;
80    let mut w = Word::<C>::new(n_qubits);
81    for (q, &b) in codes.iter().enumerate() {
82        if b > 3 {
83            return Err(Error::InvalidPauliCode { code: b });
84        }
85        if b & 1 != 0 {
86            w.xbits.set(q, true);
87        }
88        if b & 2 != 0 {
89            w.zbits.set(q, true);
90        }
91    }
92    w.rehash();
93    Ok(w)
94}
95
96/// Inverse of [`word_from_codes`]: write `n_qubits` Pauli labels into `out`.
97pub fn codes_from_word<const C: usize>(w: &Word<C>, out: &mut [u8]) {
98    debug_assert_eq!(out.len(), w.n_qubits());
99    for (q, slot) in out.iter_mut().enumerate() {
100        let xb = w.xbits[q] as u8;
101        let zb = w.zbits[q] as u8;
102        *slot = xb | (zb << 1);
103    }
104}

106/// Parse a `"IXYZ..."` string into a [`Word`] together with the list of
107/// qubits where the Pauli is non-identity (the term's support).
108pub fn parse_pauli_string<const C: usize>(
109    s: &str,
110    n_qubits: usize,
111) -> Result<(Word<C>, Vec<u32>), Error> {
112    check_width::<C>(n_qubits)?;
113    let chars: Vec<char> = s.chars().filter(|c| *c != '_').collect();
114    if chars.len() != n_qubits {
115        return Err(Error::WrongLength {
116            expected: n_qubits,
117            got: chars.len(),
118        });
119    }
120    let mut w = Word::<C>::new(n_qubits);
121    let mut support = Vec::new();
122    for (q, c) in chars.into_iter().enumerate() {
123        match c {
124            'I' => {}
125            'X' => {
126                w.xbits.set(q, true);
127                support.push(q as u32);
128            }
129            'Z' => {
130                w.zbits.set(q, true);
131                support.push(q as u32);
132            }
133            'Y' => {
134                w.xbits.set(q, true);
135                w.zbits.set(q, true);
136                support.push(q as u32);
137            }
138            other => return Err(Error::InvalidPauliChar { c: other }),
139        }
140    }
141    w.rehash();
142    Ok((w, support))
143}

145/// Compute the support (non-identity qubits) of `w`.
146pub(crate) fn word_support<const C: usize>(w: &Word<C>, out: &mut Vec<u32>) {
147    out.clear();
148    for q in 0..w.n_qubits() {
149        if w.xbits[q] || w.zbits[q] {
150            out.push(q as u32);
151        }
152    }
153}

155/// Compact 64-bit hash of a [`Word`], used as the key in cache-friendly
156/// membership tables: an `FxHashMap<u64, ()>` over the basis has a working
157/// set ~6× smaller than `FxHashMap<Word, ()>`. The hash mixes the word's
158/// cached hash once through `FxHasher` and never touches the 32-byte
159/// payload.
160#[inline(always)]
161pub(crate) fn word_hash<const C: usize>(w: &Word<C>) -> u64 {
162    use std::hash::{Hash, Hasher};
163    let mut h = fxhash::FxHasher::default();
164    w.hash(&mut h);
165    h.finish()
166}
