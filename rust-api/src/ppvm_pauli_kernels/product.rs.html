4//! The packed Pauli-word product and its phase.
5//!
6//! For words `P = ⊗ p_k`, `Q = ⊗ q_k` stored as X/Z bit planes,
7//! `P · Q = iᵏ · ⊗ (p_k ⊕ q_k)`. Each site contributes `2·sign + imag` to `k`
8//! (`site_product` in `lean/PPVM/Kernels/RowProduct.lean`), so `k` is
9//! `2·popcount(sign) + popcount(imag)` mod 4 (`popcount_exponent`). Unused
10//! (padding) bits must be zero; zero sites contribute nothing and stay zero.
11
12use crate::Bits;
13
14/// Rows up to this many words use an iterator loop, longer rows an indexed one.
15/// Measured on arm64: the iterator loop is faster on short rows, the indexed
16/// loop (which LLVM unrolls further) on long rows.
17const SHORT_ROW: usize = 8;
18
19/// The `(sign, imag)` masks for `p = (a, b)`, `q = (c, d)`.
20#[inline(always)]
21fn masks<W: Bits>(a: W, b: W, c: W, d: W) -> (W, W) {
22    let sign = (a & b & c & !d) | (a & !b & !c & d) | (!a & b & c & d);
23    let imag = (a & !b & d) | (a & !c & d) | (!a & b & c) | (b & c & !d);
24    (sign, imag)
25}
26
27/// The exponent `k mod 4` from the mask counts. 4 divides 2³², so wrapping is exact.
28#[inline(always)]
29fn exponent(sign_count: u32, imag_count: u32) -> u8 {
30    (sign_count.wrapping_mul(2).wrapping_add(imag_count) % 4) as u8
31}
32
33/// One storage word of the product `dst · src`.
34///
35/// Returns the product's bits `[dst_x ⊕ src_x, dst_z ⊕ src_z]` and this word's
36/// contribution `2·popcount(sign) + popcount(imag)` to the phase exponent. A row's
37/// exponent is the wrapping sum of its words' contributions, mod 4.
38///
39/// For containers that write their own loop around the product, e.g. to follow
40/// it with a hash of the same words: a loop the compiler sees inline lets that
41/// work reuse the words still in registers, which a call to [`row_product`] did
42/// not (measured on `PauliWord`).
43#[inline(always)]
44pub fn word_product<W: Bits>(dst_x: W, dst_z: W, src_x: W, src_z: W) -> ([W; 2], u32) {
45    let (sign, imag) = masks(dst_x, dst_z, src_x, src_z);
46    let contribution = sign
47        .popcount()
48        .wrapping_mul(2)
49        .wrapping_add(imag.popcount());
50    ([dst_x ^ src_x, dst_z ^ src_z], contribution)
51}
52
53/// `dst ← dst · src`. Returns `k ∈ 0..4` with `dst · src = iᵏ · (dst ⊕ src)`.
54///
55/// All four slices must have the same length.
56#[inline]
57pub fn row_product<W: Bits>(dst_x: &mut [W], dst_z: &mut [W], src_x: &[W], src_z: &[W]) -> u8 {
58    debug_assert!(
59        dst_x.len() == dst_z.len() && src_x.len() == dst_x.len() && src_z.len() == dst_x.len()
60    );
61    let mut sign_count = 0u32;
62    let mut imag_count = 0u32;
63    if dst_x.len() <= SHORT_ROW {
64        for (((dx, dz), &sx), &sz) in dst_x.iter_mut().zip(dst_z.iter_mut()).zip(src_x).zip(src_z) {
65            let (sign, imag) = masks(*dx, *dz, sx, sz);
66            sign_count = sign_count.wrapping_add(sign.popcount());
67            imag_count = imag_count.wrapping_add(imag.popcount());
68            *dx = *dx ^ sx;
69            *dz = *dz ^ sz;
70        }
71    } else {
72        for i in 0..dst_x.len() {
73            let (a, b, c, d) = (dst_x[i], dst_z[i], src_x[i], src_z[i]);
74            let (sign, imag) = masks(a, b, c, d);
75            sign_count = sign_count.wrapping_add(sign.popcount());
76            imag_count = imag_count.wrapping_add(imag.popcount());
77            dst_x[i] = a ^ c;
78            dst_z[i] = b ^ d;
79        }
80    }
81    exponent(sign_count, imag_count)
82}

84/// The phase exponent of `a · b` without computing the product's bits.
85///
86/// All four slices must have the same length.
87#[inline]
88pub fn row_product_phase<W: Bits>(a_x: &[W], a_z: &[W], b_x: &[W], b_z: &[W]) -> u8 {
89    debug_assert!(a_x.len() == a_z.len() && b_x.len() == a_x.len() && b_z.len() == a_x.len());
90    let mut sign_count = 0u32;
91    let mut imag_count = 0u32;
92    if a_x.len() <= SHORT_ROW {
93        for (((&ax, &az), &bx), &bz) in a_x.iter().zip(a_z).zip(b_x).zip(b_z) {
94            let (sign, imag) = masks(ax, az, bx, bz);
95            sign_count = sign_count.wrapping_add(sign.popcount());
96            imag_count = imag_count.wrapping_add(imag.popcount());
97        }
98    } else {
99        for i in 0..a_x.len() {
100            let (sign, imag) = masks(a_x[i], a_z[i], b_x[i], b_z[i]);
101            sign_count = sign_count.wrapping_add(sign.popcount());
102            imag_count = imag_count.wrapping_add(imag.popcount());
103        }
104    }
105    exponent(sign_count, imag_count)
106}
