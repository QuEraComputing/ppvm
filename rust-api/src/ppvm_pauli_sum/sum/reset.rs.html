5use ppvm_traits::char::Pauli;
6use ppvm_traits::config::Config;
7use ppvm_traits::traits::*;

9/// Reset to `|0⟩` in the Heisenberg picture.
10///
11/// The adjoint of the reset channel maps `A_q ⊗ B` to `⟨0|A|0⟩ · I_q ⊗ B`:
12/// `I` and `Z` on the reset qubit become `I`, while `X` and `Y` drop out.
13/// Terms that collide after relabelling have their coefficients added.
14/// Lost qubits are left untouched, matching the tableau, which skips
15/// inactive qubits on reset; use `ResetLossChannel` to model reloading.
16impl<T: Config> Reset for PauliSum<T> {
17    fn reset(&mut self, addr0: usize) {
18        self.data_mut()
19            .retain(|k, _| k.get_lbit(addr0) || !k.get_xbit(addr0));
20        self.map_add(|k, v| {
21            if !k.get_lbit(addr0) && k.get_zbit(addr0) {
22                (k.set_new(addr0, Pauli::I), v.clone())
23            } else {
24                (k.clone(), v.clone())
25            }
26        });
27    }
28
29    // The trait defaults run reset first (Schrödinger order); the Heisenberg
30    // picture needs the basis change applied before the reset instead.
31    fn reset_x(&mut self, addr0: usize) {
32        self.h(addr0);
33        self.reset(addr0);
34    }
35
36    fn reset_y(&mut self, addr0: usize) {
37        self.s(addr0);
38        self.h(addr0);
39        self.reset(addr0);
40    }
41}

43#[cfg(test)]
44mod tests {
45    use crate::config::fxhash::ByteF64;
46
47    use super::*;
48
49    type Sum = PauliSum<ByteF64<2>>;
50
51    fn sum(n_qubits: usize, terms: &[(&str, f64)]) -> Sum {
52        let mut s: Sum = PauliSum::builder().n_qubits(n_qubits).build();
53        for &(word, coef) in terms {
54            s += (word, coef);
55        }
56        s
57    }
58
59    #[test]
60    fn test_reset_single_qubit_paulis() {
61        let mut s = sum(1, &[("I", 0.5)]);
62        s.reset(0);
63        assert_eq!(s, sum(1, &[("I", 0.5)]));
64
65        let mut s = sum(1, &[("Z", 0.5)]);
66        s.reset(0);
67        assert_eq!(s, sum(1, &[("I", 0.5)]));
68
69        let mut s = sum(1, &[("X", 0.5), ("Y", 0.25)]);
70        s.reset(0);
71        assert!(s.is_empty());
72    }
73
74    #[test]
75    fn test_reset_merges_colliding_terms() {
76        // Qubit 0 is the leftmost character.
77        let mut s = sum(2, &[("ZX", 0.5), ("IX", 0.25), ("YZ", 1.0)]);
78        s.reset(0);
79        assert_eq!(s, sum(2, &[("IX", 0.75)]));
80    }
81
82    #[test]
83    fn test_reset_only_touches_target_qubit() {
84        let mut s = sum(2, &[("XZ", 1.0), ("ZY", 1.0)]);
85        s.reset(1);
86        assert_eq!(s, sum(2, &[("XI", 1.0)]));
87    }
88
89    #[test]
90    fn test_reset_leaves_lost_qubit_untouched() {
91        use ppvm_pauli_word::loss::LossyPauliWord;
92        type Lossy = PauliSum<
93            crate::config::fxhash::Byte<
94                1,
95                f64,
96                ppvm_traits::traits::NoStrategy,
97                LossyPauliWord<[u8; 1], fxhash::FxBuildHasher>,
98            >,
99        >;
100        let mut s: Lossy = PauliSum::builder().n_qubits(2).build();
101        s += ("LZ", 1.0);
102        s += ("ZZ", 0.5);
103        s.reset(0);
104        let mut expect: Lossy = PauliSum::builder().n_qubits(2).build();
105        expect += ("LZ", 1.0);
106        expect += ("IZ", 0.5);
107        assert_eq!(s, expect);
108    }

110    #[test]
111    fn test_reset_after_x_gives_zero_state() {
112        // X then reset: ⟨Z⟩ = +1. Heisenberg order: reset, then X.
113        let mut o = sum(1, &[("Z", 1.0)]);
114        o.reset(0);
115        o.x(0);
116        assert_eq!(o, sum(1, &[("I", 1.0)]));
117    }
118
119    #[test]
120    fn test_reset_x_prepares_plus_state() {
121        // reset_x then measure X: ⟨X⟩ = +1, ⟨Z⟩ = 0.
122        let mut o = sum(1, &[("X", 1.0)]);
123        o.reset_x(0);
124        assert_eq!(o, sum(1, &[("I", 1.0)]));
125
126        let mut o = sum(1, &[("Z", 1.0)]);
127        o.reset_x(0);
128        assert!(o.is_empty());
129    }
130
131    #[test]
132    fn test_reset_y_prepares_plus_i_state() {
133        // reset_y then measure Y: ⟨Y⟩ = +1 on |+i⟩.
134        let mut o = sum(1, &[("Y", 1.0)]);
135        o.reset_y(0);
136        assert_eq!(o, sum(1, &[("I", 1.0)]));
137
138        let mut o = sum(1, &[("X", 1.0)]);
139        o.reset_y(0);
140        assert!(o.is_empty());
141    }
