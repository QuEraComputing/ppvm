9/// Reset a live qubit to `|0⟩` in the Heisenberg picture.
10///
11/// The adjoint of the reset channel maps `A_q ⊗ B` to `⟨0|A|0⟩ · I_q ⊗ B`:
12/// `I` and `Z` on the reset qubit become `I`, while `X` and `Y` drop out.
13/// Terms that collide after relabelling have their coefficients added.
14/// Lost qubits are left untouched, matching the tableau, which skips
15/// inactive qubits on state reset.
16impl<T: Config> ResetState for PauliSum<T> {
17    fn reset_state(&mut self, addr0: usize) {
18        self.data_mut()
19            .retain(|k, _| k.get_lbit(addr0) || !k.get_xbit(addr0));
20        self.map_add(|k, v| {
21            if !k.get_lbit(addr0) && k.get_zbit(addr0) {
22                (k.set_new(addr0, Pauli::I), v.clone())
23            } else {
24                (k.clone(), v.clone())
25            }
26        });
27    }
28}
29
30/// Full reset to `|0⟩` in the Heisenberg picture, term by term via
31/// [`WordReset`]. For lossy words this also reloads lost qubits.
32impl<T: Config> Reset for PauliSum<T>
33where
34    T::PauliWordType: WordReset,
35{
36    fn reset(&mut self, addr0: usize) {
37        let (data, aux) = self.data_aux_mut();
38        aux.clear();
39        data.map_flat_add_assign(aux, |k, v| {
40            let v = v.clone();
41            k.word_reset(addr0).map(move |w| (w, v.clone()))
42        });
43        self.swap();
44    }
45
46    // The trait defaults run reset first (Schrödinger order); the Heisenberg
47    // picture needs the basis change applied before the reset instead.
48    fn reset_x(&mut self, addr0: usize) {
49        self.h(addr0);
50        self.reset(addr0);
51    }
52
53    fn reset_y(&mut self, addr0: usize) {
54        self.s(addr0);
55        self.h(addr0);
56        self.reset(addr0);
57    }
58}

61mod tests {

67    type Sum = PauliSum<ByteF64<2>>;
68
69    fn sum(n_qubits: usize, terms: &[(&str, f64)]) -> Sum {
70        let mut s: Sum = PauliSum::builder().n_qubits(n_qubits).build();
71        for &(word, coef) in terms {
72            s += (word, coef);
73        }
74        s
75    }
76
77    #[test]
78    fn test_reset_single_qubit_paulis() {
79        let mut s = sum(1, &[("I", 0.5)]);
80        s.reset(0);
81        assert_eq!(s, sum(1, &[("I", 0.5)]));
82
83        let mut s = sum(1, &[("Z", 0.5)]);
84        s.reset(0);
85        assert_eq!(s, sum(1, &[("I", 0.5)]));
86
87        let mut s = sum(1, &[("X", 0.5), ("Y", 0.25)]);
88        s.reset(0);
89        assert!(s.is_empty());
90    }
91
92    #[test]
93    fn test_reset_merges_colliding_terms() {
94        // Qubit 0 is the leftmost character.
95        let mut s = sum(2, &[("ZX", 0.5), ("IX", 0.25), ("YZ", 1.0)]);
96        s.reset(0);
97        assert_eq!(s, sum(2, &[("IX", 0.75)]));
98    }
99
100    #[test]
101    fn test_reset_only_touches_target_qubit() {
102        let mut s = sum(2, &[("XZ", 1.0), ("ZY", 1.0)]);
103        s.reset(1);
104        assert_eq!(s, sum(2, &[("XI", 1.0)]));
105    }

107    type Lossy = PauliSum<
108        crate::config::fxhash::Byte<
109            1,
110            f64,
111            ppvm_traits::traits::NoStrategy,
112            LossyPauliWord<[u8; 1], fxhash::FxBuildHasher>,
113        >,
114    >;
115
116    fn lossy(n_qubits: usize, terms: &[(&str, f64)]) -> Lossy {
117        let mut s: Lossy = PauliSum::builder().n_qubits(n_qubits).build();
118        for &(word, coef) in terms {
119            s += (word, coef);
120        }
121        s
122    }

124    #[test]
125    fn test_reset_gives_zero_state_even_from_loss() {
126        // ⟨Z⟩ = 1 after reset on any state: Z maps to the full identity I + L.
127        let mut o = lossy(1, &[("Z", 1.0)]);
128        o.reset(0);
129        assert_eq!(o, lossy(1, &[("I", 1.0), ("L", 1.0)]));
130    }
131
132    #[test]
133    fn test_reset_leaves_no_loss_probability() {
134        // The lost-projector expectation vanishes after a full reset.
135        let mut o = lossy(2, &[("LZ", 1.0)]);
136        o.reset(0);
137        assert!(o.data().iter().all(|(_, v)| *v == 0.0));
138    }
139
140    #[test]
141    fn test_reset_state_leaves_lost_qubit_untouched() {
142        let mut s: Lossy = PauliSum::builder().n_qubits(2).build();
143        s += ("LZ", 1.0);
144        s += ("ZZ", 0.5);
145        s.reset_state(0);
146        let mut expect: Lossy = PauliSum::builder().n_qubits(2).build();
147        expect += ("LZ", 1.0);
148        expect += ("IZ", 0.5);
149        assert_eq!(s, expect);
150    }

152    #[test]
153    fn test_reset_after_x_gives_zero_state() {
154        // X then reset: ⟨Z⟩ = +1. Heisenberg order: reset, then X.
155        let mut o = sum(1, &[("Z", 1.0)]);
156        o.reset(0);
157        o.x(0);
158        assert_eq!(o, sum(1, &[("I", 1.0)]));
159    }

161    #[test]
162    fn test_reset_x_prepares_plus_state() {
163        // reset_x then measure X: ⟨X⟩ = +1, ⟨Z⟩ = 0.
164        let mut o = sum(1, &[("X", 1.0)]);
165        o.reset_x(0);
166        assert_eq!(o, sum(1, &[("I", 1.0)]));
167
168        let mut o = sum(1, &[("Z", 1.0)]);
169        o.reset_x(0);
170        assert!(o.is_empty());
171    }

173    #[test]
174    fn test_reset_y_prepares_plus_i_state() {
175        // reset_y then measure Y: ⟨Y⟩ = +1 on |+i⟩.
