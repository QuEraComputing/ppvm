9pub(crate) fn rotate_1_map_insert_closure<T: Config>(
10    k: &T::PauliWordType,
11    v: &mut T::Coeff,
12    axis: Pauli,
13    addr0: usize,
14    sin: &T::Coeff,
15    cos: &T::Coeff,
16) -> Option<(T::PauliWordType, T::Coeff)> {
17    if axis == Pauli::L {
18        panic!("Rotation axis cannot be L");
19    }
20    // Check loss first — avoids reading xbit/zbit on lost qubits
21    if k.get_lbit(addr0) {
22        return None;
23    }
24    let xbit = k.get_xbit(addr0);
25    let zbit = k.get_zbit(addr0);
26    // Reconstruct 2-bit Pauli code (I=00, X=01, Z=10, Y=11) from bits
27    let p_g = (zbit as u8) << 1 | (xbit as u8);
28    let (eps, p_q) = levi_civita(p_g, axis as u8);
29    if eps == 0 {
30        None
31    } else {
32        let mut coeff = v.clone();
33        *v *= cos.clone();
34        let mut new_word = k.clone();
35        new_word.set_xbit(addr0, p_q & 0b01 != 0);
36        new_word.set_zbit(addr0, p_q & 0b10 != 0);
37        new_word.rehash();
38
39        coeff *= sin.mul_sign(eps);
40        Some((new_word, coeff))
41    }
42}

44impl<T: Config> RotationOne<T> for PauliSum<T>
45where
46    T::Coeff: std::ops::MulAssign,
47    T::Map: ACMapInsert<T::Storage, T::Coeff, T::BuildHasher, T::PauliWordType> + ACMapConsume,
48{
49    fn rotate_1(&mut self, axis: Pauli, addr0: usize, theta: <T as Config>::Coeff) {
50        let (sin, cos) = theta.sin_cos();
51        self.map_insert(|k, v| rotate_1_map_insert_closure::<T>(k, v, axis, addr0, &sin, &cos));
52    }
53
54    #[inline]
55    fn rx(&mut self, addr0: usize, theta: impl Into<T::Coeff>) {
56        // Axis = X (xbit=1, zbit=0). Commutes when zbit==false (I or X).
57        // Anticommuting: Z(xbit=0)→new Y, eps=+1; Y(xbit=1)→new Z, eps=-1
58        let (sin, cos) = theta.into().sin_cos();
59        self.map_insert(|k, v| {
60            if k.get_lbit(addr0) {
61                return None;
62            }
63            let zbit = k.get_zbit(addr0);
64            if !zbit {
65                return None;
66            }
67            let xbit = k.get_xbit(addr0);
68            let mut coeff = v.clone();
69            *v *= cos.clone();
70            let mut new_word = k.clone();
71            new_word.set_xbit(addr0, !xbit);
72            new_word.rehash();
73            let eps: i8 = if xbit { -1 } else { 1 };
74            coeff *= sin.mul_sign(eps);
75            Some((new_word, coeff))
76        });
77    }
78
79    #[inline]
80    fn ry(&mut self, addr0: usize, theta: impl Into<T::Coeff>) {
81        // Axis = Y (xbit=1, zbit=1). Commutes when xbit==zbit (I or Y).
82        // Anticommuting: X(zbit=0)→new Z, eps=+1; Z(zbit=1)→new X, eps=-1
83        let (sin, cos) = theta.into().sin_cos();
84        self.map_insert(|k, v| {
85            if k.get_lbit(addr0) {
86                return None;
87            }
88            let xbit = k.get_xbit(addr0);
89            let zbit = k.get_zbit(addr0);
90            if xbit == zbit {
91                return None;
92            }
93            let mut coeff = v.clone();
94            *v *= cos.clone();
95            let mut new_word = k.clone();
96            new_word.set_xbit(addr0, !xbit);
97            new_word.set_zbit(addr0, !zbit);
98            new_word.rehash();
99            let eps: i8 = if zbit { -1 } else { 1 };
100            coeff *= sin.mul_sign(eps);
101            Some((new_word, coeff))
102        });
103    }
104
105    #[inline]
106    fn rz(&mut self, addr0: usize, theta: impl Into<T::Coeff>) {
107        // Axis = Z (xbit=0, zbit=1). Commutes when xbit==false (I or Z).
108        // Anticommuting: X(zbit=0)→new Y, eps=-1; Y(zbit=1)→new X, eps=+1
109        let (sin, cos) = theta.into().sin_cos();
110        self.map_insert(|k, v| {
111            if k.get_lbit(addr0) {
112                return None;
113            }
114            let xbit = k.get_xbit(addr0);
115            if !xbit {
116                return None;
117            }
118            let zbit = k.get_zbit(addr0);
119            let mut coeff = v.clone();
120            *v *= cos.clone();
121            let mut new_word = k.clone();
122            new_word.set_zbit(addr0, !zbit);
123            new_word.rehash();
124            let eps: i8 = if zbit { 1 } else { -1 };
125            coeff *= sin.mul_sign(eps);
126            Some((new_word, coeff))
127        });
128    }
129}

131impl<T: Config> RotXY<T> for PauliSum<T>
132where
133    PauliSum<T>: RotationOne<T>,
134{
135    fn r(&mut self, addr0: usize, axis_angle: <T as Config>::Coeff, theta: <T as Config>::Coeff) {
136        // R(axis_angle, θ) = RZ(axis_angle)·RX(θ)·RZ(−axis_angle). PauliSum runs
137        // in the Heisenberg picture (observables propagate backward), so the
138        // sub-rotations are emitted in reverse of the tableau's forward order.
139        self.rz(addr0, axis_angle.clone());
140        self.rx(addr0, theta);
141        self.rz(addr0, -axis_angle);
142    }
143}

145impl<T: Config> TGate<T> for PauliSum<T>
146where
147    PauliSum<T>: RotationOne<T>,
148{
149    fn t(&mut self, addr0: usize) {
150        // T = e^{iπ/8} · RZ(π/4); the global phase drops out under conjugation.
151        self.rz(addr0, T::Coeff::from(std::f64::consts::FRAC_PI_4));
152    }
153
154    fn t_dag(&mut self, addr0: usize) {
155        self.rz(addr0, T::Coeff::from(-std::f64::consts::FRAC_PI_4));
156    }
157}

159impl<T: Config> U3Gate<T> for PauliSum<T>
160where
161    PauliSum<T>: RotationOne<T>,
162{
163    fn u3(&mut self, addr: usize, theta: T::Coeff, phi: T::Coeff, lambda: T::Coeff) {
164        // U3(θ, φ, λ) = RZ(φ)·RY(θ)·RZ(λ). Heisenberg picture, so the
165        // sub-rotations run in reverse of the tableau's forward order.
166        self.rz(addr, phi);
167        self.ry(addr, theta);
168        self.rz(addr, lambda);
169    }
170}

172/// 2-bit Pauli code: 00 I, 01 X, 10 Z, 11 Y
173/// Returns \(ε, k\) so that  –i \[P_i, P_j\]/2 = ε · P_k.
174/// For every commuting pair it yields (0, 0).
175#[inline]
176pub fn levi_civita(i: u8, j: u8) -> (i8, u8) {
177    // --------------------------------------------------- third Pauli by XOR
178    let k = i ^ j; // 0 when i == j
179
180    // ----------- commute ⇔ i==0  OR  j==0  OR  k==0  (no false positives)
181    let commute = ((i == 0) | (j == 0) | (k == 0)) as u8; // 1 = commute
182
183    // ------------------------------------------------------ sign ε_{ijk}
184    #[inline]
185    fn rank(p: u8) -> u8 {
186        let b1 = p >> 1; // MSB
187        (b1 << 1).wrapping_sub(b1 & (p & 1)) // 0,1,2 for X,Y,Z
188    }
189
190    let ri = rank(i);
191    let rj = rank(j);
192
193    // diff = (rj - ri) mod 3   without an actual modulus
194    let mut diff = rj.wrapping_sub(ri).wrapping_add(3); // 0…5
195    diff -= 3 & (0u8.wrapping_sub(diff >> 2)); // if ≥3 subtract 3
196
197    // +1 when diff == 1,  –1 when diff == 2
198    let eps_raw = 1i8 - 2 * ((diff >> 1) as i8);
199
200    // --------------------------------------------------- zero when commute
201    let eps = eps_raw * (1 - commute as i8); // 0 if commute
202    let k = k * (1 - commute); // 0 if commute
203
204    (eps, k)
205}

282    fn test_r() {
283        use std::f64::consts::FRAC_PI_2;
284        let theta = 2.1;
285
286        // r(axis_angle=0, θ) == rx(θ).
287        let mut via_r: PauliSum<ByteF64<2>> = PauliSum::builder().n_qubits(1).build();
288        via_r += ("Z", 1.0);
289        via_r.r(0, 0.0, theta);
290        let mut via_rx: PauliSum<ByteF64<2>> = PauliSum::builder().n_qubits(1).build();
291        via_rx += ("Z", 1.0);
292        via_rx.rx(0, theta);
293        assert!((via_r.overlap(&via_rx) - 1.0).abs() < 1e-9);
294
295        // r(axis_angle=π/2, θ) must equal ry(θ) — NOT ry(−θ). This is the case
296        // that distinguishes the Heisenberg (backward) order from the
297        // Schrödinger one: a forward-ordered impl would yield ry(−θ) here.
298        let mut via_r: PauliSum<ByteF64<2>> = PauliSum::builder().n_qubits(1).build();
299        via_r += ("Z", 1.0);
300        via_r.r(0, FRAC_PI_2, theta);
301        let mut via_ry: PauliSum<ByteF64<2>> = PauliSum::builder().n_qubits(1).build();
302        via_ry += ("Z", 1.0);
303        via_ry.ry(0, theta);
304        assert!((via_r.overlap(&via_ry) - 1.0).abs() < 1e-9);
305    }

340    /// `⟨0|O|0⟩` for a one-qubit observable: the I and Z coefficients.
341    fn zero_state_expectation(o: &PauliSum<ByteF64<2>>) -> f64 {
342        let mut zero_state: PauliSum<ByteF64<2>> = PauliSum::builder().n_qubits(1).build();
343        zero_state += ("I", 1.0);
344        zero_state += ("Z", 1.0);
345        o.overlap(&zero_state)
346    }

348    /// Heisenberg-evolve `pauli` back through `circuit` and evaluate on `|0⟩`.
349    fn expectation(pauli: &str, circuit: impl Fn(&mut PauliSum<ByteF64<2>>)) -> f64 {
350        let mut o: PauliSum<ByteF64<2>> = PauliSum::builder().n_qubits(1).build();
351        o += (pauli, 1.0);
352        circuit(&mut o);
353        zero_state_expectation(&o)
354    }

357    fn test_t_on_plus_state() {
358        // T·H|0⟩ = (|0⟩ + e^{iπ/4}|1⟩)/√2, so ⟨X⟩ = cos(π/4), ⟨Y⟩ = sin(π/4).
359        // Gates are listed in reverse (Heisenberg) order: T first, then H.
360        let s = std::f64::consts::FRAC_1_SQRT_2;
361        let x = expectation("X", |o| {
362            o.t(0);
363            o.h(0);
364        });
365        let y = expectation("Y", |o| {
366            o.t(0);
367            o.h(0);
368        });
369        assert!((x - s).abs() < 1e-12, "⟨X⟩ = {x}");
370        assert!((y - s).abs() < 1e-12, "⟨Y⟩ = {y}");
371
372        let y_dag = expectation("Y", |o| {
373            o.t_dag(0);
374            o.h(0);
375        });
376        assert!((y_dag + s).abs() < 1e-12, "⟨Y⟩ after T† = {y_dag}");
377    }

392    fn test_u3_on_zero_state() {
393        // U3(θ, φ, λ)|0⟩ = cos(θ/2)|0⟩ + e^{iφ} sin(θ/2)|1⟩ (λ only sets a phase).
394        let (theta, phi, lambda) = (0.34 * PI, 0.21 * PI, 0.46 * PI);
395        let u3 = |o: &mut PauliSum<ByteF64<2>>| o.u3(0, theta, phi, lambda);
396        let x = expectation("X", u3);
397        let y = expectation("Y", u3);
398        let z = expectation("Z", u3);
399        assert!((x - theta.sin() * phi.cos()).abs() < 1e-12, "⟨X⟩ = {x}");
400        assert!((y - theta.sin() * phi.sin()).abs() < 1e-12, "⟨Y⟩ = {y}");
401        assert!((z - theta.cos()).abs() < 1e-12, "⟨Z⟩ = {z}");
402    }
