8fn gcd(mut a: usize, mut b: usize) -> usize {
9    while b != 0 {
10        (a, b) = (b, a % b);
11    }
12    a
13}

15fn checked_lcm(a: usize, b: usize) -> Option<usize> {
16    a.checked_div(gcd(a, b)).and_then(|q| q.checked_mul(b))
17}

19/// Exact cyclic order of `perm`, or `None` if it does not fit in `u32`.
20fn permutation_order(perm: &[u32], generator: usize) -> Option<u32> {
21    let mut seen = vec![false; perm.len()];
22    let mut order = 1usize;
23    for start in 0..perm.len() {
24        if seen[start] {
25            continue;
26        }
27        let mut length = 0usize;
28        let mut q = start;
29        loop {
30            assert!(!seen[q], "generator {generator} contains a malformed cycle");
31            seen[q] = true;
32            length += 1;
33            q = perm[q] as usize;
34            if q == start {
35                break;
36            }
37        }
38        order = checked_lcm(order, length)?;
39    }
40    u32::try_from(order).ok()
41}

43fn permutations_commute(left: &[u32], right: &[u32]) -> bool {
44    (0..left.len()).all(|q| left[right[q] as usize] == right[left[q] as usize])
45}

47pub(super) fn checked_group_order(orders: &[u32]) -> Option<usize> {
48    orders
49        .iter()
50        .try_fold(1usize, |acc, &value| acc.checked_mul(value as usize))
51}

53pub(super) fn validate_site_count(n: usize, context: &str) {
54    let max_index = n
55        .checked_sub(1)
56        .unwrap_or_else(|| panic!("{context}: site count must be positive"));
57    u32::try_from(max_index)
58        .unwrap_or_else(|_| panic!("{context}: site count {n} exceeds the u32-addressable range"));
59}

61/// A precondition violation in [`TranslationGroup::try_from_generators`].
62///
63/// Every variant is caller-supplied-input error, and its [`Display`]
64/// text is exactly what [`TranslationGroup::from_generators`] panics
65/// with.
66///
67/// [`Display`]: std::fmt::Display
68#[derive(Debug, Clone, PartialEq, Eq)]
69pub enum GroupError {
70    /// `perms` and `orders` describe different numbers of generators.
71    LengthMismatch { perms: usize, orders: usize },
72    /// A generator's permutation is not `n_qubits` long.
73    PermutationLength {
74        generator: usize,
75        len: usize,
76        n_qubits: usize,
77    },
78    /// A generator maps a qubit outside `0..n_qubits`.
79    TargetOutOfRange {
80        generator: usize,
81        target: u32,
82        n_qubits: usize,
83    },
84    /// A generator maps two qubits to the same position.
85    DuplicateTarget { generator: usize, target: u32 },
86    /// A generator declares cyclic order zero.
87    ZeroOrder { generator: usize },
88    /// A generator's declared order is not its exact cyclic order.
89    OrderMismatch {
90        generator: usize,
91        declared: u32,
92        exact: u32,
93    },
94    /// Two generators do not commute, so they generate no abelian group.
95    NonCommuting { left: usize, right: usize },
96    /// A generator's exact cyclic order does not fit in `u32`.
97    PermutationOrderOverflow { generator: usize },
98    /// The group order `Π orders[g]` does not fit in `usize`.
99    GroupOrderOverflow,
100}

102impl std::fmt::Display for GroupError {
103    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
104        match self {
105            Self::LengthMismatch { perms, orders } => write!(
106                f,
107                "perms ({perms} generators) and orders ({orders}) must have the same length"
108            ),
109            Self::PermutationLength {
110                generator,
111                len,
112                n_qubits,
113            } => write!(
114                f,
115                "generator {generator}: permutation length {len} != n_qubits {n_qubits}"
116            ),
117            Self::TargetOutOfRange {
118                generator,
119                target,
120                n_qubits,
121            } => write!(
122                f,
123                "generator {generator}: target {target} out of range [0, {n_qubits})"
124            ),
125            Self::DuplicateTarget { generator, target } => write!(
126                f,
127                "generator {generator}: not a permutation (duplicate target {target})"
128            ),
129            Self::ZeroOrder { generator } => {
130                write!(f, "generator {generator} order must be nonzero")
131            }
132            Self::OrderMismatch {
133                generator,
134                declared,
135                exact,
136            } => write!(
137                f,
138                "generator {generator} declared order {declared} != exact permutation order {exact}"
139            ),
140            Self::NonCommuting { left, right } => {
141                write!(f, "generators {left} and {right} do not commute")
142            }
143            Self::PermutationOrderOverflow { generator } => write!(
144                f,
145                "generator {generator} exact permutation order does not fit in u32"
146            ),
147            Self::GroupOrderOverflow => write!(f, "group order overflows usize"),
148        }
149    }
150}

154/// A finite abelian symmetry group acting on qubit positions by
155/// permutations.
156///
157/// Build via the convenience constructors [`Self::chain_1d`],
158/// [`Self::torus_2d`], [`Self::torus_3d`], [`Self::ladder`], or
159/// [`Self::from_generators`] for an arbitrary list of generator
160/// permutations.
161///
162/// `perms[g]` is the permutation that **generator `g`** applies to qubit
163/// indices: a qubit at position `q` moves to position `perms[g][q]`
164/// under one application of generator `g`. `orders[g]` is the exact
165/// cyclic order of generator `g`. The abstract group is the direct
166/// product of these cyclic groups, with order `Π orders[g]`. Its combined
167/// permutation action may have a kernel, so distinct group elements can
168/// act identically.
169///
170/// Only the **generators** are stored; [`Self::canonicalize`] either runs
171/// the `O(N)` least-rotation scan (chain/ladder layouts) or walks the
172/// group as a mixed-radix odometer.
173#[derive(Debug, Clone)]
174pub struct TranslationGroup {
175    /// Number of qubits the group acts on.
176    n_qubits: usize,
177    /// One permutation per generator. `perms[g][q]` is the position
178    /// that qubit `q` maps to under one application of generator `g`.
179    pub(super) perms: Vec<Vec<u32>>,
180    /// Cyclic order of each generator.
181    pub(super) orders: Vec<u32>,
182    order: usize,
183    phase_modulus: usize,
184    /// Set when the group is a *single* generator acting as a cyclic
185    /// shift inside contiguous, aligned blocks of qubits — i.e. exactly
186    /// the [`Self::chain_1d`] and [`Self::ladder`] layouts. Enables the
187    /// `O(N)` least-rotation canonicalizer (see
188    /// [`Self::canonicalize_block_cyclic`]).
189    pub(super) block_cyclic: Option<BlockCyclic>,
190    /// Per generator, its block-rotation form when it has one (all lattice
191    /// translations do). Enables the masked-shift [`Self::apply_generator`].
192    pub(super) rotations: Vec<Option<BlockRotation>>,
193}

195/// Layout of a single-generator group acting as a cyclic shift within
196/// `n_blocks` contiguous, aligned blocks of `len` qubits each: qubit
197/// `b * len + j` maps to `b * len + (j + 1) % len`.
198#[derive(Debug, Clone, Copy)]
199pub(super) struct BlockCyclic {
200    n_blocks: usize,
201    len: usize,
202}
203
204/// A generator that acts as a cyclic shift by `stride` positions within
205/// aligned blocks of `block` qubits: `b·block + p ↦ b·block + (p + stride) mod block`.
206///
207/// Every lattice-translation generator has this form: the fastest axis of a
208/// torus is `stride = 1` with `block = lx`, the next is `stride = lx` with
209/// `block = lx·ly`, and so on. Recognising it lets the whole permutation be
210/// applied as a masked shift of the two bit planes rather than a per-qubit
211/// gather (see [`TranslationGroup::apply_block_rotation`]).
212#[derive(Debug, Clone)]
213pub(super) struct BlockRotation {
214    stride: usize,
215    block: usize,
216    /// Destinations that survive the plain left shift: everything except the
217    /// low `stride` slots of each block (which receive the previous block's
218    /// spill) and everything at or beyond `n_qubits`.
219    keep: Vec<u64>,
220    /// Sources that wrap: the top `stride` slots of each block.
221    high: Vec<u64>,
222}
223
224/// Widest storage the masked-shift path handles, in 64-bit words.
225const MAX_ROT_WORDS: usize = 16;
226
227/// Recognise a generator permutation as a [`BlockRotation`], and precompute
228/// its masks. Returns `None` for permutations that are not block rotations.
229fn detect_block_rotation(n_qubits: usize, perm: &[u32]) -> Option<BlockRotation> {
230    if n_qubits == 0 {
231        return None;
232    }
233    let stride = perm[0] as usize;
234    if stride == 0 {
235        return None;
236    }
237    // Whatever maps to qubit 0 sits `stride` below the top of block 0.
238    let block = perm.iter().position(|&t| t == 0)? + stride;
239    if block > n_qubits || stride >= block || !n_qubits.is_multiple_of(block) {
240        return None;
241    }
242    for b in 0..n_qubits / block {
243        for p in 0..block {
244            if perm[b * block + p] as usize != b * block + (p + stride) % block {
245                return None;
246            }
247        }
248    }
249    let mut keep = vec![0u64; n_qubits.div_ceil(64)];
250    let mut high = keep.clone();
251    for q in 0..n_qubits {
252        let p = q % block;
253        if p >= stride {
254            keep[q / 64] |= 1u64 << (q % 64);
255        }
256        if p >= block - stride {
257            high[q / 64] |= 1u64 << (q % 64);
258        }
259    }
260    Some(BlockRotation {
261        stride,
262        block,
263        keep,
264        high,
265    })
266}

268/// `dst = src << s` over a little-endian multiword bit array.
269#[inline]
270fn shl_words(src: &[u64], dst: &mut [u64], s: usize) {
271    let (ws, bs) = (s / 64, s % 64);
272    for i in (0..src.len()).rev() {
273        let lo = if i >= ws { src[i - ws] } else { 0 };
274        dst[i] = if bs == 0 {
275            lo
276        } else {
277            let hi = if i > ws {
278                src[i - ws - 1] >> (64 - bs)
279            } else {
280                0
281            };
282            (lo << bs) | hi
283        };
284    }
285}

287/// `dst = src >> s` over a little-endian multiword bit array.
288#[inline]
289fn shr_words(src: &[u64], dst: &mut [u64], s: usize) {
290    let n = src.len();
291    let (ws, bs) = (s / 64, s % 64);
292    for i in 0..n {
293        let hi = if i + ws < n { src[i + ws] } else { 0 };
294        dst[i] = if bs == 0 {
295            hi
296        } else {
297            let lo = if i + ws + 1 < n {
298                src[i + ws + 1] << (64 - bs)
299            } else {
300                0
301            };
302            (hi >> bs) | lo
303        };
304    }
305}

307/// Detect the [`BlockCyclic`] layout, if the generators have it.
308fn detect_block_cyclic(n_qubits: usize, perms: &[Vec<u32>], orders: &[u32]) -> Option<BlockCyclic> {
309    if perms.len() != 1 {
310        return None;
311    }
312    let len = orders[0] as usize;
313    if len == 0 || n_qubits == 0 || !n_qubits.is_multiple_of(len) {
314        return None;
315    }
316    let n_blocks = n_qubits / len;
317    let perm = &perms[0];
318    for b in 0..n_blocks {
319        for j in 0..len {
320            if perm[b * len + j] as usize != b * len + (j + 1) % len {
321                return None;
322            }
323        }
324    }
325    Some(BlockCyclic { n_blocks, len })
326}

328/// Start index of the lexicographically smallest rotation of an abstract
329/// `m`-symbol cyclic sequence, via the two-pointer (Booth/Duval) scan.
330///
331/// `cmp(a, b)` compares the symbols at positions `a` and `b`. `O(m)`
332/// comparisons, no allocation.
333fn least_rotation<F>(m: usize, cmp: &F) -> usize
334where
335    F: Fn(usize, usize) -> std::cmp::Ordering,
336{
337    let (mut i, mut j, mut k) = (0usize, 1usize, 0usize);
338    while i < m && j < m && k < m {
339        match cmp((i + k) % m, (j + k) % m) {
340            std::cmp::Ordering::Equal => {
341                k += 1;
342                continue;
343            }
344            std::cmp::Ordering::Greater => i += k + 1,
345            std::cmp::Ordering::Less => j += k + 1,
346        }
347        if i == j {
348            j += 1;
349        }
350        k = 0;
351    }
352    i.min(j)
353}

355/// Period of the cyclic sequence `t ↦ start + t (mod m)` — the smallest
356/// `p` dividing `m` with `s[t] == s[t + p]` for all `t`.
357///
358/// Computed as the length of the first Lyndon factor (Duval): the minimal
359/// rotation of a sequence is a power `w^{m/|w|}` of a Lyndon word `w`, and
360/// `|w|` is the period. `O(m)` comparisons, no allocation. Callers pass the
361/// `start` returned by [`least_rotation`]; the count of rotations achieving
362/// the minimum is then `m / period`, spaced `period` apart.
363fn minimal_rotation_period<F>(m: usize, start: usize, cmp: &F) -> usize
364where
365    F: Fn(usize, usize) -> std::cmp::Ordering,
366{
367    let at = |t: usize| (start + t) % m;
368    let (mut j, mut k) = (1usize, 0usize);
369    while j < m {
370        match cmp(at(k), at(j)) {
371            std::cmp::Ordering::Less => {
372                k = 0;
373                j += 1;
374            }
375            std::cmp::Ordering::Equal => {
376                k += 1;
377                j += 1;
378            }
379            std::cmp::Ordering::Greater => break,
380        }
381    }
382    let len = j - k;
383    if m.is_multiple_of(len) { len } else { m }
384}

386impl TranslationGroup {
387    /// Construct from explicit generator permutations and orders,
388    /// panicking on any precondition violation.
389    ///
390    /// Each `perm` must be a permutation of `0..n_qubits`. Each `order`
391    /// must be the permutation's exact cyclic order, not merely a
392    /// multiple for which `perm^order == identity`. Generators must
393    /// commute, but their combined action may still have a kernel.
394    ///
395    /// Use [`Self::try_from_generators`] when the generators come from
396    /// outside the program (an FFI boundary, a config file) and a
397    /// precondition violation should be reported rather than abort.
398    pub fn from_generators(n_qubits: usize, perms: Vec<Vec<u32>>, orders: Vec<u32>) -> Self {
399        Self::try_from_generators(n_qubits, perms, orders)
400            .unwrap_or_else(|err| panic!("TranslationGroup::from_generators: {err}"))
401    }
402
403    /// Fallible [`Self::from_generators`]: validates every precondition
404    /// on the caller-supplied generators and reports the first violation
405    /// as a [`GroupError`] instead of panicking.
406    pub fn try_from_generators(
407        n_qubits: usize,
408        perms: Vec<Vec<u32>>,
409        orders: Vec<u32>,
410    ) -> Result<Self, GroupError> {
411        if perms.len() != orders.len() {
412            return Err(GroupError::LengthMismatch {
413                perms: perms.len(),
414                orders: orders.len(),
415            });
416        }
417        for (generator, perm) in perms.iter().enumerate() {
418            if perm.len() != n_qubits {
419                return Err(GroupError::PermutationLength {
420                    generator,
421                    len: perm.len(),
422                    n_qubits,
423                });
424            }
425            let mut seen = vec![false; n_qubits];
426            for &target in perm {
427                if target as usize >= n_qubits {
428                    return Err(GroupError::TargetOutOfRange {
429                        generator,
430                        target,
431                        n_qubits,
432                    });
433                }
434                if seen[target as usize] {
435                    return Err(GroupError::DuplicateTarget { generator, target });
436                }
437                seen[target as usize] = true;
438            }
439        }
440        for (generator, &declared) in orders.iter().enumerate() {
441            if declared == 0 {
442                return Err(GroupError::ZeroOrder { generator });
443            }
444            let exact = permutation_order(&perms[generator], generator)
445                .ok_or(GroupError::PermutationOrderOverflow { generator })?;
446            if declared != exact {
447                return Err(GroupError::OrderMismatch {
448                    generator,
449                    declared,
450                    exact,
451                });
452            }
453        }
454        for left in 0..perms.len() {
455            for right in left + 1..perms.len() {
456                if !permutations_commute(&perms[left], &perms[right]) {
457                    return Err(GroupError::NonCommuting { left, right });
458                }
459            }
460        }
461        let order = checked_group_order(&orders).ok_or(GroupError::GroupOrderOverflow)?;
462        // lcm divides the product, so this cannot overflow once `order` fits.
463        let phase_modulus = orders
464            .iter()
465            .try_fold(1usize, |acc, &value| checked_lcm(acc, value as usize))
466            .ok_or(GroupError::GroupOrderOverflow)?;
467        let block_cyclic = detect_block_cyclic(n_qubits, &perms, &orders);
468        let rotations = perms
469            .iter()
470            .map(|p| detect_block_rotation(n_qubits, p))
471            .collect();
472        Ok(Self {
473            n_qubits,
474            perms,
475            orders,
476            order,
477            phase_modulus,
478            block_cyclic,
479            rotations,
480        })
481    }

483    /// 1D chain of `n` sites with periodic boundary conditions.
484    /// Single generator: cyclic shift by one site.
485    pub fn chain_1d(n: usize) -> Self {
486        assert!(n > 0, "chain_1d: n must be positive");
487        let order =
488            u32::try_from(n).unwrap_or_else(|_| panic!("chain_1d: n={n} does not fit in u32"));
489        let perm: Vec<u32> = (0..n)
490            .map(|q| {
491                u32::try_from((q + 1) % n).expect("chain_1d: target index does not fit in u32")
492            })
493            .collect();
494        Self::from_generators(n, vec![perm], vec![order])
495    }

497    /// 2D `lx × ly` torus, qubit at `(i, j)` indexed as `j*lx + i`.
498    /// Two generators: x-shift (i → i+1 mod lx) and y-shift (j → j+1 mod ly).
499    pub fn torus_2d(lx: usize, ly: usize) -> Self {
500        assert!(lx > 0, "torus_2d: lx must be positive");
501        assert!(ly > 0, "torus_2d: ly must be positive");
502        let n = lx
503            .checked_mul(ly)
504            .unwrap_or_else(|| panic!("torus_2d: lx * ly overflow"));
505        validate_site_count(n, "torus_2d");
506        let lx_u32 =
507            u32::try_from(lx).unwrap_or_else(|_| panic!("torus_2d: lx={lx} does not fit in u32"));
508        let ly_u32 =
509            u32::try_from(ly).unwrap_or_else(|_| panic!("torus_2d: ly={ly} does not fit in u32"));
510        let perm_x: Vec<u32> = (0..n)
511            .map(|q| {
512                let (i, j) = (q % lx, q / lx);
513                u32::try_from(j * lx + (i + 1) % lx)
514                    .expect("torus_2d: x-shift target index does not fit in u32")
515            })
516            .collect();
517        let perm_y: Vec<u32> = (0..n)
518            .map(|q| {
519                let (i, j) = (q % lx, q / lx);
520                u32::try_from(((j + 1) % ly) * lx + i)
521                    .expect("torus_2d: y-shift target index does not fit in u32")
522            })
523            .collect();
524        Self::from_generators(n, vec![perm_x, perm_y], vec![lx_u32, ly_u32])
525    }

527    /// 3D `lx × ly × lz` torus, qubit at `(i, j, k)` indexed as
528    /// `k*lx*ly + j*lx + i`.
529    pub fn torus_3d(lx: usize, ly: usize, lz: usize) -> Self {
530        assert!(lx > 0, "torus_3d: lx must be positive");
531        assert!(ly > 0, "torus_3d: ly must be positive");
532        assert!(lz > 0, "torus_3d: lz must be positive");
533        let n = lx
534            .checked_mul(ly)
535            .and_then(|v| v.checked_mul(lz))
536            .unwrap_or_else(|| panic!("torus_3d: lx * ly * lz overflow"));
537        validate_site_count(n, "torus_3d");
538        let lx_u32 =
539            u32::try_from(lx).unwrap_or_else(|_| panic!("torus_3d: lx={lx} does not fit in u32"));
540        let ly_u32 =
541            u32::try_from(ly).unwrap_or_else(|_| panic!("torus_3d: ly={ly} does not fit in u32"));
542        let lz_u32 =
543            u32::try_from(lz).unwrap_or_else(|_| panic!("torus_3d: lz={lz} does not fit in u32"));
544        let perm_x: Vec<u32> = (0..n)
545            .map(|q| {
546                let i = q % lx;
547                let j = (q / lx) % ly;
548                let k = q / (lx * ly);
549                u32::try_from(k * lx * ly + j * lx + (i + 1) % lx)
550                    .expect("torus_3d: x-shift target index does not fit in u32")
551            })
552            .collect();
553        let perm_y: Vec<u32> = (0..n)
554            .map(|q| {
555                let i = q % lx;
556                let j = (q / lx) % ly;
557                let k = q / (lx * ly);
558                u32::try_from(k * lx * ly + ((j + 1) % ly) * lx + i)
559                    .expect("torus_3d: y-shift target index does not fit in u32")
560            })
561            .collect();
562        let perm_z: Vec<u32> = (0..n)
563            .map(|q| {
564                let i = q % lx;
565                let j = (q / lx) % ly;
566                let k = q / (lx * ly);
567                u32::try_from(((k + 1) % lz) * lx * ly + j * lx + i)
568                    .expect("torus_3d: z-shift target index does not fit in u32")
569            })
570            .collect();
571        Self::from_generators(
572            n,
573            vec![perm_x, perm_y, perm_z],
574            vec![lx_u32, ly_u32, lz_u32],
575        )
576    }

578    /// Multi-leg ladder: `l` sites along the chain × `n_legs` legs.
579    /// Single generator: cyclic shift along the chain direction (all
580    /// legs simultaneously). Qubit at `(leg, j)` indexed as
581    /// `leg * l + j`. No translation along the leg axis (legs are
582    /// distinguished).
583    pub fn ladder(l: usize, n_legs: usize) -> Self {
584        assert!(l > 0, "ladder: l must be positive");
585        assert!(n_legs > 0, "ladder: n_legs must be positive");
586        let n = l
587            .checked_mul(n_legs)
588            .unwrap_or_else(|| panic!("ladder: l * n_legs overflow"));
589        validate_site_count(n, "ladder");
590        let l_u32 =
591            u32::try_from(l).unwrap_or_else(|_| panic!("ladder: l={l} does not fit in u32"));
592        let perm: Vec<u32> = (0..n)
593            .map(|q| {
594                let leg = q / l;
595                let j = q % l;
596                u32::try_from(leg * l + (j + 1) % l)
597                    .expect("ladder: shift target index does not fit in u32")
598            })
599            .collect();
600        Self::from_generators(n, vec![perm], vec![l_u32])
601    }

637    /// Apply a single generator's permutation to a Pauli word: for each
638    /// qubit `q` of the input, the `(xbit, zbit)` pair is placed at position
639    /// `perm[q]` of the output.
640    ///
641    /// Does **not** refresh the cached hash. Equality and ordering compare
642    /// the bit planes, so an unhashed word is safe to compare and to keep as
643    /// an intermediate; only words that escape into a hash container need
644    /// `rehash`. The odometer walk applies a generator per group element and
645    /// hashes just the winner.
646    pub(super) fn apply_generator<A, S, const R: bool>(
647        &self,
648        w: &PauliWord<A, S, R>,
649        g: usize,
650    ) -> PauliWord<A, S, R>
651    where
652        A: PauliStorage,
653        S: BuildHasher + Clone + Default + HashFinalize,
654    {
655        if let Some(rot) = &self.rotations[g]
656            && let Some(out) = Self::apply_block_rotation(w, rot)
657        {
658            return out;
659        }
660        let perm = &self.perms[g];
661        let mut out: PauliWord<A, S, R> = PauliWord::new(self.n_qubits);
662        for (q, &pq) in perm.iter().enumerate().take(self.n_qubits) {
663            let xb = w.get_xbit(q);
664            let zb = w.get_zbit(q);
665            if xb {
666                out.set_xbit(pq as usize, true);
667            }
668            if zb {
669                out.set_zbit(pq as usize, true);
670            }
671        }
672        out
673    }
674
675    /// Apply a block-rotation generator as a masked shift of both bit
676    /// planes: `out = ((in << stride) & keep) | ((in & high) >> (block − stride))`.
677    ///
678    /// This is the same permutation as the per-qubit gather, in `O(N/64)`
679    /// word operations instead of `O(N)` bit operations. The cached hash is
680    /// *not* refreshed. Returns `None` on big-endian targets (where the byte
681    /// view of the bit planes is not in bit order) or if the storage is wider
682    /// than [`MAX_ROT_WORDS`], leaving the caller on the general path.
683    pub(super) fn apply_block_rotation<A, S, const R: bool>(
684        w: &PauliWord<A, S, R>,
685        rot: &BlockRotation,
686    ) -> Option<PauliWord<A, S, R>>
687    where
688        A: PauliStorage,
689        S: BuildHasher + Clone + Default + HashFinalize,
690    {
691        if !cfg!(target_endian = "little") || size_of::<A>() > MAX_ROT_WORDS * 8 {
692            return None;
693        }
694        let mut out = *w;
695        for plane in 0..2 {
696            let (src_arr, dst_arr) = if plane == 0 {
697                (&w.xbits.data, &mut out.xbits.data)
698            } else {
699                (&w.zbits.data, &mut out.zbits.data)
700            };
701            let bytes = bytemuck::bytes_of(src_arr);
702            let nw = bytes.len().div_ceil(8);
703            let (mut src, mut shifted, mut wrapped) = (
704                [0u64; MAX_ROT_WORDS],
705                [0u64; MAX_ROT_WORDS],
706                [0u64; MAX_ROT_WORDS],
707            );
708            for (i, chunk) in bytes.chunks(8).enumerate() {
709                let mut b = [0u8; 8];
710                b[..chunk.len()].copy_from_slice(chunk);
711                src[i] = u64::from_le_bytes(b);
712            }
713            shl_words(&src[..nw], &mut shifted[..nw], rot.stride);
714            for (i, s) in src[..nw].iter_mut().enumerate() {
715                *s &= rot.high.get(i).copied().unwrap_or(0);
716            }
717            shr_words(&src[..nw], &mut wrapped[..nw], rot.block - rot.stride);
718            for i in 0..nw {
719                shifted[i] = (shifted[i] & rot.keep.get(i).copied().unwrap_or(0)) | wrapped[i];
720            }
721            let dst = bytemuck::bytes_of_mut(dst_arr);
722            for (i, chunk) in dst.chunks_mut(8).enumerate() {
723                let b = shifted[i].to_le_bytes();
724                let n = chunk.len();
725                chunk.copy_from_slice(&b[..n]);
726            }
727        }
728        Some(out)
729    }

731    /// Odometer step: advance `cur` from the group element with
732    /// mixed-radix index `idx - 1` to the one with index `idx`.
733    ///
734    /// Generator `0` is the fastest-varying digit, so it advances on
735    /// every step; digit `g` advances only when all lower digits roll
736    /// over, i.e. when `idx` is a multiple of `orders[0..=g-1]`. Applying
737    /// generator `g` once always moves digit `g` forward *cyclically*
738    /// (the `orders[g]`-th application is the identity), so a roll-over
739    /// is just one more application — no rebuild from the identity.
740    ///
741    /// Cost: `O(1)` generator applications amortised, hence `O(|G| × N)`
742    /// for a full walk. Leaves the cached hash of `cur` stale.
743    #[inline]
744    fn advance<A, S, const R: bool>(&self, cur: &mut PauliWord<A, S, R>, idx: usize)
745    where
746        A: PauliStorage,
747        S: BuildHasher + Clone + Default + HashFinalize,
748    {
749        let mut p = 1usize;
750        for (g, &o) in self.orders.iter().enumerate() {
751            if o > 1 {
752                *cur = self.apply_generator(cur, g);
753            }
754            p *= o as usize;
755            if !idx.is_multiple_of(p) {
756                break;
757            }
758        }
759    }

782    /// Lex-min canonical representative of `w`'s translation orbit
783    /// under this group.
784    ///
785    /// For chain/ladder layouts this is `O(N)` via the least-rotation
786    /// canonicalizer ([`Self::canonicalize_block_cyclic`]); otherwise it
787    /// walks the full group as a mixed-radix odometer, `O(|G| × N)`.
788    pub fn canonicalize<A, S, const R: bool>(&self, w: &PauliWord<A, S, R>) -> PauliWord<A, S, R>
789    where
790        A: PauliStorage,
791        S: BuildHasher + Clone + Default + HashFinalize,
792    {
793        self.canonicalize_with_index(w).0
794    }

796    /// Lex-min canonical representative `r` of `w` together with the
797    /// **mixed-radix counter** `c = (c_0, c_1, …)` of the group element
798    /// `g` such that `g·r = w`.
799    ///
800    /// In other words: if `r = self.canonicalize(w)`, this returns
801    /// `(r, c)` where applying generator `i` exactly `c[i]` times in
802    /// sequence to `r` produces `w`. It returns the first valid counter
803    /// selected by the deterministic mixed-radix traversal. Counters are
804    /// not unique when `r` has a non-trivial stabilizer (or when the
805    /// combined action has a kernel). The counter is used to compute
806    /// momentum phases by the phase-aware merge routines.
807    ///
808    /// Same cost as [`Self::canonicalize`], plus the counter `Vec`. Hot
809    /// paths should prefer [`Self::canonicalize_with_index`], which is
810    /// allocation-free and indexes a precomputed
811    /// [`Self::character_table`](crate::symmetry::TranslationGroup::character_table).
812    pub fn canonicalize_with_shift<A, S, const R: bool>(
813        &self,
814        w: &PauliWord<A, S, R>,
815    ) -> (PauliWord<A, S, R>, Vec<u32>)
816    where
817        A: PauliStorage,
818        S: BuildHasher + Clone + Default + HashFinalize,
819    {
820        let (rep, idx) = self.canonicalize_with_index(w);
821        (rep, self.counter_from_index(idx))
822    }

824    /// Lex-min canonical representative `r` of `w` together with the
825    /// **mixed-radix index** (generator `0` fastest) of the group element
826    /// `g` such that `g·r = w` — i.e. the index of the counter returned by
827    /// [`Self::canonicalize_with_shift`].
828    ///
829    /// The index is directly usable as a subscript into a
830    /// [`CharacterTable`](crate::symmetry::CharacterTable), which is how the
831    /// phase-aware evolution gets `χ_k(g)` without decoding a counter or
832    /// calling `sin`/`cos` per term.
833    ///
834    /// Cost: `O(N)` for chain/ladder layouts, else `O(|G| × N)`.
835    /// Allocation-free apart from the returned word.
836    pub fn canonicalize_with_index<A, S, const R: bool>(
837        &self,
838        w: &PauliWord<A, S, R>,
839    ) -> (PauliWord<A, S, R>, usize)
840    where
841        A: PauliStorage,
842        S: BuildHasher + Clone + Default + HashFinalize,
843    {
844        assert_eq!(
845            w.n_qubits(),
846            self.n_qubits,
847            "word and group must agree on n_qubits"
848        );
849        match self.block_cyclic {
850            Some(bc) => {
851                let (rep, r, _) = self.canonicalize_block_cyclic(w, bc);
852                (rep, r)
853            }
854            None => {
855                let (rep, idx, _) = self.canonicalize_odometer(w, |_| true);
856                (rep, idx)
857            }
858        }
859    }

861    /// Canonical rep, the index of the group element mapping it back to `w`,
862    /// and the **stabilizer** of `w` checked against `trivial`: returns
863    /// `None` as soon as a stabilizer element `s` (`s·w = w`) with
864    /// `!trivial(index(s))` is found, else `Some((rep, index, |stabilizer|))`.
865    ///
866    /// One traversal gives everything the momentum-sector routines need.
867    /// `trivial` is only consulted on stabilizer elements, which are rare
868    /// (none but the identity for a free orbit).
869    pub(super) fn canonicalize_with_stabilizer<A, S, const R: bool, F>(
870        &self,
871        w: &PauliWord<A, S, R>,
872        trivial: F,
873    ) -> Option<(PauliWord<A, S, R>, usize, usize)>
874    where
875        A: PauliStorage,
876        S: BuildHasher + Clone + Default + HashFinalize,
877        F: Fn(usize) -> bool,
878    {
879        assert_eq!(
880            w.n_qubits(),
881            self.n_qubits,
882            "word and group must agree on n_qubits"
883        );
884        match self.block_cyclic {
885            Some(bc) => {
886                let (rep, r, step) = self.canonicalize_block_cyclic(w, bc);
887                // The stabilizer of a single-generator group is the cyclic
888                // subgroup generated by `g^step`; its characters are all
889                // trivial iff that generator's is.
890                if step < bc.len && !trivial(step) {
891                    return None;
892                }
893                Some((rep, r, bc.len / step))
894            }
895            None => {
896                let (rep, idx, stabilizer) = self.canonicalize_odometer(w, trivial);
897                (stabilizer != 0).then_some((rep, idx, stabilizer))
898            }
899        }
900    }

902    /// Reference canonicalizer: walk the whole group as a mixed-radix
903    /// odometer (see [`Self::advance`]), keeping the first smallest word
904    /// seen. Returns the rep, the index of the group element mapping it
905    /// back to `w`, and the stabilizer size — or `0` for the latter if a
906    /// stabilizer element fails `trivial` (the walk stops there).
907    ///
908    /// `O(|G| × N)`; used for groups without a [`BlockCyclic`] layout, and
909    /// as the test oracle for the fast path.
910    pub(super) fn canonicalize_odometer<A, S, const R: bool, F>(
911        &self,
912        w: &PauliWord<A, S, R>,
913        trivial: F,
914    ) -> (PauliWord<A, S, R>, usize, usize)
915    where
916        A: PauliStorage,
917        S: BuildHasher + Clone + Default + HashFinalize,
918        F: Fn(usize) -> bool,
919    {
920        let mut best = *w;
921        let mut best_idx = 0usize;
922        let mut stabilizer = 1usize;
923        let mut cur = *w;
924        for idx in 1..self.order {
925            self.advance(&mut cur, idx);
926            if cur == *w {
927                if !trivial(idx) {
928                    return (best, 0, 0);
929                }
930                stabilizer += 1;
931            }
932            if cur < best {
933                best = cur;
934                best_idx = idx;
935            }
936        }
937        // `advance` leaves the cached hash stale; the winner escapes to the
938        // caller (and into hash containers), so refresh it here.
939        best.rehash();
940        // The walk found `best = g·w` at index `best_idx`, so `w = g⁻¹·best`
941        // and the element we must report is the inverse.
942        (best, self.invert_index(best_idx), stabilizer)
943    }

945    /// `O(N)` canonicalizer for single-generator cyclic-block groups
946    /// (chain, ladder): returns the same rep as the odometer walk — the
947    /// `Ord`-lex-min of the orbit — the index `r` of the group element
948    /// with `g^r · rep = w` (the odometer's choice), and the smallest
949    /// `step > 0` with `g^step · w = w` (`step == len` for a free orbit).
950    ///
951    /// ## Why this is not one Booth call
952    ///
953    /// `PauliWord`'s `Ord` compares the whole x-bit plane in qubit order,
954    /// *then* the whole z-bit plane. Under a shift by `r`, the comparison
955    /// key is therefore the concatenation
956    /// `rot_r(x_block0) ‖ … ‖ rot_r(z_block0) ‖ …` — `2 · n_blocks` strings
957    /// rotated *together*, not one rotated string, so lex-min over rotations
958    /// is not a single least-rotation problem. (Running Booth on an
959    /// interleaved per-site symbol would be one call, but it minimises a
960    /// different order and so would silently change which orbit member is
961    /// canonical.)
962    ///
963    /// Instead we refine the candidate rotation set plane by plane. After
964    /// each plane the surviving rotations form a residue class
965    /// `{start + i·step}` of size `m = L / step`, because the rotations
966    /// achieving a minimum are exactly those spaced by the *period* of that
967    /// minimal rotation. Plane `p + 1` then compares its own string only at
968    /// those rotations — which is again a least-rotation problem, over `m`
969    /// super-symbols of `step` bits each. Every plane costs `O(L)` symbol
970    /// comparisons of `O(step)` bits = `O(L)`, so the whole call is
971    /// `O(n_blocks · L) = O(N)`, allocation-free apart from the output word.
972    /// The final survivors are one coset of the stabilizer, so `step` is
973    /// its generator.
974    pub(super) fn canonicalize_block_cyclic<A, S, const R: bool>(
975        &self,
976        w: &PauliWord<A, S, R>,
977        bc: BlockCyclic,
978    ) -> (PauliWord<A, S, R>, usize, usize)
979    where
980        A: PauliStorage,
981        S: BuildHasher + Clone + Default + HashFinalize,
982    {
983        let l = bc.len;
984        // Surviving rotations: { (start + i·step) mod l : i < m }, with
985        // step · m == l throughout, and `start < step` (the smallest one).
986        let (mut start, mut step, mut m) = (0usize, 1usize, l);
987        for plane in 0..2 * bc.n_blocks {
988            if m == 1 {
989                break;
990            }
991            let is_x = plane < bc.n_blocks;
992            let base = (if is_x { plane } else { plane - bc.n_blocks }) * l;
993            // Symbol `j` is the run of `step` bits of this plane starting at
994            // rotation offset `start + j·step`.
995            let bit = |j: usize, t: usize| -> bool {
996                let pos = base + (start + j * step + t) % l;
997                if is_x {
998                    w.get_xbit(pos)
999                } else {
1000                    w.get_zbit(pos)
1001                }
1002            };
1003            let cmp = |a: usize, b: usize| -> std::cmp::Ordering {
1004                for t in 0..step {
1005                    let (x, y) = (bit(a, t), bit(b, t));
1006                    if x != y {
1007                        // `false < true`, matching bit-slice lex order.
1008                        return x.cmp(&y);
1009                    }
1010                }
1011                std::cmp::Ordering::Equal
1012            };
1013            let j0 = least_rotation(m, &cmp);
1014            let period = minimal_rotation_period(m, j0, &cmp);
1015            start = (start + j0 * step) % l;
1016            step *= period;
1017            m /= period;
1018            start %= step; // smallest member of the surviving residue class
1019        }
1020        // Tie-break exactly as the odometer does: it keeps the *first*
1021        // minimal word it meets, i.e. the smallest number of generator
1022        // applications `idx = (l − r) mod l`. That is `r = 0` when `r = 0`
1023        // survives, and otherwise the largest surviving `r`.
1024        let r = if start == 0 {
1025            0
1026        } else {
1027            start + (m - 1) * step
1028        };
1029        // rep = g^{−r}·w, i.e. rep[base + j] = w[base + (j + r) mod l].
1030        let mut rep: PauliWord<A, S, R> = PauliWord::new(self.n_qubits);
1031        for b in 0..bc.n_blocks {
1032            let base = b * l;
1033            for j in 0..l {
1034                let src = base + (j + r) % l;
1035                if w.get_xbit(src) {
1036                    rep.set_xbit(base + j, true);
1037                }
1038                if w.get_zbit(src) {
1039                    rep.set_zbit(base + j, true);
1040                }
1041            }
1042        }
1043        rep.rehash();
1044        (rep, r, step)
1045    }

1047    /// Decode a group-element index (mixed-radix, generator `0` fastest)
1048    /// into its per-generator counter.
1049    pub fn counter_from_index(&self, idx: usize) -> Vec<u32> {
1050        let mut rem = idx;
1051        let mut counter: Vec<u32> = Vec::with_capacity(self.orders.len());
1052        for &o in &self.orders {
1053            counter.push((rem % o as usize) as u32);
1054            rem /= o as usize;
1055        }
1056        counter
1057    }

1059    /// Index of the inverse of the group element with index `idx`. In an
1060    /// abelian product of cyclic groups that is `(orders[g] − c[g]) mod
1061    /// orders[g]` componentwise.
1062    pub(super) fn invert_index(&self, idx: usize) -> usize {
1063        let mut rem = idx;
1064        let mut out = 0usize;
1065        let mut stride = 1usize;
1066        for &o in &self.orders {
1067            let o = o as usize;
1068            let c = rem % o;
1069            rem /= o;
1070            out += ((o - c) % o) * stride;
1071            stride *= o;
1072        }
1073        out
1074    }

1078    /// element).

1081    /// has a kernel; this is not an iterator over distinct orbit members.
1082    pub fn orbit<'a, A, S, const R: bool>(

1084        w: &'a PauliWord<A, S, R>,

1091    }

1094pub(super) struct GroupOrbit<'a, A, S, const R: bool>
1095where
1096    A: PauliStorage,
1097{
1098    group: &'a TranslationGroup,
1099    current: PauliWord<A, S, R>,
1100    counter: Vec<u32>,
1101    remaining: usize,
1102}
1103
1104impl<A, S, const R: bool> Iterator for GroupOrbit<'_, A, S, R>
1105where
1106    A: PauliStorage,
1107    S: BuildHasher + Clone + Default + HashFinalize,
1108{
1109    type Item = (PauliWord<A, S, R>, Vec<u32>);
1110
1111    fn next(&mut self) -> Option<Self::Item> {
1112        if self.remaining == 0 {
1113            return None;
1114        }
1115        // `apply_generator` skips hashing; yielded words may become hash
1116        // keys, so hash each one on the way out.
1117        let mut word = self.current;
1118        word.rehash();
1119        let item = (word, self.counter.clone());
1120        self.remaining -= 1;
1121        if self.remaining == 0 {
1122            return Some(item);
1123        }
1124        for g in 0..self.group.orders.len() {
1125            if self.group.orders[g] == 1 {
1126                continue;
1127            }
1128            self.current = self.group.apply_generator(&self.current, g);
1129            self.counter[g] += 1;
1130            if self.counter[g] < self.group.orders[g] {
1131                break;
1132            }
1133            self.counter[g] = 0;
1134        }
1135        Some(item)
1136    }
