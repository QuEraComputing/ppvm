4//! Python bindings for the symmetry-merging primitive.
5//!
6//! Exposes:
7//! - [`TranslationGroup`] PyO3 class with constructors for 1D, 2D, 3D
8//!   tori and multi-leg ladders, plus a generic generator-list path.
9//! - [`canonicalize_basis_arr`] / [`canonicalize_basis_arr_complex`] free
10//!   functions that merge the numpy `(basis_arr, coeffs)` representation
11//!   used by `Lindbladian.pc_step_arr`.

23type PyPauliMap<'py> = (Bound<'py, PyArray2<u8>>, Bound<'py, PyArray1<f64>>);

26/// A finite abelian symmetry group acting on qubit positions by
27/// permutations. Use this to merge translation-equivalent Pauli strings
28/// in either the `Lindbladian.pc_step_arr` basis or the `PauliSum`
29/// dictionary, reducing per-step memory by up to `|G|×`.
30///
31/// Build via the static methods:
32/// - `TranslationGroup.chain_1d(n)` — 1D chain of `n` sites with PBC.
33/// - `TranslationGroup.torus_2d(lx, ly)` — 2D torus; qubit `(i, j)` at
34///   index `j*lx + i`.
35/// - `TranslationGroup.torus_3d(lx, ly, lz)` — 3D torus; qubit
36///   `(i, j, k)` at index `k*lx*ly + j*lx + i`.
37/// - `TranslationGroup.ladder(l, n_legs)` — `n_legs`-leg ladder of `l`
38///   sites, translation only along chain direction; qubit `(leg, j)` at
39///   index `leg*l + j`.
40/// - `TranslationGroup.from_generators(n_qubits, perms, orders)` —
41///   arbitrary list of generator permutations + cyclic orders.
42#[pyclass(frozen)]
43pub struct TranslationGroup {
44    pub(crate) inner: core_sym::TranslationGroup,
45}
46
47impl TranslationGroup {
48    /// Accessor for the underlying [`ppvm_pauli_sum::symmetry::TranslationGroup`].
49    /// Used by other crate-internal modules (e.g. the PauliSum interface
50    /// macro) to call into the core merging API.
51    pub fn core(&self) -> &core_sym::TranslationGroup {
52        &self.inner
53    }
54}

56/// Validate lattice extents before handing them to a core constructor,
57/// which asserts these preconditions rather than reporting them. Each
58/// extent must be positive and `u32`-addressable, and the qubit count
59/// (their product) must not overflow.
60fn check_lattice(dims: &[(&str, usize)]) -> PyResult<()> {
61    let mut n_qubits = 1usize;
62    for &(name, dim) in dims {
63        if dim == 0 {
64            return Err(PyValueError::new_err(format!("{name} must be positive")));
65        }
66        n_qubits = n_qubits
67            .checked_mul(dim)
68            .ok_or_else(|| PyValueError::new_err(format!("qubit count overflows: {name}={dim}")))?;
69    }
70    u32::try_from(n_qubits - 1).map_err(|_| {
71        PyValueError::new_err(format!(
72            "qubit count {n_qubits} exceeds the u32-addressable range"
73        ))
74    })?;
75    Ok(())
76}

78#[pymethods]
79impl TranslationGroup {
80    #[staticmethod]
81    pub fn chain_1d(n: usize) -> PyResult<Self> {
82        check_lattice(&[("n", n)])?;
83        Ok(Self {
84            inner: core_sym::TranslationGroup::chain_1d(n),
85        })
86    }
87
88    #[staticmethod]
89    pub fn torus_2d(lx: usize, ly: usize) -> PyResult<Self> {
90        check_lattice(&[("lx", lx), ("ly", ly)])?;
91        Ok(Self {
92            inner: core_sym::TranslationGroup::torus_2d(lx, ly),
93        })
94    }
95
96    #[staticmethod]
97    pub fn torus_3d(lx: usize, ly: usize, lz: usize) -> PyResult<Self> {
98        check_lattice(&[("lx", lx), ("ly", ly), ("lz", lz)])?;
99        Ok(Self {
100            inner: core_sym::TranslationGroup::torus_3d(lx, ly, lz),
101        })
102    }
103
104    #[staticmethod]
105    pub fn ladder(l: usize, n_legs: usize) -> PyResult<Self> {
106        check_lattice(&[("l", l), ("n_legs", n_legs)])?;
107        Ok(Self {
108            inner: core_sym::TranslationGroup::ladder(l, n_legs),
109        })
110    }
111
112    /// Every precondition — permutation shape and validity, exact cyclic
113    /// orders, pairwise commutation — is checked by the core's fallible
114    /// constructor, so bad generators raise `ValueError` here instead of
115    /// aborting.
116    #[staticmethod]
117    pub fn from_generators(
118        n_qubits: usize,
119        perms: Vec<Vec<u32>>,
120        orders: Vec<u32>,
121    ) -> PyResult<Self> {
122        let inner = core_sym::TranslationGroup::try_from_generators(n_qubits, perms, orders)
123            .map_err(|e| PyValueError::new_err(e.to_string()))?;
124        Ok(Self { inner })
125    }
126
127    /// Number of qubits this group acts on.
128    #[getter]
129    pub fn n_qubits(&self) -> usize {
130        self.inner.n_qubits()
131    }
132
133    /// Number of generators (rank as an abelian product group).
134    #[getter]
135    pub fn n_generators(&self) -> usize {
136        self.inner.n_generators()
137    }
138
139    /// Total group order: product of generator orders.
140    #[getter]
141    pub fn order(&self) -> usize {
142        self.inner.order()
143    }
144
145    /// Return the canonical (lex-min) orbit representative of `pauli`.
146    /// `pauli` is a length-`n_qubits` uint8 array with the encoding
147    /// `0=I, 1=X, 2=Z, 3=Y`. Result is the same shape.
148    pub fn canonicalize<'py>(
149        &self,
150        py: Python<'py>,
151        pauli: PyReadonlyArray1<'py, u8>,
152    ) -> PyResult<Bound<'py, PyArray1<u8>>> {
153        let codes = pauli.as_slice()?;
154        if codes.len() != self.inner.n_qubits() {
155            return Err(PyValueError::new_err(format!(
156                "pauli has length {} but group expects {} qubits",
157                codes.len(),
158                self.inner.n_qubits()
159            )));
160        }
161        let mut out = vec![0u8; codes.len()];
162        with_width!(codes.len(), C => {
163            let w = word_from_codes::<C>(codes).map_err(|e| PyValueError::new_err(e.to_string()))?;
164            let canon = self.inner.canonicalize(&w);
165            codes_from_word(&canon, &mut out);
166            PyResult::Ok(())
167        })?;
168        Ok(out.into_pyarray(py))
169    }
170}

172/// Phase-aware merge of a complex-coefficient `(basis_arr, coeffs)`
173/// Pauli sum into orbit-rep form, projected onto momentum sector
174/// `momentum`.
175///
176/// `momentum` is a length-`group.n_generators` integer array of mode
177/// indices; the wavenumber along generator `g` is
178/// `2π · momentum[g] / group.generator_order(g)`. Use `momentum=[0, …]`
179/// for the trivial (k=0) sector — equivalent to plain merging modulo the
180/// `1/|orbit|` normalization this projection applies (it *averages* over
181/// each orbit's distinct members; `PauliSum.momentum_merge` sums).
182///
183/// If the input is **not** in sector `momentum`, the projection
184/// silently throws away the other components. Use
185/// [`check_momentum_sector_arr`] beforehand to validate.
186#[pyfunction]
187pub fn canonicalize_basis_arr_complex<'py>(
188    py: Python<'py>,
189    basis: PyReadonlyArray2<'py, u8>,
190    coeffs: PyReadonlyArray1<'py, Complex64>,
191    group: &TranslationGroup,
192    momentum: PyReadonlyArray1<'py, i32>,
193) -> PyResult<PyPauliMapComplex<'py>> {
194    let basis_view = basis.as_array();
195    let n_q = group.inner.n_qubits();
196    check_group_width(&basis_view, n_q)?;
197    let coeffs_slice = coeffs.as_slice()?;
198    check_coeffs_len(coeffs_slice.len(), basis_view.shape()[0])?;
199    let k_slice = momentum.as_slice()?;
200    check_momentum_len(k_slice.len(), group.inner.n_generators())?;
201    with_width!(n_q, C => {
202        let mut basis_words = decode_basis::<C>(&basis_view, n_q)?;
203        let mut coeffs_vec: Vec<Complex<f64>> = coeffs_slice
204            .iter()
205            .map(|c| Complex::new(c.re, c.im))
206            .collect();
207
208        core_sym::canonicalize_pauli_sum_complex(
209            &mut basis_words,
210            &mut coeffs_vec,
211            &group.inner,
212            k_slice,
213        );
214
215        let out_coeffs: Vec<Complex64> = coeffs_vec
216            .iter()
217            .map(|c| Complex64::new(c.re, c.im))
218            .collect();
219        let basis_arr = encode_basis(py, &basis_words, n_q)?;
220        Ok((basis_arr, out_coeffs.into_pyarray(py)))
221    })
222}

224/// Verify that a `(basis_arr, complex_coeffs)` Pauli sum lies in the
225/// momentum sector `momentum` under `group`. Returns `None` on pass,
226/// raises a `ValueError` with diagnostic info on fail.
227///
228/// `tol` is the relative tolerance on coefficient comparison; default
229/// `1e-8`.
230#[pyfunction]
231#[pyo3(signature = (basis, coeffs, group, momentum, tol = 1e-8))]
232pub fn check_momentum_sector_arr<'py>(
233    basis: PyReadonlyArray2<'py, u8>,
234    coeffs: PyReadonlyArray1<'py, Complex64>,
235    group: &TranslationGroup,
236    momentum: PyReadonlyArray1<'py, i32>,
237    tol: f64,
238) -> PyResult<()> {
239    let basis_view = basis.as_array();
240    let n_q = group.inner.n_qubits();
241    check_group_width(&basis_view, n_q)?;
242    let coeffs_slice = coeffs.as_slice()?;
243    check_coeffs_len(coeffs_slice.len(), basis_view.shape()[0])?;
244    let k_slice = momentum.as_slice()?;
245    check_momentum_len(k_slice.len(), group.inner.n_generators())?;
246    with_width!(n_q, C => {
247        let basis_words = decode_basis::<C>(&basis_view, n_q)?;
248        let coeffs_vec: Vec<Complex<f64>> = coeffs_slice
249            .iter()
250            .map(|c| Complex::new(c.re, c.im))
251            .collect();
252        core_sym::check_momentum_sector(&basis_words, &coeffs_vec, &group.inner, k_slice, tol)
253            .map_err(|e| PyValueError::new_err(format!("{e}")))
254    })
255}

257/// Merge a `(basis_arr, coeffs)` Pauli sum (the representation used by
258/// `Lindbladian.pc_step_arr`) into orbit-representative form.
259/// Each row of `basis_arr` is replaced by its canonical
260/// representative; coefficients of rows collapsing to the same rep are
261/// summed.
262///
263/// Returns `(merged_basis_arr, merged_coeffs)`. Output length ≤ input
264/// length.
265///
266/// For dynamics that commute with `group` and initial states that are
267/// `group`-invariant, this preserves all `group`-invariant expectation
268/// values (Theorem 1 of Teng et al., arXiv:2512.12094).
269#[pyfunction]
270pub fn canonicalize_basis_arr<'py>(
271    py: Python<'py>,
272    basis: PyReadonlyArray2<'py, u8>,
273    coeffs: PyReadonlyArray1<'py, f64>,
274    group: &TranslationGroup,
275) -> PyResult<PyPauliMap<'py>> {
276    let basis_view = basis.as_array();
277    let n_q = group.inner.n_qubits();
278    check_group_width(&basis_view, n_q)?;
279    let coeffs_slice = coeffs.as_slice()?;
280    check_coeffs_len(coeffs_slice.len(), basis_view.shape()[0])?;
281
282    with_width!(n_q, C => {
283        let mut basis_words = decode_basis::<C>(&basis_view, n_q)?;
284        let mut coeffs_vec = coeffs_slice.to_vec();
285
286        core_sym::canonicalize_pauli_sum(&mut basis_words, &mut coeffs_vec, &group.inner);
287
288        let basis_arr = encode_basis(py, &basis_words, n_q)?;
289        Ok((basis_arr, coeffs_vec.into_pyarray(py)))
290    })
291}
