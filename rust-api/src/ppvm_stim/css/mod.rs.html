4//! Exact sampling of CSS circuits with atom loss and leakage, many shots per machine word.
5//!
6//! A circuit is *CSS up to Hadamards* when conjugating some of its qubit segments (between resets
7//! and measurements) by `H` turns every two-qubit gate into a `CX`. Preparations and
8//! measurements may use either basis. Following Gidney & Jones, *CSS Circuits: useful quantum
9//! circuits that can be simulated with O(1) cost per gate* ([arXiv:2611.14944]), the all-zero
10//! record is then a valid noiseless outcome for every pattern of gates cancelled by loss, so a
11//! Pauli frame per shot samples loss and leakage exactly. Gates touching a lost or leaked qubit
12//! are skipped, with the same semantics as the generalized-tableau backend ([`sample`]).
13//!
14//! Supported: `R`, `RX`, `M`, `MX`, `MR`, `MRX`, `MPAD`, `X`, `Y`, `Z`, `H`, `SQRT_Y`,
15//! `SQRT_Y_DAG`, `CX`, `CZ`, classical feedback (`CX` / `CY` / `CZ` controlled by `rec[-k]`),
16//! Pauli noise, measurement flips, `REPEAT`, and the `I_ERROR` loss, leakage and reset tags.
17//!
18//! ```
19//! use ppvm_stim::{css::CssProgram, parse_extended};
20//!
21//! let program = parse_extended("R 0 1\nH 0\nI_ERROR[loss](0.1) 0\nCX 0 1\nM 0 1")?;
22//! let css = CssProgram::compile(&program)?;
23//! let shots = css.sample(1000, 7); // 1000 x 2 bytes: 0 / 1 outcome, 2 = lost
24//! assert_eq!(shots.len(), 2000);
25//! # Ok::<(), Box<dyn std::error::Error>>(())
26//! ```
27//!
28//! [arXiv:2611.14944]: https://arxiv.org/abs/2611.14944
29//! [`sample`]: crate::sample

37/// A Stim program compiled for the CSS frame sampler; see the [module docs](self).
38#[derive(Debug, Clone)]
39pub struct CssProgram {
40    ops: Vec<compile::Op>,
41    num_qubits: usize,
42    num_measurements: usize,
43}
44
45impl CssProgram {
46    /// Number of measurement-record entries per shot.
47    pub fn num_measurements(&self) -> usize {
48        self.num_measurements
49    }
50}
51
52/// Why a program cannot run on the CSS frame sampler.
53#[derive(Debug, PartialEq, thiserror::Error)]
54pub enum CssError {
55    /// Invalid program, or an instruction outside the CSS gate set.
56    #[error(transparent)]
57    Exec(#[from] ExecError),
58    /// The two-qubit gates admit no consistent choice of Hadamard-conjugated segments.
59    #[error("the circuit is not CSS up to Hadamard gates (line {line})")]
60    NotCss { line: usize },
61}
