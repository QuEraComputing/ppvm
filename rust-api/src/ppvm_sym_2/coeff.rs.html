13impl Coefficient for Term {
14    /// Flips the stored signs in place (or does nothing for `+1`); never a
15    /// `Term` product. Panics on a bare variable.
16    #[inline]
17    fn mul_sign(&self, sign: i8) -> Self {
18        let mut ret = self.clone();
19        ret.mul_sign_assign(sign);
20        ret
21    }
22
23    #[inline]
24    fn mul_sign_assign(&mut self, sign: i8) {
25        if sign < 0 {
26            self.negate();
27        }
28    }
29
30    /// `|c|` for a constant (including a `Sum` with no monomials), `+∞` otherwise.
31    ///
32    /// So a `CoefficientThreshold` policy never drops a symbolic coefficient;
33    /// symbolic terms truncate through `max_sin`/`min_eps` instead. This breaks the
34    /// law `N(x) == 0 iff x == 0`, but no absolute value exists on this ring, and an
35    /// `ℓ¹` norm (a usable substitute) would start dropping symbolic coefficients.
36    fn magnitude(&self) -> f64 {
37        self.as_const().map_or(f64::INFINITY, f64::abs)
38    }
39}

41/// `x_u ↦ (sin x_u, cos x_u)`. The ring is the free polynomial ring, so
42/// `sin² + cos² = 1` holds only after evaluation.
43impl Angle<Term> for Var {
44    #[inline]
45    fn sin_cos(&self) -> (Term, Term) {
46        (self.sin(), self.cos())
47    }
48}

50/// An `f64` angle on a symbolic sum, folded to two constants.
51impl Angle<Term> for f64 {
52    #[inline]
53    fn sin_cos(&self) -> (Term, Term) {
54        let (s, c) = f64::sin_cos(*self);
55        (Term::from_f64(s), Term::from_f64(c))
56    }
57}

59impl Term {
60    /// Multiply by `i^phase`: `eval_complex(x.mul_phase(k)) == iᵏ · eval_complex(x)`.
61    /// Even phases scale in place; odd ones flip every key's `i` bit and rebuild.
62    #[inline]
63    pub fn mul_phase(&self, phase: u8) -> Self {
64        match (&self.inner, phase & 3) {
65            (_, 0) => self.clone(),
66            (_, 2) => -self.clone(),
67            (_, k) => self.rotate_phase(k),
68        }
69    }
70
71    /// [`Term::mul_phase`] for odd `k`.
72    fn rotate_phase(&self, k: u8) -> Self {
73        let relabel = |p: &Prod, c: f64| {
74            let mut q = p.clone();
75            let sign = q.mul_phase(k);
76            (q, c * sign)
77        };
78        match self.inner {
79            Inner::Sum(ref s) => {
80                let mut ret = Sum::new();
81                for (p, c) in s.iter().chain([(&ONE, s.c0)]) {
82                    let (q, c) = relabel(p, c);
83                    ret.add_term(q, c, self.max_sin, self.min_eps);
84                }
85                self.with_inner(Inner::Sum(ret))
86            }
87            Inner::Const(c) => {
88                let (q, c) = relabel(&ONE, c);
89                self.with_inner(Inner::truncated(q, c, self.max_sin, self.min_eps))
90            }
91            Inner::One(ref p, c) => {
92                let (q, c) = relabel(p, c);
93                self.with_inner(Inner::truncated(q, c, self.max_sin, self.min_eps))
94            }
95        }
96    }
97}

99/// `i` is `One(i, 1.0)`; `i·i` normalises back to `Const(−1.0)`.
100impl ImaginaryUnit for Term {
101    fn imaginary_unit() -> Self {
102        Term::from_f64(1.0).mul_phase(1)
103    }
104
105    /// Folds `i` into the monomials rather than running a `Term` product.
106    fn mul_i(&self) -> Self {
107        self.mul_phase(1)
108    }
109
110    /// Folds `iᵏ` into the monomials.
111    fn mul_i_pow(&self, k: u8) -> Self {
112        self.mul_phase(k & 3)
113    }
114}

116/// `i ↦ −i`: negates every imaginary monomial in place, since no key changes.
117impl Conjugate for Term {
118    fn conj(&self) -> Self {
119        let mut ret = self.clone();
120        match ret.inner {
121            Inner::Sum(ref mut s) => {
122                if let Some(maps) = &mut s.maps {
123                    for (_, c) in maps.terms.iter_mut().filter(|(p, _)| p.imag) {
124                        *c = -*c;
125                    }
126                }
127            }
128            Inner::One(ref p, ref mut c) if p.imag => *c = -*c,
129            Inner::One(..) | Inner::Const(_) => {}
130        }
131        ret
132    }
133}

135/// Folds from the first summand, so its truncation settings apply, as for `a + b
136/// + …`; an empty iterator gives `Const(0.0)` with the default settings.
137impl std::iter::Sum for Term {
138    #[inline]
139    fn sum<I: Iterator<Item = Self>>(mut iter: I) -> Self {
140        let mut total = iter.next().unwrap_or_else(|| Term::from_f64(0.0));
141        for t in iter {
142            total += t;
143        }
144        total
145    }
146}

148/// Exact in `f64`: scales the coefficients by `0.5` without pruning, so
149/// `x.half() + x.half() == x`.
150impl ppvm_traits_2::Halvable for Term {
151    #[inline]
152    fn half(&self) -> Self {
153        let mut ret = self.clone();
154        match ret.inner {
155            Inner::Sum(ref mut s) => s.scale(0.5),
156            Inner::One(_, ref mut c) | Inner::Const(ref mut c) => *c *= 0.5,
157        }
158        ret
159    }
160}

184/// `is_zero()` holds for a constant (`Const`, or a `Sum` with no monomials) with `|c| <= min_eps`.
185impl num::Zero for Term {
186    fn zero() -> Self {
187        Term::from_f64(0.0)
188    }
189
190    fn is_zero(&self) -> bool {
191        self.as_const().is_some_and(|c| c.abs() <= self.min_eps)
192    }
193}

208    /// Constants in any form are zero at or below `min_eps` and have `|c|` as their
209    /// magnitude; symbolic forms never are zero and never get dropped by a threshold.
210    #[test]
211    fn constants_in_any_form_are_zero_and_have_a_magnitude() {
212        assert!(Term::from_f64(0.0).is_zero());
213        assert_eq!(Term::from_f64(-1e-30).magnitude(), 1e-30);
214        let mut t = Var(0).sin();
215        t += Var(0).sin() * -1.0;
216        assert!(matches!(t.inner(), Inner::Sum(_)));
217        assert!(t.is_zero(), "a cancelled `Sum` is zero");
218        assert_eq!(t.magnitude(), 0.0);
219        t += -2.5;
220        assert!(!t.is_zero());
221        assert_eq!(t.magnitude(), 2.5);
222        let mut tiny = Var(0).sin();
223        tiny.set_min_eps(0.0);
224        tiny *= 1e-30;
225        assert!(!tiny.is_zero());
226        assert_eq!(tiny.magnitude(), f64::INFINITY);
227        // With the default `min_eps` the product truncates to an exact zero.
228        assert!((Var(0).sin() * 1e-30).is_zero());
229    }

231    #[test]
232    fn equality_looks_through_the_representation() {
233        let with = |inner| Term {
234            inner,
235            max_sin: usize::MAX,
236            min_eps: f64::EPSILON,
237        };
238        let mut c0_only = Sum::new();
239        c0_only.c0 = 1.0;
240        assert_eq!(with(Inner::Sum(c0_only.clone())), Term::from(1.0));
241        assert_eq!(Term::from(1.0), with(Inner::Sum(c0_only)));
242        assert_eq!(with(Inner::Sum(Sum::new())), Term::from(0.0));
243
244        let x = Var(0).sin() * 2.0;
245        let mut one_entry = Term::from(0.0) + x.clone();
246        assert!(matches!(one_entry.inner(), Inner::Sum(_)));
247        assert_eq!(one_entry, x);
248        assert_eq!(x, one_entry);
249        assert_ne!(one_entry, x.mul_phase(1), "the `i` bit is part of the key");
250        one_entry += 1.0;
251        assert_ne!(one_entry, x, "an extra c₀ differs");
252        assert_ne!(Var(0).sin() * 3.0, x, "coefficients differ");
253        assert_ne!(Var(1).sin() * 2.0, x, "keys differ");
254
255        // `x − x` is the zero constant in every form.
256        let mut t = Var(0).sin() + Var(1).cos();
257        t += (Var(0).sin() + Var(1).cos()) * -1.0;
258        assert_eq!(t, Term::from(0.0));
259        assert_eq!(
260            Term::imaginary_unit() * Term::imaginary_unit(),
261            -Term::from(1.0)
262        );
263
264        assert_ne!(
265            Term::from(1.0),
266            Term::from(1.0) + x.clone(),
267            "extra monomials differ"
268        );
269        assert_ne!(Term::from(1.0) + x.clone(), Term::from(1.0));
270        let mut tight = Term::from(1.0);
271        tight.set_max_sin(2);
272        assert_ne!(tight, Term::from(1.0), "truncation settings are compared");
273    }

300        let x = Var(0).sin();
301        let s = Term::from(0.5) + Var(1).cos();
302        let mut c0_only = Term::from(0.0) + Var(1).cos();
303        c0_only *= 0.0;

323    pub(crate) const PT: [f64; 41] = {
324        let mut v = [0.0; 41];
325        v[0] = 0.3;
326        v[1] = 1.1;
327        v[40] = -0.7;
328        v
329    };

331    pub(crate) fn ev(t: &Term) -> Complex<f64> {
332        t.eval_complex(&PT).unwrap()
333    }
334
335    pub(crate) fn close(a: Complex<f64>, b: Complex<f64>) -> bool {
336        (a - b).norm() < 1e-12 * (1.0 + a.norm().max(b.norm()))
337    }

339    /// `eval_complex` is a ring homomorphism to `ℂ` commuting with `conj`,
340    /// `mul_phase` and `mul_sign`; `eval` is its real part, and borrowed ops
341    /// match owned ones.
342    #[test]
343    fn evaluation_is_a_ring_homomorphism() {
344        let i = Term::imaginary_unit();
345        let forms = forms();
346        for a in &forms {
347            let ea = ev(a);
348            assert!((a.eval(&PT).unwrap() - ea.re).abs() < 1e-12, "eval of {a}");
349            assert!(close(ev(&a.conj()), ea.conj()), "conj of {a}");
350            assert!(close(ev(&(a.clone() + 1.5)), ea + 1.5), "{a} + 1.5");
351            assert!(close(ev(&(1.5 + a.clone())), ea + 1.5), "1.5 + {a}");
352            assert!(close(ev(&(a.clone() - 1.5)), ea - 1.5), "{a} - 1.5");
353            assert!(close(ev(&(a.clone() * -3.0)), ea * -3.0), "{a} · -3");
354            assert!(close(ev(&(-3.0 * a.clone())), ea * -3.0), "-3 · {a}");
355            assert!(close(ev(&-a.clone()), -ea), "-{a}");
356            assert_eq!(a.mul_sign(-1), -a.clone(), "{a}");
357            assert_eq!(a.mul_sign(1), *a, "{a}");
358            let mut b = a.clone();
359            b.mul_sign_assign(-1);
360            assert_eq!(b, a.mul_sign(-1));
361            for k in 0..4u8 {
362                assert!(close(ev(&a.mul_phase(k)), ea * Complex::i().powu(k as u32)));
363            }
364            assert_eq!(a.clone() * i.clone(), a.mul_phase(1), "{a} · i");
365            for b in &forms {
366                let eb = ev(b);
367                assert!(close(ev(&(a.clone() + b.clone())), ea + eb), "{a} + {b}");
368                assert!(close(ev(&(a.clone() - b.clone())), ea - eb), "{a} - {b}");
369                assert!(close(ev(&(a.clone() * b.clone())), ea * eb), "{a} · {b}");
370
371                let mut by_ref = a.clone();
372                by_ref += b;
373                assert_eq!(by_ref, a.clone() + b.clone(), "{a} += &{b}");
374                let mut by_ref = a.clone();
375                by_ref -= b;
376                assert_eq!(by_ref, a.clone() - b.clone(), "{a} -= &{b}");
377                let mut by_ref = a.clone();
378                by_ref *= b;
379                assert_eq!(by_ref, a.clone() * b.clone(), "{a} *= &{b}");
380            }
381        }
382        assert_eq!(i.conj(), -i);
383    }

385    /// Associativity and distributivity hold in value for every triple of forms.
386    #[test]
387    fn ring_axioms_hold_after_evaluation() {
388        let forms = forms();
389        for a in &forms {
390            for b in &forms {
391                for c in &forms {
392                    let (a, b, c) = (a.clone(), b.clone(), c.clone());
393                    let abc = ev(&((a.clone() * b.clone()) * c.clone()));
394                    assert!(close(abc, ev(&(a.clone() * (b.clone() * c.clone())))));
395                    let lhs = ev(&(a.clone() * (b.clone() + c.clone())));
396                    assert!(close(
397                        lhs,
398                        ev(&(a.clone() * b.clone() + a.clone() * c.clone()))
399                    ));
400                    let lhs = ev(&((a.clone() + b.clone()) + c.clone()));
401                    assert!(close(lhs, ev(&(a + (b + c)))));
402                }
403            }
404        }
405    }

455    /// The identity monomial never appears as a key or inside `One`.
456    #[test]
457    fn identity_monomial_is_always_normalised() {
458        let i = Term::imaginary_unit();
459        assert!(matches!(i.inner(), Inner::One(p, c) if p.is_imag() && *c == 1.0));
460        assert_eq!(i.clone() * i.clone(), -Term::from(1.0));
461        assert_eq!(i.mul_phase(3), Term::from(1.0));
462        assert_eq!(Term::from(2.0).mul_i_pow(0), Term::from(2.0));
463        assert_eq!(Term::from(2.0).mul_i_pow(2), Term::from(-2.0));
464        let no_identity_key = |t: &Term| match t.inner() {
465            Inner::Sum(s) => s.iter().all(|(p, _)| !p.is_identity()),
466            Inner::One(p, _) => !p.is_identity(),
467            _ => true,
468        };
469        for a in forms() {
470            for b in forms() {
471                let ab = a.clone() * b.clone();
472                assert!(no_identity_key(&ab), "{a} · {b}");
473                assert!(no_identity_key(&(a.clone() + b.clone())), "{a} + {b}");
474                assert!(no_identity_key(&ab.mul_phase(1)), "i · {a} · {b}");
475            }
476        }
477    }
