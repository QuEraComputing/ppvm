4//! Evaluation at concrete angles. Within a monomial the fold multiplies the
5//! `sin` factors, then the `cos` factors, each in ascending variable order;
6//! that order fixes the floating-point result and is part of the contract.
7
8use std::mem::MaybeUninit;
9
10use num::Complex;
11
12use crate::prod::Prod;
13use crate::sum::Sum;
14use crate::term::{Inner, Term};

16/// Evaluation needed a variable `vals` has no value for: `vals.len() <= k`.
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub struct MissingVariable(pub u32);
19
20impl std::fmt::Display for MissingVariable {
21    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
22        write!(f, "variable %{} not found", self.0)
23    }
24}
25
26impl std::error::Error for MissingVariable {}
27
28/// Variables [`AngleCache`] memoizes; higher ids are recomputed per access.
29const INLINE_VARS: usize = 32;
30
31/// Per-call memo of `sin`/`cos` of each variable, so a [`Sum`] with many
32/// monomials computes each transcendental once. Bit-identical to the uncached
33/// fold, and bounds-checked per access so the same [`MissingVariable`] is
34/// reported first. Slots stay uninitialized until their validity bit is set.
35struct AngleCache<'a> {
36    vals: &'a [f64],
37    /// Bit `i` set <=> the corresponding slot is initialized.
38    sin_valid: u32,
39    cos_valid: u32,
40    sin: [MaybeUninit<f64>; INLINE_VARS],
41    cos: [MaybeUninit<f64>; INLINE_VARS],
42}
43
44impl<'a> AngleCache<'a> {
45    #[inline]
46    fn new(vals: &'a [f64]) -> Self {
47        Self {
48            vals,
49            sin_valid: 0,
50            cos_valid: 0,
51            sin: [MaybeUninit::uninit(); INLINE_VARS],
52            cos: [MaybeUninit::uninit(); INLINE_VARS],
53        }
54    }
55
56    /// `sin` or `cos` of `vals[k]`, memoized for `k < INLINE_VARS`.
57    #[inline]
58    fn get(&mut self, k: u32, trig: Trig) -> Result<f64, MissingVariable> {
59        let i = k as usize;
60        let v = *self.vals.get(i).ok_or(MissingVariable(k))?;
61        let f = match trig {
62            Trig::Sin => f64::sin,
63            Trig::Cos => f64::cos,
64        };
65        if i >= INLINE_VARS {
66            return Ok(f(v));
67        }
68        let (valid, slots) = match trig {
69            Trig::Sin => (&mut self.sin_valid, &mut self.sin),
70            Trig::Cos => (&mut self.cos_valid, &mut self.cos),
71        };
72        if *valid & (1 << i) == 0 {
73            slots[i].write(f(v));
74            *valid |= 1 << i;
75        }
76        // SAFETY: the validity bit is set only after this slot is initialized.
77        Ok(unsafe { slots[i].assume_init() })
78    }
79}
80
81/// Which trigonometric function a fold step needs.
82#[derive(Clone, Copy)]
83enum Trig {
84    Sin,
85    Cos,
86}
87
88impl Prod {
89    /// The real part of [`Prod::eval_complex`] at `vals` (`vals[i]` binds
90    /// variable `i`): `0.0` for an imaginary monomial, `1.0` for the empty one.
91    #[cfg(test)]
92    #[inline]
93    pub(crate) fn eval(&self, vals: &[f64]) -> Result<f64, MissingVariable> {
94        Ok(self.eval_complex(vals)?.re)
95    }
96
97    /// The fold in contract order; every evaluation path goes through it.
98    #[inline]
99    fn fold(
100        &self,
101        mut trig_of: impl FnMut(u32, Trig) -> Result<f64, MissingVariable>,
102    ) -> Result<f64, MissingVariable> {
103        let mut res = 1.0;
104        for f in self.factors.iter().filter(|f| f.sin > 0) {
105            res *= trig_of(f.var, Trig::Sin)?.powi(f.sin as i32);
106        }
107        for f in self.factors.iter().filter(|f| f.cos > 0) {
108            res *= trig_of(f.var, Trig::Cos)?.powi(f.cos as i32);
109        }
110        Ok(res)
111    }
112
113    /// The value at `vals`, including the phase.
114    pub(crate) fn eval_complex(&self, vals: &[f64]) -> Result<Complex<f64>, MissingVariable> {
115        Ok(self.with_phase(self.fold_uncached(vals)?))
116    }
117
118    /// [`Prod::fold`] reading `vals` directly, for a one-off monomial.
119    #[inline]
120    fn fold_uncached(&self, vals: &[f64]) -> Result<f64, MissingVariable> {
121        self.fold(|k, trig| {
122            let v = vals.get(k as usize).ok_or(MissingVariable(k))?;
123            Ok(match trig {
124                Trig::Sin => v.sin(),
125                Trig::Cos => v.cos(),
126            })
127        })
128    }
129
130    /// `v`, or `i·v` for an imaginary monomial.
131    #[inline]
132    fn with_phase(&self, v: f64) -> Complex<f64> {
133        if self.imag {
134            Complex::new(0.0, v)
135        } else {
136            Complex::new(v, 0.0)
137        }
138    }
139}
140
141impl Sum {
142    /// Phase-aware evaluation; see [`Prod::eval_complex`].
143    pub(crate) fn eval_complex(&self, vals: &[f64]) -> Result<Complex<f64>, MissingVariable> {
144        let mut cache = AngleCache::new(vals);
145        let mut res = Complex::new(self.c0, 0.0);
146        for (p, c) in self.iter() {
147            res += p.with_phase(p.fold(|k, trig| cache.get(k, trig))?) * c;
148        }
149        Ok(res)
150    }
151}
152
153impl Term {
154    /// The real part of [`Term::eval_complex`].
155    #[inline]
156    pub fn eval(&self, vals: &[f64]) -> Result<f64, MissingVariable> {
157        Ok(self.eval_complex(vals)?.re)
158    }
159
160    /// The value at `vals` (`vals[i]` binds variable `i`), including the phase.
161    pub fn eval_complex(&self, vals: &[f64]) -> Result<Complex<f64>, MissingVariable> {
162        match self.inner {
163            Inner::Const(c) => Ok(Complex::new(c, 0.0)),
164            Inner::One(ref p, c) => Ok(p.eval_complex(vals)? * c),
165            Inner::Sum(ref s) => s.eval_complex(vals),
166        }
167    }
168}

170#[cfg(test)]
171mod tests {
172    use super::*;
173    use crate::term::Var;
174    use crate::term::{DEFAULT_MAX_SIN, DEFAULT_MIN_EPS};
175
176    #[test]
177    fn missing_variable_is_a_typed_error() {
178        let err = Var(5).sin().eval(&[0.1]).unwrap_err();
179        assert_eq!(err, MissingVariable(5));
180        assert_eq!(err.to_string(), "variable %5 not found");
181
182        // The cached `Sum` path reports the first missing variable the
183        // sin-then-cos walk reaches (`%7`), not the lowest missing id (`%5`).
184        let mut p = Prod::sin(7);
185        p.mul_cos(5);
186        let mut s = Sum::new();
187        s.add_term(p, 1.0, DEFAULT_MAX_SIN, DEFAULT_MIN_EPS);
188        assert_eq!(
189            s.eval_complex(&[0.1, 0.2, 0.3, 0.4]).map(|z| z.re),
190            Err(MissingVariable(7))
191        );
192    }
193
194    #[test]
195    fn fold_order_is_sin_ascending_then_cos_ascending() {
196        // Bit-identical to a hand-written fold in contract order.
197        let mut p = Prod::sin(0);
198        p.mul_sin(0);
199        p.mul_sin(0);
200        p.mul_sin(1);
201        p.mul_sin(1);
202        p.mul_cos(0);
203        let vals = [1.1, 2.1];
204        let expected = 1.0f64 * 1.1f64.sin().powi(3) * 2.1f64.sin().powi(2) * 1.1f64.cos().powi(1);
205        assert_eq!(p.eval(&vals).unwrap().to_bits(), expected.to_bits());
206    }
207
208    /// table order, across the inline window and with repeated variables.
209    #[test]
210    fn cached_sum_eval_is_bit_identical_to_uncached() {
211        let n = INLINE_VARS as u32 + 4;
212        let vals: Vec<f64> = (0..n).map(|i| 0.37 * i as f64 - 3.0).collect();
213        let mut s = Sum::new();
214        s.c0 = 0.125;
215        for i in 0..n {
216            let mut p = Prod::sin(i);
217            p.mul_cos((i * 7) % n);
218            p.mul_sin((i * 3) % n);
219            if i % 3 == 0 {
220                p.mul_phase(1);
221            }
222            s.add_term(p, 1.0 + i as f64, DEFAULT_MAX_SIN, DEFAULT_MIN_EPS);
223        }
224        let mut re = s.c0;
225        let mut z = Complex::new(s.c0, 0.0);
226        for (p, c) in s.iter() {
227            re += p.eval(&vals).unwrap() * c;
228            z += p.eval_complex(&vals).unwrap() * c;
229        }
230        assert_eq!(s.eval_complex(&vals).unwrap().re.to_bits(), re.to_bits());
231        let got = s.eval_complex(&vals).unwrap();
232        assert_eq!(
233            (got.re.to_bits(), got.im.to_bits()),
234            (z.re.to_bits(), z.im.to_bits())
235        );
236    }

238    #[test]
239    fn small_forms_and_the_real_part() {
240        assert_eq!(Prod::new().eval(&[]), Ok(1.0));
241        assert_eq!(Sum::new().eval_complex(&[]), Ok(Complex::new(0.0, 0.0)));
242        let mut p = Prod::cos(0);
243        p.mul_phase(1);
244        assert_eq!(p.eval(&[0.0]), Ok(0.0));
245        assert_eq!(p.eval_complex(&[0.0]), Ok(Complex::new(0.0, 1.0)));
246    }
