4//! [`TableauLike`](crate::tableau_like::TableauLike) trait: shared interface for stabilizer-tableau backends.
5//!
6//! Any type implementing [`TableauLike`](crate::tableau_like::TableauLike) gets default implementations of
7//! single- and two-qubit Pauli noise channels (Depolarizing, PauliError,
8//! Depolarizing2, TwoQubitPauliError). New tableau backends only need to
9//! provide [`TableauLike::rng_mut`](crate::tableau_like::TableauLike::rng_mut) and (optionally) override
10//! [`TableauLike::is_qubit_lost`](crate::tableau_like::TableauLike::is_qubit_lost)
11//! (which reports lost *or* leaked qubits; two-qubit channels marginalize them out).
12
13use num::Zero;
14use rand::{Rng, RngExt};
15
16use ppvm_traits::traits::{Clifford, Coefficient};
17
18// `Zero` is used for `Self::Coeff::zero()` inside default method bodies; the
19// bound itself is redundant on `Self::Coeff` because `Coefficient: num::Zero`.
20
21/// `true` iff `p` lies in the closed unit interval `[0, 1]`, up to a `1e-12`
22/// rounding slack above 1 (e.g. summed marginals of a normalized channel).
23///
24/// Spelled out with two explicit comparisons rather than
25/// `(0.0..=1.0).contains(p)` because the coefficient types here only implement
26/// `PartialOrd<f64>`, not the `PartialOrd<Self>` that `RangeInclusive::contains`
27/// requires. Keeping the two comparisons in separate statements also avoids the
28/// `clippy::manual_range_contains` rewrite, which would not compile here.
29#[inline]
30fn is_probability<C: PartialOrd<f64>>(p: &C) -> bool {
31    if *p < 0.0 {
32        return false;
33    }
34    *p <= 1.0 + 1e-12
35}
36
37/// Marginal X/Y/Z probabilities on one qubit of a two-qubit Pauli channel.
38///
39/// `p` uses the [`TableauLike::two_qubit_pauli_error_impl`] layout, where
40/// `p[4a + b - 1]` is the probability of `P_a ⊗ P_b` (0 = I, 1 = X, 2 = Y,
41/// 3 = Z). Sums out the other qubit's Pauli: `q_s = Σ_o p(s, o)` when keeping
42/// the first qubit, `q_s = Σ_o p(o, s)` when keeping the second.
43fn marginal_pauli_probs<C: Coefficient>(p: &[C; 15], keep_first: bool) -> [C; 3] {
44    core::array::from_fn(|i| {
45        let s = i + 1;
46        (0..4)
47            .map(|o| if keep_first { 4 * s + o } else { 4 * o + s })
48            .fold(C::zero(), |acc, k| acc + p[k - 1].clone())
49    })
50}

52/// A stabilizer-tableau-like backend that supports Clifford gates and an RNG.
53///
54/// Implementing this trait grants default implementations of the Pauli noise
55/// channels via [`TableauLike::depolarize_impl`] and friends. The associated
56/// `Rng` type lets each backend choose its own RNG; nothing in this trait
57/// depends on `SmallRng`.
58pub trait TableauLike: Clifford {
59    /// Coefficient type used for probabilities.
60    type Coeff: Coefficient + PartialOrd<f64>;
61
62    /// RNG type backing the stochastic channels.
63    type Rng: Rng + RngExt;
64
65    /// Mutable access to the backend's RNG.
66    fn rng_mut(&mut self) -> &mut Self::Rng;
67
68    /// Whether the qubit at `addr` is inactive (lost or leaked), so the
69    /// two-qubit Pauli channels act only on the surviving qubit. Default: never inactive.
70    #[inline]
71    fn is_qubit_lost(&self, _addr: usize) -> bool {
72        false
73    }
74
75    /// Single-qubit depolarizing channel.
76    ///
77    /// RNG is consumed unconditionally; the selected Clifford gate is expected
78    /// to no-op on lost qubits. This preserves seeded RNG sequences across
79    /// loss events.
80    #[inline]
81    fn depolarize_impl(&mut self, addr0: usize, p: Self::Coeff) {
82        debug_assert!(is_probability(&p));
83        let r = self.rng_mut().random::<f64>();
84        if p <= r {
85            return;
86        }
87        if p > r * 3.0 {
88            // p / 3 > r >= 0
89            self.x(addr0);
90        } else if p > r * 1.5 {
91            // 2p/3 > r >= p / 3
92            self.y(addr0);
93        } else {
94            // p > r >= 2p/3
95            self.z(addr0);
96        }
97    }
98
99    /// Single-qubit Pauli-error channel (X, Y, Z with given probabilities).
100    ///
101    /// RNG is consumed unconditionally; the selected Clifford gate is expected
102    /// to no-op on lost qubits. This preserves seeded RNG sequences across
103    /// loss events.
104    #[inline]
105    fn pauli_error_impl(&mut self, addr0: usize, p: [Self::Coeff; 3]) {
106        debug_assert!(p.iter().all(is_probability));
107        let r = self.rng_mut().random::<f64>();
108        let mut cumulative = Self::Coeff::zero();
109        for (i, p_) in p.iter().enumerate() {
110            cumulative += p_.clone();
111            if cumulative > r {
112                match i {
113                    0 => self.x(addr0),
114                    1 => self.y(addr0),
115                    _ => self.z(addr0),
116                }
117                return;
118            }
119        }
120    }
121
122    /// Two-qubit Pauli-error channel (15 non-identity Pauli combinations).
123    ///
124    /// If exactly one qubit is lost or leaked, the survivor gets the marginal
125    /// single-qubit channel (the dead qubit's Pauli is summed out). If both
126    /// are, nothing is applied.
127    #[inline]
128    fn two_qubit_pauli_error_impl(&mut self, addr0: usize, addr1: usize, p: [Self::Coeff; 15]) {
129        debug_assert!(p.iter().all(is_probability));
130
131        let is_lost0 = self.is_qubit_lost(addr0);
132        let is_lost1 = self.is_qubit_lost(addr1);
133        if is_lost0 && is_lost1 {
134            return;
135        }
136
137        if is_lost0 || is_lost1 {
138            let addr = if is_lost0 { addr1 } else { addr0 };
139            let marginal_probs = marginal_pauli_probs(&p, !is_lost0);
140            self.pauli_error_impl(addr, marginal_probs);
141            return;
142        }
143
144        let r = self.rng_mut().random::<f64>();
145        let sum = Self::Coeff::zero();
146        let idx = p
147            .iter()
148            .scan(sum, |acc, p_| {
149                *acc += p_.clone();
150                Some(acc.clone())
151            })
152            .position(|cum_prob| cum_prob > r);
153
154        if let Some(i) = idx {
155            // Two-qubit Pauli pairs indexed by `i + 1` (so II at index 0 is
156            // skipped). Encoding: 0 = I, 1 = X, 2 = Y, 3 = Z; the first entry
157            // acts on addr0, the second on addr1.
158            //
159            // `rustfmt::skip` keeps the rows grouped by the first Pauli (a
160            // readable 4-wide grid); without it rustfmt repacks the tuples to
161            // fill the line width and the grouping is lost.
162            #[rustfmt::skip]
163            const PAULI_PAIRS: [(u8, u8); 16] = [
164                (0,0),(0,1),(0,2),(0,3),
165                (1,0),(1,1),(1,2),(1,3),
166                (2,0),(2,1),(2,2),(2,3),
167                (3,0),(3,1),(3,2),(3,3),
168            ];
169            let cartesian_index = PAULI_PAIRS[i + 1]; // skip II entry
170
171            match cartesian_index.0 {
172                0 => {}
173                1 => self.x(addr0),
174                2 => self.y(addr0),
175                _ => self.z(addr0),
176            }
177
178            match cartesian_index.1 {
179                0 => {}
180                1 => self.x(addr1),
181                2 => self.y(addr1),
182                _ => self.z(addr1),
183            }
184        }
185    }
186
187    /// Two-qubit depolarizing channel: spreads `p` over the 15 non-identity
188    /// two-qubit Pauli errors. A lone surviving qubit is depolarized with `4p/5`.
189    #[inline]
190    fn depolarize2_impl(&mut self, addr0: usize, addr1: usize, p: Self::Coeff) {
191        debug_assert!(is_probability(&p));
192        let p_arr: [Self::Coeff; 15] = core::array::from_fn(|_| p.clone() * (1.0 / 15.0));
193        self.two_qubit_pauli_error_impl(addr0, addr1, p_arr);
194    }
195}
