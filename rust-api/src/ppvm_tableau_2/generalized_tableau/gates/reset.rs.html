4use ppvm_traits_2::{Clifford, Reset, ResetState};
5
6use super::super::GeneralizedTableau;
7use super::super::status::QubitStatus;
8use crate::VirtualState;
9
10/// Measure, then flip the qubit if it was `|1⟩`. Lost and leaked qubits are
11/// left alone.
12impl<V: VirtualState> ResetState for GeneralizedTableau<V> {
13    fn reset_state(&mut self, q: usize) {
14        if !self.is_inactive(q) && self.sample_z(q) {
15            self.x(q);
16        }
17    }
18}
19
20/// A full reset: a lost or leaked qubit becomes live, then its state is reset.
21impl<V: VirtualState> Reset for GeneralizedTableau<V> {
22    fn reset(&mut self, q: usize) {
23        self.status[q] = QubitStatus::Live;
24        self.reset_state(q);
25    }
26}
