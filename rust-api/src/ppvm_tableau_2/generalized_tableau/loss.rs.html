4//! Loss and leakage. A lost qubit is collapsed, reset to `|0⟩` and skipped by
5//! gates; measuring it gives `None`. A leaked qubit is collapsed and pinned to
6//! `|0⟩` or `|1⟩`; gates skip it, and measuring it gives the pinned bit. Loss
7//! overwrites leakage. Neither event is a measurement outcome.
8

21impl<V: VirtualState> GeneralizedTableau<V> {
22    /// Reset `q`, collapsed onto outcome `m`, to `|0⟩` and mark it lost. The `X`
23    /// goes straight to the tableau, as `q` may be leaked.
24    fn mark_lost(&mut self, q: usize, m: bool) {
25        if m {
26            self.tableau.x(q);
27        }
28        self.status[q] = QubitStatus::Lost;
29    }
30
31    /// Collapse `q` by sampling `Z_q`, then reset it and mark it lost.
32    fn lose(&mut self, q: usize) {
33        let m = self.sample_z(q);
34        self.mark_lost(q, m);
35    }
36}
37
38/// With probability `p`, lose `q`. A lost qubit stays lost.
39impl<V: VirtualState> LossChannel<f64> for GeneralizedTableau<V> {
40    fn loss_channel(&mut self, q: usize, p: f64) {
41        if !self.is_lost(q) && self.rng.random::<f64>() < p {
42            self.lose(q);
43        }
44    }
45}
46
47/// Bring a lost qubit back as live; it is already in `|0⟩`.
48impl<V: VirtualState> ResetLossChannel for GeneralizedTableau<V> {
49    fn reset_loss_channel(&mut self, q: usize) {
50        if self.is_lost(q) {
51            self.status[q] = QubitStatus::Live;
52        }
53    }
54}

56/// With probability `p0` pin `q` to `|0⟩` and mark it leaked, with `p1` to
57/// `|1⟩`. Lost and leaked qubits are left alone.
58impl<V: VirtualState> LeakageChannel<f64> for GeneralizedTableau<V> {
59    fn leakage_channel(&mut self, q: usize, p0: f64, p1: f64) {
60        debug_assert!(p0 >= 0.0 && p1 >= 0.0 && p0 + p1 <= 1.0 + 1e-9);
61        if self.is_inactive(q) {
62            return;
63        }
64        let r = self.rng.random::<f64>();
65        if r >= p0 + p1 {
66            return;
67        }
68        let pin = r >= p0;
69        if self.sample_z(q) != pin {
70            self.tableau.x(q);
71        }
72        self.status[q] = QubitStatus::Leaked;
73    }
74}
75
76/// Bring a leaked qubit back as live `|0⟩`. Live and lost qubits are untouched.
77impl<V: VirtualState> ResetLeakageChannel for GeneralizedTableau<V> {
78    fn reset_leakage_channel(&mut self, q: usize) {
79        if self.is_leaked(q) {
80            self.status[q] = QubitStatus::Live;
81            self.reset_state(q);
82        }
83    }
84}

86/// `p = [p_LL, p_LQ, p_LN]` as the trait documents: if one of the pair is
87/// already lost, the other is lost with `p[2]`; otherwise both with `p[0]` and
88/// each named one alone with `p[1]`. A leaked partner counts as present.
89impl<V: VirtualState> CorrelatedLossChannel<f64> for GeneralizedTableau<V> {
90    fn correlated_loss_channel(&mut self, a: usize, b: usize, p: [f64; 3]) {
91        debug_assert!(
92            p[0] >= 0.0 && p[1] >= 0.0 && p[0] + 2.0 * p[1] <= 1.0 + 1e-9,
93            "correlated loss needs p0, p1 >= 0 and p0 + 2·p1 <= 1; got {p:?}"
94        );
95        debug_assert!((0.0..=1.0).contains(&p[2]), "p2 in [0, 1]; got {p:?}");
96        if self.is_lost(a) {
97            self.loss_channel(b, p[2]);
98        } else if self.is_lost(b) {
99            self.loss_channel(a, p[2]);
100        } else {
101            match self.draw(&[p[0], p[1], p[1]]) {
102                Some(0) => {
103                    self.lose(a);
104                    self.lose(b);
105                }
106                Some(1) => self.lose(a),
107                Some(_) => self.lose(b),
108                None => {}
109            }
110        }
111    }
112}

114/// State-dependent loss, with its back-action (issue #249). With Born weights
115/// `P(m)` of `Z_q`, `q` is lost from `|m⟩` with probability `p_m·P(m)`;
116/// otherwise the survival Kraus operator
117/// `K₀ = √(1−p0)|0⟩⟨0| + √(1−p1)|1⟩⟨1| = a·I + b·Z_q` is applied and the state
118/// renormalized. On a qubit in superposition `K₀` is non-Clifford and
119/// activates an axis.
120impl<V: VirtualState> AsymmetricLossChannel<f64> for GeneralizedTableau<V> {
121    fn asymmetric_loss_channel(&mut self, q: usize, p0: f64, p1: f64) {
122        debug_assert!((0.0..=1.0).contains(&p0) && (0.0..=1.0).contains(&p1));
123        if self.is_lost(q) {
124            return;
125        }
126        let [w0, w1] = self.z_weights(q);
127        let pop = [w0 / (w0 + w1), w1 / (w0 + w1)];
128        if let Some(k) = self.draw(&[p0 * pop[0], p1 * pop[1]]) {
129            // Lost from |m⟩: the outcome comes from the branch, not from `P(m)`.
130            let m = k == 1;
131            self.collapse(q, m);
132            self.mark_lost(q, m);
133            return;
134        }
135        // K₀ ∝ I: the renormalized state is unchanged, and rotating would
136        // activate an axis for nothing.
137        if p0 == p1 {
138            return;
139        }
140        // Scaled so the norm is unchanged: ‖K₀ψ‖² = Σ (1 − p_m)·P(m).
141        let scale = ((1.0 - p0) * pop[0] + (1.0 - p1) * pop[1]).sqrt().recip();
142        let (s0, s1) = ((1.0 - p0).sqrt(), (1.0 - p1).sqrt());
143        let a = Complex64::from((s0 + s1) / 2.0 * scale);
144        let b = Complex64::from((s0 - s1) / 2.0 * scale);
145        self.rotate_z(&[q], a, b);
146    }
147}

160    /// Whether `count` of `shots` is within 5σ of `rate`.
161    fn within(count: usize, shots: usize, rate: f64) -> bool {
162        let sigma = (rate * (1.0 - rate) / shots as f64).sqrt();
163        ((count as f64 / shots as f64) - rate).abs() <= 5.0 * sigma
164    }

166    /// A lost qubit reads `None` and no gate reaches it, including a CX from a
167    /// live `|1⟩` control. Clearing the flag finds it still in `|0⟩`.
168    #[test]
169    fn lost_qubit_is_skipped_and_reads_none() {
170        let mut s = Sim::new_with_seed(2, 0);
171        s.loss_channel(0, 1.0);
172        assert!(s.is_lost(0));
173        s.x(1);
174        s.cnot(1, 0);
175        s.x(0);
176        s.h(0);
177        s.rx(0, 1.0);
178        s.t(0);
179        assert_eq!(s.measure(0), None);
180        assert_eq!(s.measure(1), Some(true));
181        s.reset_loss_channel(0);
182        assert!(!s.is_lost(0));
183        assert_eq!(s.measure(0), Some(false));
184    }

186    /// `exp(−iπ/2 X⊗X)` with the second qubit lost acts as `exp(−iπ/2 X)` on
187    /// the first and flips it; skipping it whole would leave `|0⟩`.
188    #[test]
189    fn rotation_with_a_lost_partner_acts_on_the_survivor() {
190        let mut s = Sim::new_with_seed(2, 0);
191        s.loss_channel(1, 1.0);
192        s.rotate_2(Pauli::X, Pauli::X, 0, 1, PI);
193        assert_eq!(s.measure(0), Some(true));
194        assert_eq!(s.measure(1), None);

197    /// Loss is an unrecorded measurement: losing half of a Bell pair leaves the

241    /// `leakage_channel(p0, p1)` pins to `|0⟩` with `p0` and `|1⟩` with `p1`,

243    /// pinned bit.

245    fn leakage_pins_with_the_branch_probabilities() {
246        let (p0, p1, shots) = (0.2, 0.3, 20_000);
247        let (mut zeros, mut ones) = (0, 0);
248        for seed in 0..shots {
249            let mut s = Sim::new_with_seed(1, seed as u64);
250            s.h(0);
251            s.leakage_channel(0, p0, p1);
252            if s.is_leaked(0) {
253                let m = s.measure(0).unwrap();
254                s.x(0);
255                s.h(0);
256                s.rx(0, 1.0);

259                    ones += 1;

261                    zeros += 1;

263            }
264        }
265        assert!(within(zeros, shots, p0), "{zeros} of {shots}");
266        assert!(within(ones, shots, p1), "{ones} of {shots}");

308    /// `p = [p_LL, p_LQ, p_LN]`: from two present qubits both are lost with
309    /// `p_LL`, each named one alone with `p_LQ`; with one already lost, the
310    /// other is lost with `p_LN`.
311    #[test]
312    fn correlated_loss_event_rates() {
313        let (p, shots) = ([0.1, 0.2, 0.5], 20_000);
314        let mut counts = [0; 4];
315        let mut second = 0;
316        for seed in 0..shots {
317            let mut s = Sim::new_with_seed(2, seed as u64);
318            s.h(0);
319            s.cnot(0, 1);
320            s.correlated_loss_channel(0, 1, p);
321            counts[usize::from(s.is_lost(0)) | usize::from(s.is_lost(1)) << 1] += 1;
322            let mut t = Sim::new_with_seed(2, seed as u64);
323            t.loss_channel(0, 1.0);
324            t.correlated_loss_channel(0, 1, p);
325            second += usize::from(t.is_lost(1));
326        }
327        let rates = [1.0 - p[0] - 2.0 * p[1], p[1], p[1], p[0]];
328        for (k, (&count, rate)) in counts.iter().zip(rates).enumerate() {
329            assert!(
330                within(count, shots, rate),
331                "event {k:#b}: {count} of {shots}"
332            );
333        }
334        assert!(within(second, shots, p[2]), "{second} of {shots}");
335    }

337    /// The Bell-pair example of #249 with `p0 = 0, p1 = 1`: the qubit is lost
338    /// exactly from `|1⟩`, so a loss leaves the partner in `|1⟩`; survival
339    /// projects onto `|0⟩`, leaving `|00⟩`. Each happens half of the time.
340    #[test]
341    fn asymmetric_loss_on_a_bell_pair() {
342        let shots = 4000;
343        let mut lost = 0;
344        for seed in 0..shots {
345            let mut s = Sim::new_with_seed(2, seed as u64);
346            s.h(0);
347            s.cnot(0, 1);
348            s.asymmetric_loss_channel(0, 0.0, 1.0);
349            if s.is_lost(0) {
350                lost += 1;
351                assert_eq!(s.measure(1), Some(true), "lost from |1⟩");
352            } else {
353                assert_eq!((s.measure(0), s.measure(1)), (Some(false), Some(false)));
354            }
355        }
356        assert!(within(lost, shots, 0.5), "{lost} of {shots}");
357    }
