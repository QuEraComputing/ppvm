4//! The storage word and the only code that knows its width. Everything else
5//! touches words through these items and `Bits`, so a wider word can be tried
6//! by changing the alias.
7
8/// Storage word. Swap to benchmark other widths.
9pub(super) type BitWord = u64;
10pub(super) const BITWORD_BITS: usize = BitWord::BITS as usize;
11pub(super) const ZERO: BitWord = 0;
12/// The highest bit of a word.
13pub(super) const TOP_BIT: BitWord = 1 << (BITWORD_BITS - 1);
14
15/// Bit `i` of a packed bit string.
16#[inline(always)]
17pub(super) fn get_bit(words: &[BitWord], i: usize) -> bool {
18    (words[i / BITWORD_BITS] >> (i % BITWORD_BITS)) & 1 == 1
19}
20
21#[inline(always)]
22pub(super) fn flip_bit(words: &mut [BitWord], i: usize) {
23    words[i / BITWORD_BITS] ^= 1 << (i % BITWORD_BITS);
24}
25
26/// Word index and mask of bit `i`.
27#[inline(always)]
28pub(super) fn bit_position(i: usize) -> (usize, BitWord) {
29    (i / BITWORD_BITS, 1 << (i % BITWORD_BITS))
30}
31
32/// All ones if `b`, else zero (for branch-free masked updates).
33#[inline(always)]
34pub(super) fn mask_if(b: bool) -> BitWord {
35    ZERO.wrapping_sub(BitWord::from(b))
36}
37
38/// Indices of the set bits of a packed bit string, ascending.
39pub(super) fn set_bits(words: &[BitWord]) -> impl Iterator<Item = usize> + '_ {
40    set_bits_of(words.iter().copied())
41}
42
43/// [`set_bits`] over words computed on the fly (e.g. masked).
44pub(super) fn set_bits_of(words: impl Iterator<Item = BitWord>) -> impl Iterator<Item = usize> {
45    words.enumerate().flat_map(|(w, word)| {
46        let mut rest = word;
47        std::iter::from_fn(move || {
48            (rest != ZERO).then(|| {
49                let b = rest.trailing_zeros() as usize;
50                rest &= rest - 1;
51                w * BITWORD_BITS + b
52            })
53        })
54    })
55}
56
57/// Column blocks transposed together by [`transpose_64x64`].
58pub(super) const LANES: usize = 4;
59
60/// Transpose [`LANES`] independent 64 × 64 bit matrices in place: lane `l` of
61/// word `i` is word `i` of matrix `l`, and bit `j` of word `i` swaps with bit
62/// `i` of word `j` in each. Running every step on all lanes lets the compiler
63/// vectorize across matrices. Width-specific (64-bit words).
64#[inline(always)]
65pub(super) fn transpose_64x64(a: &mut [[BitWord; LANES]; 64]) {
66    const PASSES: [(BitWord, usize); 6] = [
67        (0x5555_5555_5555_5555, 1),
68        (0x3333_3333_3333_3333, 2),
69        (0x0F0F_0F0F_0F0F_0F0F, 4),
70        (0x00FF_00FF_00FF_00FF, 8),
71        (0x0000_FFFF_0000_FFFF, 16),
72        (0x0000_0000_FFFF_FFFF, 32),
73    ];
74    for (mask, shift) in PASSES {
75        for base in (0..64).step_by(2 * shift) {
76            for k in base..base + shift {
77                let (x, y) = (a[k], a[k + shift]);
78                for l in 0..LANES {
79                    a[k][l] = (x[l] & mask) | ((y[l] & mask) << shift);
80                    a[k + shift][l] = ((x[l] & !mask) >> shift) | (y[l] & !mask);
81                }
82            }
83        }
84    }
85}
