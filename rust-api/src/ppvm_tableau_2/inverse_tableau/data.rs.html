4//! The tableau's storage: rows, signs, and where rows sit in the arena.
5
6use super::bitword::{
7    BITWORD_BITS, BitWord, TOP_BIT, ZERO, bit_position, flip_bit, get_bit, mask_if, set_bits,
8    set_bits_of,
9};
10use super::view::QubitMask;
11
12/// The inverse tableau of an `n`-qubit Clifford.
13#[derive(Clone, Debug, PartialEq, Eq)]
14pub struct InverseTableau {
15    pub(super) n: usize,
16    /// Words per row and plane: `n.div_ceil(BITWORD_BITS)`.
17    pub(super) stride: usize,
18    /// X and Z words of all `2n` rows, arranged by `offsets`.
19    pub(super) words: Vec<BitWord>,
20    /// Bit `r` is the sign of row `r` (set = −).
21    pub(super) signs: Vec<BitWord>,
22}
23
24// NOTE: we could impl Word for RowRef to make it a true PauliWord if we ever need it
25/// A borrowed row: its X words, Z words and sign.
26#[derive(Clone, Copy, Debug)]
27pub struct RowRef<'a> {
28    pub x: &'a [BitWord],
29    pub z: &'a [BitWord],
30    pub negative: bool,
31}
32
33impl<'a> RowRef<'a> {
34    /// X bit of the row at qubit `q`.
35    pub fn x_bit(&self, q: usize) -> bool {
36        get_bit(self.x, q)
37    }
38
39    /// Z bit of the row at qubit `q`.
40    pub fn z_bit(&self, q: usize) -> bool {
41        get_bit(self.z, q)
42    }
43
44    /// Qubits where the row has an X or Y, ascending.
45    pub fn x_support(&self) -> impl Iterator<Item = usize> + 'a {
46        set_bits(self.x)
47    }
48
49    /// Qubits outside `mask` where the row has an X or Y, ascending.
50    pub fn x_outside(&self, mask: &'a QubitMask) -> impl Iterator<Item = usize> + 'a {
51        set_bits_of(self.x.iter().zip(&mask.words).map(|(&x, &m)| x & !m))
52    }
53
54    /// Qubits in `mask` where the row has a Z or Y, ascending.
55    pub fn z_inside(&self, mask: &'a QubitMask) -> impl Iterator<Item = usize> + 'a {
56        set_bits_of(self.z.iter().zip(&mask.words).map(|(&z, &m)| z & m))
57    }
58}
59
60impl InverseTableau {
61    /// The identity Clifford, i.e. the state `|0…0⟩`: `ix_q = X_q`, `iz_q = Z_q`.
62    pub fn new(n: usize) -> Self {
63        let stride = n.div_ceil(BITWORD_BITS);
64        let mut t = Self {
65            n,
66            stride,
67            words: vec![ZERO; 4 * n * stride],
68            signs: vec![ZERO; (2 * n).div_ceil(BITWORD_BITS)],
69        };
70        for q in 0..n {
71            let (ix, iz) = (t.x_at(q), t.z_at(n + q));
72            flip_bit(&mut t.words[ix..], q);
73            flip_bit(&mut t.words[iz..], q);
74        }
75        t
76    }
77
78    pub fn num_qubits(&self) -> usize {
79        self.n
80    }
81
82    /// Row `ix_q = C⁻¹ X_q C`.
83    pub fn ix(&self, q: usize) -> RowRef<'_> {
84        assert!(q < self.n, "qubit {q} out of range");
85        self.row(q)
86    }
87
88    /// Row `iz_q = C⁻¹ Z_q C`.
89    pub fn iz(&self, q: usize) -> RowRef<'_> {
90        assert!(q < self.n, "qubit {q} out of range");
91        self.row(self.n + q)
92    }
93
94    pub(super) fn row(&self, r: usize) -> RowRef<'_> {
95        let (x, z) = (self.x_at(r), self.z_at(r));
96        RowRef {
97            x: &self.words[x..x + self.stride],
98            z: &self.words[z..z + self.stride],
99            negative: self.negative(r),
100        }
101    }
102
103    /// Whether row `r` has sign `−`.
104    #[inline(always)]
105    pub(super) fn negative(&self, r: usize) -> bool {
106        get_bit(&self.signs, r)
107    }
108
109    pub(super) fn flip_sign(&mut self, r: usize) {
110        flip_bit(&mut self.signs, r);
111    }
112
113    // Arrangement: blocks by row kind, ix-X, ix-Z, iz-X, iz-Z (#204's order).
114    // Row r is row i = r % n of kind r / n (0 for ix, 1 for iz).
115    #[inline(always)]
116    pub(super) fn offsets(n: usize, stride: usize, kind: usize, i: usize) -> (usize, usize) {
117        (
118            ((2 * kind) * n + i) * stride,
119            ((2 * kind + 1) * n + i) * stride,
120        )
121    }
122
123    /// The rows of `qubits` in generator order: `ix_q`, `iz_q` per qubit. `M`
124    /// is twice the number of qubits.
125    #[inline(always)]
126    pub(super) fn qubit_rows<const M: usize>(&self, qubits: &[usize]) -> [usize; M] {
127        let mut rows = [0; M];
128        for (k, &q) in qubits.iter().enumerate() {
129            rows[2 * k] = q;
130            rows[2 * k + 1] = self.n + q;
131        }
132        rows
133    }
134
135    /// The X words and the Z words of the rows of distinct qubits, in the
136    /// order of [`qubit_rows`](Self::qubit_rows), mutably.
137    ///
138    /// Panics unless every qubit is in range and no qubit repeats: distinct
139    /// qubits own distinct rows, which makes the ranges disjoint, so this is
140    /// the one place that checks it.
141    #[inline(always)]
142    pub(super) fn qubit_rows_mut<const M: usize>(
143        &mut self,
144        qubits: &[usize],
145    ) -> ([&mut [BitWord]; M], [&mut [BitWord]; M]) {
146        assert_eq!(M, 2 * qubits.len());
147        for (i, &q) in qubits.iter().enumerate() {
148            assert!(q < self.n, "qubit {q} out of range");
149            assert!(!qubits[..i].contains(&q), "qubit {q} repeated");
150        }
151        let rows: [usize; M] = self.qubit_rows(qubits);
152        let (xs, zs) = (rows.map(|r| self.x_at(r)), rows.map(|r| self.z_at(r)));
153        let (stride, words) = (self.stride, self.words.as_mut_ptr());
154        // SAFETY: distinct qubits in range own distinct rows, whose
155        // `stride`-word ranges of `words` are disjoint; the slices borrow
156        // `self` mutably for their lifetime.
157        unsafe {
158            (
159                xs.map(|x| std::slice::from_raw_parts_mut(words.add(x), stride)),
160                zs.map(|z| std::slice::from_raw_parts_mut(words.add(z), stride)),
161            )
162        }
163    }
164
165    /// Kind and index of row `r < 2n`, without dividing.
166    #[inline(always)]
167    fn kind_and_index(&self, r: usize) -> (usize, usize) {
168        let kind = usize::from(r >= self.n);
169        (kind, r - kind * self.n)
170    }
171
172    #[inline(always)]
173    pub(super) fn x_at(&self, r: usize) -> usize {
174        let (kind, i) = self.kind_and_index(r);
175        Self::offsets(self.n, self.stride, kind, i).0
176    }
177
178    #[inline(always)]
179    pub(super) fn z_at(&self, r: usize) -> usize {
180        let (kind, i) = self.kind_and_index(r);
181        Self::offsets(self.n, self.stride, kind, i).1
182    }
183
184    /// Visit every row in order with its X words and Z words; `f` returns
185    /// whether the row's sign flips. Signs are updated a word (64 rows) at a
186    /// time. Walks the blocks of `offsets` directly: rows of one kind are
187    /// `stride` words apart.
188    #[inline(always)]
189    pub(super) fn for_each_row(
190        &mut self,
191        mut f: impl FnMut(&mut [BitWord], &mut [BitWord]) -> bool,
192    ) {
193        let (n, stride) = (self.n, self.stride);
194        if n == 0 {
195            return;
196        }
197        debug_assert_eq!(
198            Self::offsets(n, stride, 1, 0),
199            (2 * n * stride, 3 * n * stride)
200        );
201        let (ix, iz) = self.words.split_at_mut(2 * n * stride);
202        let (ix_x, ix_z) = ix.split_at_mut(n * stride);
203        let (iz_x, iz_z) = iz.split_at_mut(n * stride);
204        let (mut flips, mut r) = (ZERO, 0);
205        for (xs, zs) in [(ix_x, ix_z), (iz_x, iz_z)] {
206            for (x, z) in xs.chunks_exact_mut(stride).zip(zs.chunks_exact_mut(stride)) {
207                let (sw, sm) = bit_position(r);
208                flips |= sm & mask_if(f(x, z));
209                if sm == TOP_BIT {
210                    self.signs[sw] ^= flips;
211                    flips = ZERO;
212                }
213                r += 1;
214            }
215        }
216        if !r.is_multiple_of(BITWORD_BITS) {
217            self.signs[r / BITWORD_BITS] ^= flips;
218        }
219    }
220}

222#[cfg(test)]
223mod tests {
224    use super::super::test_support::*;
225    use super::*;
226
227    #[test]
228    fn identity_rows_are_single_paulis() {
229        for n in SIZES {
230            let t = InverseTableau::new(n);
231            for q in 0..n {
232                for j in 0..n {
233                    assert_eq!(t.ix(q).x_bit(j), j == q);
234                    assert!(!t.ix(q).z_bit(j));
235                    assert!(!t.iz(q).x_bit(j));
236                    assert_eq!(t.iz(q).z_bit(j), j == q);
237                }
238                assert!(!t.ix(q).negative && !t.iz(q).negative);
239            }
240        }
241    }
242
243    #[test]
244    fn identity_has_canonical_commutation() {
245        for n in SIZES {
246            assert_canonical_commutation(&InverseTableau::new(n));
247        }
248    }
