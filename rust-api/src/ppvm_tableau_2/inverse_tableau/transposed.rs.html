4//! A column-major copy of the tableau, for batches of changes of basis.
5//!
6//! Column `c` (virtual qubit `c`) of a row block is `stride` contiguous words,
7//! one bit per row, so a change of basis runs the gate rules on 64 rows per
8//! word (`B = u64`) instead of one bit per row. Building the copy and writing
9//! it back each cost a transpose, so it only pays for a batch.
10
11use ppvm_pauli_kernels::heisenberg;
12
13use super::InverseTableau;
14use super::bitword::{BitWord, LANES, ZERO, bit_position, flip_bit, get_bit, transpose_64x64};
15
16pub(super) struct TransposedTableau {
17    n: usize,
18    stride: usize,
19    /// `x[k][c * stride + w]`: X bits of column `c` for rows `64w..` of
20    /// block `k` (0: the `ix` rows, 1: the `iz` rows).
21    x: [Vec<BitWord>; 2],
22    z: [Vec<BitWord>; 2],
23    /// `signs[k]`: bit `i` is the sign of row `i` of block `k`.
24    signs: [Vec<BitWord>; 2],
25}
26
27impl TransposedTableau {
28    /// The column-major copy of `t`.
29    pub(super) fn from_rows(t: &InverseTableau) -> Self {
30        let (n, stride) = (t.n, t.stride);
31        let empty = || [vec![ZERO; n * stride], vec![ZERO; n * stride]];
32        let mut c = Self {
33            n,
34            stride,
35            x: empty(),
36            z: empty(),
37            signs: [vec![ZERO; stride], vec![ZERO; stride]],
38        };
39        for k in 0..2 {
40            for plane in 0..2 {
41                let dst = if plane == 0 { &mut c.x[k] } else { &mut c.z[k] };
42                for_each_block(stride, |rows, cols, lanes| {
43                    let mut block = [[ZERO; LANES]; 64];
44                    for (r, words) in block.iter_mut().enumerate() {
45                        let i = rows + r;
46                        if i < n {
47                            let (x, z) = InverseTableau::offsets(n, stride, k, i);
48                            let at = if plane == 0 { x } else { z } + cols / 64;
49                            // A loop: `copy_from_slice` (or a zip) of < 4 words calls `memmove`.
50                            #[allow(clippy::manual_memcpy)]
51                            for l in 0..lanes {
52                                words[l] = t.words[at + l];
53                            }
54                        }
55                    }
56                    transpose_64x64(&mut block);
57                    for (j, words) in block.iter().enumerate() {
58                        for (l, &word) in words[..lanes].iter().enumerate() {
59                            let c = cols + 64 * l + j;
60                            if c < n {
61                                dst[c * stride + rows / 64] = word;
62                            }
63                        }
64                    }
65                });
66            }
67            for i in 0..n {
68                if t.negative(k * n + i) {
69                    flip_bit(&mut c.signs[k], i);
70                }
71            }
72        }
73        c
74    }
75
76    /// Write this copy back into the rows of `t`.
77    pub(super) fn write_rows(&self, t: &mut InverseTableau) {
78        let (n, stride) = (self.n, self.stride);
79        for k in 0..2 {
80            for plane in 0..2 {
81                let src = if plane == 0 { &self.x[k] } else { &self.z[k] };
82                for_each_block(stride, |rows, cols, lanes| {
83                    let mut block = [[ZERO; LANES]; 64];
84                    for (j, words) in block.iter_mut().enumerate() {
85                        for (l, word) in words[..lanes].iter_mut().enumerate() {
86                            let c = cols + 64 * l + j;
87                            if c < n {
88                                *word = src[c * stride + rows / 64];
89                            }
90                        }
91                    }
92                    transpose_64x64(&mut block);
93                    for (r, words) in block.iter().enumerate() {
94                        let i = rows + r;
95                        if i < n {
96                            let (x, z) = InverseTableau::offsets(n, stride, k, i);
97                            let at = if plane == 0 { x } else { z } + cols / 64;
98                            #[allow(clippy::manual_memcpy)]
99                            for l in 0..lanes {
100                                t.words[at + l] = words[l];
101                            }
102                        }
103                    }
104                });
105            }
106            for i in 0..n {
107                if t.negative(k * n + i) != get_bit(&self.signs[k], i) {
108                    t.flip_sign(k * n + i);
109                }
110            }
111        }
112    }
113
114    /// X and Z bits of row `i` of block `k` at column `c`.
115    pub(super) fn bits(&self, k: usize, i: usize, c: usize) -> [bool; 2] {
116        let w = c * self.stride;
117        [
118            get_bit(&self.x[k][w..w + self.stride], i),
119            get_bit(&self.z[k][w..w + self.stride], i),
120        ]
121    }
122
123    /// Columns where row `i` of block `k` has an X bit, ascending, into `out`:
124    /// word `i / 64` of each column, without touching the Z bits.
125    pub(super) fn x_support(&self, k: usize, i: usize, out: &mut Vec<usize>) {
126        let (w, mask) = bit_position(i);
127        let words = self.x[k][w..].iter().step_by(self.stride);
128        out.extend(
129            words
130                .enumerate()
131                .filter(|&(_, &word)| word & mask != ZERO)
132                .map(|(c, _)| c),
133        );
134    }
135
136    pub(super) fn negative(&self, k: usize, i: usize) -> bool {
137        get_bit(&self.signs[k], i)
138    }
139
140    /// [`InverseTableau::fan_out`] without CZ targets, on columns: the rules
141    /// run on 64 rows per word.
142    pub(super) fn fan_out(
143        &mut self,
144        pivot: usize,
145        cx_targets: &[usize],
146        last: impl Fn(BitWord, BitWord) -> ([BitWord; 2], BitWord),
147    ) {
148        let s = self.stride;
149        for k in 0..2 {
150            let (x, z) = (&mut self.x[k], &mut self.z[k]);
151            for w in 0..s {
152                let p = pivot * s + w;
153                let (mut px, mut pz, mut flips) = (x[p], z[p], ZERO);
154                for &t in cx_targets {
155                    let i = t * s + w;
156                    let ([nx, nz, tx, tz], f) = heisenberg::cnot([px, pz, x[i], z[i]]);
157                    (x[i], z[i], px, pz) = (tx, tz, nx, nz);
158                    flips ^= f;
159                }
160                let ([nx, nz], f) = last(px, pz);
161                (x[p], z[p]) = (nx, nz);
162                self.signs[k][w] ^= flips ^ f;
163            }
164        }
165    }
166}
167
168/// Visit the 64 × 64 blocks of a `stride`-word square bit matrix in groups of up to
169/// [`LANES`] side by side: first row, first column, and blocks in the group.
170fn for_each_block(stride: usize, mut f: impl FnMut(usize, usize, usize)) {
171    for rows in (0..stride).map(|w| 64 * w) {
172        for w in (0..stride).step_by(LANES) {
173            f(rows, 64 * w, LANES.min(stride - w));
174        }
175    }
176}

178#[cfg(test)]
179mod tests {
180    use super::super::test_support::*;
181    use super::*;
182
183    /// Building the copy and writing it back restores the tableau exactly,
184    /// and the copy's bits and signs are the rows' bits and signs.
185    #[test]
186    fn round_trip_and_bits_match() {
187        // 300 qubits: one full group of `LANES` column blocks and a partial one.
188        for (i, n) in SIZES.into_iter().chain([300]).enumerate() {
189            let mut t = InverseTableau::new(n);
190            random_circuit(n, 150, 0x7a + i as u64)
191                .into_iter()
192                .for_each(|g| apply(&mut t, g));
193            let c = TransposedTableau::from_rows(&t);
194            for q in 0..n {
195                for col in 0..n {
196                    assert_eq!(c.bits(0, q, col), [t.ix(q).x_bit(col), t.ix(q).z_bit(col)]);
197                    assert_eq!(c.bits(1, q, col), [t.iz(q).x_bit(col), t.iz(q).z_bit(col)]);
198                }
199                assert_eq!(c.negative(1, q), t.iz(q).negative);
200            }
201            let mut back = InverseTableau::new(n);
202            random_circuit(n, 20, 0x99)
203                .into_iter()
204                .for_each(|g| apply(&mut back, g));
205            c.write_rows(&mut back);
206            assert_eq!(back, t, "n = {n}");
207        }
208    }
