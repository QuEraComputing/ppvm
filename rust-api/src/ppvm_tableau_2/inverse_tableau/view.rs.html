4//! [`VirtualView`]: the part of the tableau a virtual state may use.

11/// A set of qubits as packed bits, e.g. the active virtual qubits, so rows
12/// can be classified against it word by word.
13#[derive(Clone, Debug, PartialEq, Eq)]
14pub struct QubitMask {
15    pub(super) words: Vec<BitWord>,
16}
17
18impl QubitMask {
19    /// The empty set over `n` qubits.
20    pub fn new(n: usize) -> Self {
21        Self {
22            words: vec![ZERO; n.div_ceil(BITWORD_BITS)],
23        }
24    }
25
26    pub fn contains(&self, q: usize) -> bool {
27        get_bit(&self.words, q)
28    }
29
30    pub fn set(&mut self, q: usize, member: bool) {
31        let (w, m) = bit_position(q);
32        self.words[w] = self.words[w] & !m | m & mask_if(member);
33    }
34}

36/// An owned row, e.g. a product of rows: its X words, Z words and sign.
37#[derive(Clone, Debug, PartialEq, Eq)]
38pub struct OwnedRow {
39    x: Vec<BitWord>,
40    z: Vec<BitWord>,
41    negative: bool,
42}
43
44impl OwnedRow {
45    pub fn row(&self) -> RowRef<'_> {
46        RowRef {
47            x: &self.x,
48            z: &self.z,
49            negative: self.negative,
50        }
51    }
52}

54/// The virtual side of an [`InverseTableau`]: row reads and changes of basis
55/// `C ← C·V`, but no physical gates. A virtual state gets this view, so it can
56/// change basis together with its own state while `|ψ⟩` stays the same.
57pub struct VirtualView<'a> {
58    t: &'a mut InverseTableau,
59}
60
61impl InverseTableau {
62    pub fn virtual_view(&mut self) -> VirtualView<'_> {
63        VirtualView { t: self }
64    }
65}
66
67impl VirtualView<'_> {
68    pub fn num_qubits(&self) -> usize {
69        self.t.num_qubits()
70    }
71
72    /// Row `ix_q = C⁻¹ X_q C`.
73    pub fn ix(&self, q: usize) -> RowRef<'_> {
74        self.t.ix(q)
75    }
76
77    /// Row `iz_q = C⁻¹ Z_q C`.
78    pub fn iz(&self, q: usize) -> RowRef<'_> {
79        self.t.iz(q)
80    }
81
82    /// `O = C⁻¹ (∏ Z_q) C = ∏ iz_q` over `qubits`, which must be distinct. The
83    /// `Z_q` commute, so `O` is a Hermitian Pauli string with sign ±1.
84    pub fn iz_product(&self, qubits: &[usize]) -> OwnedRow {
85        let stride = self.t.stride;
86        let (mut x, mut z) = (vec![ZERO; stride], vec![ZERO; stride]);
87        // Power of `i` in front of the product of the bare rows.
88        let mut phase = 0u8;
89        for &q in qubits {
90            let r = self.t.iz(q);
91            let k = row_product(&mut x, &mut z, r.x, r.z) + 2 * u8::from(r.negative);
92            phase = phase.wrapping_add(k);
93        }
94        debug_assert_eq!(phase % 2, 0, "product of commuting rows is Hermitian");
95        OwnedRow {
96            x,
97            z,
98            negative: phase % 4 == 2,
99        }
100    }
101
102    /// See [`InverseTableau::change_basis1`].
103    pub fn change_basis1(&mut self, v: usize, rule: impl Fn(bool, bool) -> ([bool; 2], bool)) {
104        self.t.change_basis1(v, rule);
105    }
106
107    /// See [`InverseTableau::change_basis2`].
108    pub fn change_basis2(
109        &mut self,
110        a: usize,
111        b: usize,
112        rule: impl Fn([bool; 4]) -> ([bool; 4], bool),
113    ) {
114        self.t.change_basis2(a, b, rule);
115    }
116
117    /// See [`InverseTableau::fan_out`].
118    pub fn fan_out(
119        &mut self,
120        pivot: usize,
121        cx_targets: &[usize],
122        cz_targets: &[usize],
123        last: impl Fn(bool, bool) -> ([bool; 2], bool),
124    ) {
125        self.t.fan_out(pivot, cx_targets, cz_targets, last);
126    }
127
128    /// Measure `Z_q` with every virtual qubit dormant, i.e. on `C |0…0⟩`. See
129    /// [`InverseTableau::measure`]; the collapse is a change of basis only.
130    pub fn measure(&mut self, q: usize, coin: impl FnOnce() -> bool) -> bool {
131        self.t.measure(q, coin)
132    }
133
134    /// See [`InverseTableau::collapse_dormant`].
135    pub fn collapse_dormant(&mut self, q: usize, active: &QubitMask, m: bool) {
136        self.t.collapse_dormant(q, active, m);
137    }
138}
