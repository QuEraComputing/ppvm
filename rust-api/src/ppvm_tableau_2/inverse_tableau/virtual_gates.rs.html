4//! Virtual gates, i.e. changes of basis `C ← C·V`: they rewrite columns.
5
6use ppvm_pauli_kernels::heisenberg;
7
8use super::InverseTableau;
9use super::bitword::{BitWord, ZERO, bit_position, mask_if};
10
11impl InverseTableau {
12    /// Change the basis of virtual qubit `v` by the one-qubit Clifford `V`
13    /// whose Heisenberg rule is `rule`: `C ← C·V`.
14    ///
15    /// Updates the tableau only. The state `C (|φ⟩ ⊗ |0…0⟩)` stays the same if
16    /// the caller also applies `V†` to `φ`, or if `V` acts trivially on it (e.g.
17    /// `v` dormant and `V` fixes `|0⟩`).
18    ///
19    /// Every row `R` becomes `V† R V`: the rule runs on each row's bits in
20    /// column `v`. Strided: one word per row.
21    ///
22    /// Used by: dormant-pivot collapse (`H` on the pivot, `X` to fix the
23    /// outcome), active collapse (the final rotation onto `Z_h`), reset.
24    pub fn change_basis1(&mut self, v: usize, rule: impl Fn(bool, bool) -> ([bool; 2], bool)) {
25        debug_assert!(v < self.n);
26        let (w, m) = bit_position(v);
27        self.for_each_row(|x, z| {
28            let old = [x[w] & m != ZERO, z[w] & m != ZERO];
29            let (new, flip) = rule(old[0], old[1]);
30            x[w] ^= m & mask_if(new[0] != old[0]);
31            z[w] ^= m & mask_if(new[1] != old[1]);
32            flip
33        });
34    }
35
36    /// Change the basis of virtual qubits `a`, `b` by the two-qubit Clifford
37    /// `V` whose Heisenberg rule is `rule` (`a` is the control for controlled
38    /// gates): `C ← C·V`. Same contract as [`change_basis1`](Self::change_basis1).
39    ///
40    /// Used by: T-gate localization (`CX(p, v)` collects the X bits of `O` onto
41    /// the pivot `p`), dormant-pivot collapse (`CX(p, v)`, `CZ(p, c)`), active
42    /// collapse (the `CX` / `CZ` that reduce `P` to one qubit).
43    pub fn change_basis2(
44        &mut self,
45        a: usize,
46        b: usize,
47        rule: impl Fn([bool; 4]) -> ([bool; 4], bool),
48    ) {
49        debug_assert!(a < self.n && b < self.n && a != b);
50        let [(wa, ma), (wb, mb)] = [bit_position(a), bit_position(b)];
51        self.for_each_row(|x, z| {
52            let old = [
53                x[wa] & ma != ZERO,
54                z[wa] & ma != ZERO,
55                x[wb] & mb != ZERO,
56                z[wb] & mb != ZERO,
57            ];
58            let (new, flip) = rule(old);
59            x[wa] ^= ma & mask_if(new[0] != old[0]);
60            z[wa] ^= ma & mask_if(new[1] != old[1]);
61            x[wb] ^= mb & mask_if(new[2] != old[2]);
62            z[wb] ^= mb & mask_if(new[3] != old[3]);
63            flip
64        });
65    }
66
67    /// Change basis by a pivot fan-out in one pass over the rows:
68    /// `C ← C·V` with `V = CX(p, t₁)⋯CX(p, t_k) · CZ(p, c₁)⋯CZ(p, c_m) · L(p)`,
69    /// where `L` is the one-qubit Clifford whose Heisenberg rule is `last`.
70    ///
71    /// Same result and contract as the corresponding [`change_basis2`] /
72    /// [`change_basis1`] calls in that order, but each row loads and stores the
73    /// pivot's bits and its sign once instead of once per gate. Targets must be
74    /// distinct and differ from the pivot.
75    ///
76    /// Used by: T-gate localization (CX only, `last` the identity), both
77    /// random measurement collapses (CX, CZ, then `H`/`S·H` and an `X`).
78    ///
79    /// [`change_basis1`]: Self::change_basis1
80    /// [`change_basis2`]: Self::change_basis2
81    pub fn fan_out(
82        &mut self,
83        pivot: usize,
84        cx_targets: &[usize],
85        cz_targets: &[usize],
86        last: impl Fn(bool, bool) -> ([bool; 2], bool),
87    ) {
88        debug_assert!(
89            cx_targets
90                .iter()
91                .chain(cz_targets)
92                .all(|&t| t < self.n && t != pivot)
93        );
94        let (pw, pm) = bit_position(pivot);
95        self.for_each_row(|x, z| {
96            let old_pivot = [x[pw] & pm != ZERO, z[pw] & pm != ZERO];
97            let (mut pivot_bits, mut flip) = (old_pivot, false);
98            if old_pivot[0] {
99                pivot_bits[0] = true;
100                for &t in cx_targets {
101                    flip ^= target_step(x, z, t, &mut pivot_bits, heisenberg::cnot);
102                }
103                for &t in cz_targets {
104                    flip ^= target_step(x, z, t, &mut pivot_bits, heisenberg::cz);
105                }
106            } else {
107                pivot_bits[0] = false;
108                for &t in cx_targets {
109                    flip ^= target_step(x, z, t, &mut pivot_bits, heisenberg::cnot);
110                }
111                for &t in cz_targets {
112                    flip ^= target_step(x, z, t, &mut pivot_bits, heisenberg::cz);
113                }
114            }
115            let (new_pivot, f) = last(pivot_bits[0], pivot_bits[1]);
116            x[pw] ^= pm & mask_if(new_pivot[0] != old_pivot[0]);
117            z[pw] ^= pm & mask_if(new_pivot[1] != old_pivot[1]);
118            flip ^ f
119        });
120    }
121}

123/// Run a two-qubit `rule` on the pivot's bits (held in `pivot_bits`) and
124/// target `t`'s bits in one row (`x`, `z`); store the target's changed bits.
125/// Returns the sign flip.
126#[inline(always)]
127fn target_step(
128    x: &mut [BitWord],
129    z: &mut [BitWord],
130    t: usize,
131    pivot_bits: &mut [bool; 2],
132    rule: impl Fn([bool; 4]) -> ([bool; 4], bool),
133) -> bool {
134    let (w, m) = bit_position(t);
135    let old = [x[w] & m != ZERO, z[w] & m != ZERO];
136    let ([px, pz, tx, tz], flip) = rule([pivot_bits[0], pivot_bits[1], old[0], old[1]]);
137    x[w] ^= m & mask_if(tx != old[0]);
138    z[w] ^= m & mask_if(tz != old[1]);
139    *pivot_bits = [px, pz];
140    flip
141}

148    #[test]
149    fn change_basis_preserves_commutation() {
150        for (i, n) in [1, 2, 7, 65].into_iter().enumerate() {
151            let mut t = InverseTableau::new(n);
152            for gate in random_circuit(n, 100, 0xba5e + i as u64) {
153                change_basis(&mut t, gate);
154                assert_canonical_commutation(&t);
155            }
156        }
157    }

159    /// `(G·C)·V = G·(C·V)`: the order of physical gates relative to changes of
160    /// basis does not matter.
161    #[test]
162    fn physical_gates_commute_with_changes_of_basis() {
163        for (i, n) in SIZES.into_iter().filter(|&n| n > 1).enumerate() {
164            let physical = random_circuit(n, 100, 0xc0de + i as u64);
165            let basis = random_circuit(n, 100, 0xfeed + i as u64);
166            let (mut first, mut second) = (InverseTableau::new(n), InverseTableau::new(n));
167            physical.iter().for_each(|&g| apply(&mut first, g));
168            basis.iter().for_each(|&g| change_basis(&mut first, g));
169            basis.iter().for_each(|&g| change_basis(&mut second, g));
170            physical.iter().for_each(|&g| apply(&mut second, g));
171            assert_eq!(first, second, "n = {n}");
172        }
173    }

175    type Rule1 = fn(bool, bool) -> ([bool; 2], bool);

177    /// The fused fan-out equals the separate changes of basis it replaces.
178    #[test]
179    fn fan_out_matches_separate_changes_of_basis() {
180        use ppvm_pauli_kernels::heisenberg as rules;
181        let lasts: [Rule1; 6] = [
182            |x, z| ([x, z], false),
183            rules::h,
184            rules::s,
185            rules::x,
186            rules::sqrt_x,
187            rules::sqrt_y_dag,
188        ];
189        let mut seed = 0xfa40_u64;
190        let mut next = move || {
191            seed ^= seed << 13;
192            seed ^= seed >> 7;
193            seed ^= seed << 17;
194            seed as usize
195        };
196        for n in SIZES.into_iter().filter(|&n| n > 1) {
197            for round in 0..20 {
198                let mut start = InverseTableau::new(n);
199                random_circuit(n, 60, next() as u64)
200                    .into_iter()
201                    .for_each(|g| apply(&mut start, g));
202                random_circuit(n, 20, next() as u64)
203                    .into_iter()
204                    .for_each(|g| change_basis(&mut start, g));
205                let pivot = next() % n;
206                // Each other qubit is a CX target, a CZ target, or neither.
207                let (mut cx, mut cz) = (Vec::new(), Vec::new());
208                for t in (0..n).filter(|&t| t != pivot) {
209                    match next() % 4 {
210                        0 => cx.push(t),
211                        1 => cz.push(t),
212                        _ => {}
213                    }
214                }
215                let last = lasts[round % lasts.len()];
216                let mut fused = start.clone();
217                fused.fan_out(pivot, &cx, &cz, last);
218                let mut separate = start;
219                cx.iter()
220                    .for_each(|&t| separate.change_basis2(pivot, t, rules::cnot));
221                cz.iter()
222                    .for_each(|&t| separate.change_basis2(pivot, t, rules::cz));
223                separate.change_basis1(pivot, last);
224                assert_eq!(fused, separate, "n = {n}, round {round}");
225            }
226        }
227    }
