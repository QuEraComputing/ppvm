4//! The interface of the active part.
5
6use num_complex::Complex64;
7
8use super::axis::{AxisPauli, RotationCollapse};
9
10/// Storage for `φ` over the active axes `0..k`. Only three things happen to
11/// it: a Pauli rotation, an activation, and a collapse; every other change of
12/// basis is trivial on the state. It sees axes only, never the tableau.
13///
14/// `P` is unsigned: the caller folds signs into the coefficients and the
15/// outcome. Weights are unnormalized, so `rotate` need not be unitary.
16pub trait ActiveState: Clone {
17    fn num_axes(&self) -> usize;
18
19    /// Append axis `k` in `|0⟩`.
20    fn activate(&mut self);
21
22    /// `φ ← a·φ + b·P φ`.
23    fn rotate(&mut self, p: AxisPauli, a: Complex64, b: Complex64);
24
25    /// Unnormalized weights `‖½(1 + P) φ‖²` and `‖½(1 − P) φ‖²` of the
26    /// outcomes `+1` and `−1`.
27    fn branch_weights(&self, p: AxisPauli) -> [f64; 2];
28
29    /// Project onto outcome `m` of `p` (eigenvalue `(−1)^m`), apply `V†` for
30    /// `v = RotationCollapse::collapse(p, m)`, drop axis `v.pivot` (now `|0⟩`), and
31    /// renormalize. Axes above the pivot shift down by one. The branch must
32    /// have nonzero weight.
33    fn collapse(&mut self, p: AxisPauli, m: bool, v: &RotationCollapse);
34}
