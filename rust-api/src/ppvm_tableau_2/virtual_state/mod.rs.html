4//! The state `|χ⟩` of the virtual qubits, with `|ψ⟩ = C |χ⟩`, behind the
5//! [`VirtualState`] interface.
6//!
7//! - `active_dormant`: [`ActiveDormant`], `|χ⟩ = |φ⟩ ⊗ |0…0⟩` with `φ` over
8//!   the active qubits, and the storage of `φ`.
9
10mod active_dormant;
11
12use num_complex::Complex64;
13
14use crate::VirtualView;
15
16pub use active_dormant::{ActiveDormant, ActiveState, AxisPauli, DenseState, RotationCollapse};

18/// The `2ⁿ`-dimensional state `|χ⟩` of the `n` virtual qubits, in a
19/// representation of the implementor's choice.
20///
21/// It sees the tableau only through a [`VirtualView`]: it may change basis
22/// (`C ← C·V` with `χ ← V† χ`, so `|ψ⟩` is unchanged), but never applies a
23/// physical gate. Arguments are physical qubits; the implementor reads
24/// `O = C⁻¹ Z_q C` from the view. The caller samples outcomes, so weights are
25/// unnormalized and `rotate` need not be unitary.
26pub trait VirtualState: Clone {
27    /// `|0…0⟩` on `num_qubits` virtual qubits.
28    fn new(num_qubits: usize) -> Self;
29
30    /// Whether `χ = |0…0⟩`, so `|ψ⟩ = C |0…0⟩` is a stabilizer state and the
31    /// caller may measure on the tableau alone. `false` is always correct.
32    fn is_zero(&self) -> bool {
33        false
34    }
35
36    /// `χ ← (a·I + b·O) χ` with `O = C⁻¹ (∏ Z_q) C` over `qubits`.
37    fn rotate(&mut self, t: &mut VirtualView<'_>, qubits: &[usize], a: Complex64, b: Complex64);
38
39    /// Unnormalized weights of the outcomes `0` and `1` of `Z_q`. Takes `&mut`
40    /// so an implementor may change basis here and reuse it in `collapse`.
41    fn z_weights(&mut self, t: &mut VirtualView<'_>, q: usize) -> [f64; 2];
42
43    /// Project onto outcome `m` of `Z_q` and renormalize. The outcome must have
44    /// nonzero weight.
45    fn collapse(&mut self, t: &mut VirtualView<'_>, q: usize, m: bool);
46}
