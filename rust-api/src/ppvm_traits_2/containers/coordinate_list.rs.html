4//! An unsorted `Vec<(K, C)>` backend using linear scans and `K: Eq + Clone`.
5//! Suitable for small supports; no hashing is required.
6
7use crate::algebra::{ImaginaryUnit, KeyProduct};
8use crate::arithmetic::Coefficient;
9use crate::containers::{Accumulate, Multiply, Pair, Retain, Scale, Support, TermBatch};
10
11impl<K, C> Support for Vec<(K, C)>
12where
13    K: Eq + Clone,
14    C: Coefficient,
15{
16    type Key = K;
17    type Coeff = C;
18
19    #[inline]
20    fn len(&self) -> usize {
21        self.as_slice().len()
22    }
23
24    #[inline]
25    fn get(&self, key: &K) -> Option<C> {
26        self.as_slice()
27            .iter()
28            .find(|(k, _)| k == key)
29            .map(|(_, c)| c.clone())
30    }
31
32    #[inline]
33    fn iter(&self) -> impl Iterator<Item = (K, C)> {
34        self.as_slice().iter().map(|(k, c)| (k.clone(), c.clone()))
35    }
36
37    /// The borrowing scan: no clone at all, so a reader that rejects most terms
38    /// pays nothing for the ones it rejects.
39    #[inline]
40    fn for_each_ref(&self, mut f: impl FnMut(&K, &C)) {
41        for (k, c) in self.as_slice() {
42            f(k, c);
43        }
44    }
45}
46
47impl<K, C> Accumulate for Vec<(K, C)>
48where
49    K: Eq + Clone,
50    C: Coefficient,
51{
52    /// Linear-scan hash-join: for each produced term, find the matching key and
53    /// add onto it, else append. `O(n·m)` in the support size, which is the
54    /// right cost model for the small support this backend targets.
55    #[inline]
56    fn accumulate_batch(&mut self, terms: &TermBatch<K, C>) {
57        for (k, c) in terms.iter() {
58            if let Some(slot) = self.iter_mut().find(|(ek, _)| ek == k) {
59                slot.1 += c;
60            } else {
61                self.push((k.clone(), c.clone()));
62            }
63        }
64    }
65
66    /// Drop every zero-coefficient term (`reduce_structural`): canonicalize to
67    /// reduced finite support. Runs only at finalize, never inline.
68    #[inline]
69    fn reduce(&mut self) {
70        self.retain(|(_, v)| !v.is_zero());
71    }
72}

74impl<K, C> Scale for Vec<(K, C)>
75where
76    K: Eq + Clone,
77    C: Coefficient,
78{
79    #[inline]
80    fn scale(&mut self, s: &C) {
81        for (_, v) in self.iter_mut() {
82            *v *= s;
83        }
84    }
85}

87/// Takes every [`Pair`] default. Each probe is a linear [`Support::get`] scan,
88/// so pairing costs `O(|self|·|other|)` — fine for the small supports this
89/// backend targets.
90impl<K, C> Pair for Vec<(K, C)>
91where
92    K: Eq + Clone,
93    C: Coefficient,
94{
95}

97impl<K, C> Retain<K, C> for Vec<(K, C)>
98where
99    K: Eq + Clone,
100    C: Coefficient,
101{
102    #[inline]
103    fn retain(&mut self, keep: impl Fn(&K, &C) -> bool) {
104        // Inherent `Vec::retain` shadows the trait method (inherent-first
105        // resolution), so this does not recurse.
106        self.retain(|(k, v)| keep(k, v));
107    }
108}

110impl<K, C> Multiply for Vec<(K, C)>
111where
112    K: KeyProduct,
113    C: ImaginaryUnit,
114{
115    /// Accumulate `(A·B)[k] = Σ_{p·q=k} A[p] B[q] i^{β(p,q)}` into `acc`.
116    /// No reduction or truncation runs; exact-zero entries remain until explicit reduction.
117    fn multiply_into(&self, other: &Self, acc: &mut Self) {
118        for (p, a) in self.as_slice() {
119            for (q, b) in other.as_slice() {
120                let (k, phase) = p.key_mul(q);
121                let mut product = a.clone();
122                product *= b;
123                let c = phase.apply(&product);
124                if let Some(slot) = acc.iter_mut().find(|(ek, _)| *ek == k) {
125                    slot.1 += c;
126                } else {
127                    acc.push((k, c));
128                }
129            }
130        }
131    }
132}

134#[cfg(test)]
135mod tests {
136    use super::*;
137    use crate::containers::TermSink;
138
139    fn batch(terms: &[(&str, f64)]) -> TermBatch<String, f64> {
140        let mut b = TermBatch::with_capacity(terms.len());
141        for (k, c) in terms {
142            b.push((*k).to_string(), *c);
143        }
144        b
145    }
146
147    #[test]
148    fn vec_accumulate_combines_keys() {
149        let mut v: Vec<(String, f64)> = Vec::new();
150        v.accumulate_batch(&batch(&[("a", 1.0), ("b", 2.0), ("a", 3.0)]));
151        assert_eq!(Support::get(&v, &"a".to_string()), Some(4.0));
152        assert_eq!(Support::get(&v, &"b".to_string()), Some(2.0));
153        assert_eq!(Support::len(&v), 2);
154    }
155
156    #[test]
157    fn vec_reduce_drops_zero() {
158        let mut v: Vec<(String, f64)> = Vec::new();
159        v.accumulate_batch(&batch(&[("a", 1.0), ("a", -1.0), ("b", 2.0)]));
160        v.reduce();
161        assert_eq!(Support::len(&v), 1);
162        assert_eq!(Support::get(&v, &"b".to_string()), Some(2.0));
163    }
164
165    #[test]
166    fn vec_scale_and_overlap() {
167        let mut a: Vec<(String, f64)> = Vec::new();
168        a.accumulate_batch(&batch(&[("x", 2.0), ("y", 3.0)]));
169        let mut b: Vec<(String, f64)> = Vec::new();
170        b.accumulate_batch(&batch(&[("x", 5.0), ("z", 7.0)]));
171        a.scale(&2.0);
172        // overlap = (2*2)*5 = 20; y and z do not match.
173        assert_eq!(Pair::overlap(&a, &b), 20.0);
174    }
175
176    #[test]
177    fn for_each_ref_agrees_with_iter() {
178        let terms = batch(&[("a", 1.0), ("b", 2.0), ("c", -3.0), ("a", 0.5)]);
179
180        let mut v: Vec<(String, f64)> = Vec::new();
181        v.accumulate_batch(&terms);
182        let mut seen: Vec<(String, f64)> = Vec::new();
183        v.for_each_ref(|k, c| seen.push((k.clone(), *c)));
184        seen.sort_by(|a, b| a.0.cmp(&b.0));
185        let mut want: Vec<(String, f64)> = Support::iter(&v).collect();
186        want.sort_by(|a, b| a.0.cmp(&b.0));
187        assert_eq!(seen, want);
188        assert_eq!(seen.len(), 3);
189    }
190
191    #[test]
192    fn vec_retain_filters() {
193        let mut v: Vec<(String, f64)> = Vec::new();
194        v.accumulate_batch(&batch(&[("keep", 2.0), ("drop", 0.5)]));
195        Retain::retain(&mut v, |_, c| *c >= 1.0);
196        assert_eq!(Support::len(&v), 1);
197        assert_eq!(Support::get(&v, &"keep".to_string()), Some(2.0));
198    }
