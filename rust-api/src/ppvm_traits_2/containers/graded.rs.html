4//! Container algebra over `C[K]`: support, accumulation, scaling, pairing, and products.
5//! Keys require `Eq + Clone`; hashing and Pauli capabilities are backend-specific.
6//! [`Retain`] is separate because truncation can break algebraic exactness.
7
8use num::Zero;
9
10use crate::algebra::{Conjugate, ImaginaryUnit, KeyProduct};
11use crate::arithmetic::Coefficient;
12use crate::containers::{KeyBatch, TermBatch, TermSink};

14/// A finitely supported map from keys to coefficients.
15/// Read-only export avoids requiring mutable pair slots in columnar backends.
16pub trait Support {
17    /// The key type — minimal `Eq + Clone`; hash backends add `Indexable`.
18    type Key: Eq + Clone;
19    /// The coefficient ring.
20    type Coeff: Coefficient;
21
22    /// Number of terms in the (reduced) support.
23    fn len(&self) -> usize;
24
25    /// Whether the support is empty.
26    fn is_empty(&self) -> bool {
27        self.len() == 0
28    }
29
30    /// The coefficient at `key`, if present.
31    fn get(&self, key: &Self::Key) -> Option<Self::Coeff>;
32
33    /// Read-only export of the support as `(key, coeff)` pairs. A SoA backend
34    /// synthesizes the pairs from its columns.
35    fn iter(&self) -> impl Iterator<Item = (Self::Key, Self::Coeff)>;
36
37    /// Visit each supported key and coefficient by reference, in backend order.
38    /// The default materializes owned pairs through [`Self::iter`]; stored backends
39    /// should override it to avoid cloning coefficients before filtering.
40    fn for_each_ref(&self, mut f: impl FnMut(&Self::Key, &Self::Coeff)) {
41        for (k, c) in self.iter() {
42            f(&k, &c);
43        }
44    }
45}

47/// Accumulate linear combinations and explicitly reduce exact-zero coefficients.
48pub trait Accumulate: Support {
49    /// Merge a batch, adding to existing keys or inserting new ones.
50    /// The algebraic multiset contract permits reordering and partitioning;
51    /// floating-point accumulation can still differ through rounding.
52    fn accumulate_batch(&mut self, terms: &TermBatch<Self::Key, Self::Coeff>);
53
54    /// Canonicalize to reduced finite-support form: drop every key whose
55    /// coefficient `is_zero()`. First-class and run **only** at finalize — never
56    /// inline during accumulation.
57    fn reduce(&mut self);
58
59    /// Accumulate one term through a singleton batch.
60    fn accumulate(&mut self, key: Self::Key, coeff: Self::Coeff) {
61        let mut batch = TermBatch::with_capacity(1);
62        batch.push(key, coeff);
63        self.accumulate_batch(&batch);
64    }
65}

67/// Scale coefficients by a scalar without changing their keys.
68pub trait Scale: Support {
69    /// Multiply every coefficient by `s`: `∀ k. c_k *= s`.
70    fn scale(&mut self, s: &Self::Coeff);
71}

73/// Pair supports: `overlap` computes `Σ a_k b_k` without conjugation.
74/// `hermitian_overlap` computes `Σ conj(a_k) b_k` and requires [`Conjugate`].
75/// For Pauli coefficients, the bilinear pairing is the normalized trace `Tr(A B)/2ⁿ`.
76pub trait Pair: Support {
77    /// Read-only probe of a key column: `out[i]` is the coefficient at
78    /// `keys[i]`, or `None` on a miss.
79    fn probe_batch(&self, keys: &KeyBatch<Self::Key>, out: &mut [Option<Self::Coeff>]) {
80        debug_assert!(out.len() >= keys.keys().len());
81        for (slot, k) in out.iter_mut().zip(keys.keys().iter()) {
82            *slot = self.get(k);
83        }
84    }
85
86    /// The symmetric bilinear trace pairing `∑_k a_k b_k`.
87    fn overlap(&self, other: &Self) -> Self::Coeff {
88        fold_common(self, other, |a, b| {
89            let mut product = a.clone();
90            product *= b;
91            product
92        })
93    }
94
95    /// The sesquilinear inner product `∑_k conj(a_k)·b_k`, conjugating `self`.
96    fn hermitian_overlap(&self, other: &Self) -> Self::Coeff
97    where
98        Self::Coeff: Conjugate,
99    {
100        fold_common(self, other, |a, b| {
101            let mut product = a.conj();
102            product *= b;
103            product
104        })
105    }
106}

108/// Sum `combine` over keys present in both supports, scanning the smaller side
109/// (`O(min(|lhs|, |rhs|))` probes, so summation order varies with the sizes).
110/// `combine` always receives `(lhs coeff, rhs coeff)`, whichever side scanned.
111fn fold_common<S>(
112    lhs: &S,
113    rhs: &S,
114    mut combine: impl FnMut(&S::Coeff, &S::Coeff) -> S::Coeff,
115) -> S::Coeff
116where
117    S: Support + ?Sized,
118{
119    let mut acc = S::Coeff::zero();
120    if lhs.len() <= rhs.len() {
121        lhs.for_each_ref(|k, a| {
122            if let Some(b) = rhs.get(k) {
123                acc += combine(a, &b);
124            }
125        });
126    } else {
127        rhs.for_each_ref(|k, b| {
128            if let Some(a) = lhs.get(k) {
129                acc += combine(&a, b);
130            }
131        });
132    }
133    acc
134}

136/// Accumulate a ring product using [`KeyProduct`] and [`ImaginaryUnit`].
137/// Key-product phases are absorbed into coefficients; types without a product
138/// need not implement this layer.
139pub trait Multiply: Accumulate
140where
141    Self::Key: KeyProduct,
142    Self::Coeff: ImaginaryUnit,
143{
144    /// Accumulate the ring product `self · other` into `acc`.
145    fn multiply_into(&self, other: &Self, acc: &mut Self);
146}

148/// Retain terms selected by a predicate, independently of the algebraic layers.
149/// Used by truncation policies; error guarantees depend on the coefficient magnitude.
150pub trait Retain<W, C> {
151    /// Retain exactly the terms for which `keep(&word, &coeff)` is `true`.
152    fn retain(&mut self, keep: impl Fn(&W, &C) -> bool);
153}
