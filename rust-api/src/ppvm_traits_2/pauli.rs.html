4/// A single-qubit Pauli symbol — the site alphabet of an ordinary packed Pauli
5/// word (`Word<Site = Pauli>`).
6#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
7pub enum Pauli {
8    /// Identity `I`.
9    I,
10    /// Pauli `X`.
11    X,
12    /// Pauli `Y`.
13    Y,
14    /// Pauli `Z`.
15    Z,
16}
17/// Phase-free symplectic operations on X/Z bit columns.
18/// Shared by single-word bits and multi-row tableau columns.
19pub trait SymplecticColumns {
20    /// Number of qubits (columns) this operator spans.
21    fn n_qubits(&self) -> usize;
22
23    /// `H` on `q`: swap the X and Z columns.
24    fn swap_xz(&mut self, q: usize);
25
26    /// S bit update on `q`: `z_q ⊕= x_q` (maps X to Y).
27    fn xor_z_from_x(&mut self, q: usize);
28
29    /// `CNOT` bit rule, part one: `x_tgt ⊕= x_ctrl`.
30    fn xor_x_col(&mut self, ctrl: usize, tgt: usize);
31
32    /// `CNOT` bit rule, part two: `z_ctrl ⊕= z_tgt`.
33    fn xor_z_col(&mut self, tgt: usize, ctrl: usize);
34
35    /// CZ bit update: `z_a ⊕= x_b` and `z_b ⊕= x_a`.
36    fn cz_bits(&mut self, a: usize, b: usize);
37}
38
39/// Extension-part: the phase algebra. `ℤ₄` for a phased word, `ℤ₂` + the
40/// Aaronson–Gottesman `g`-rule for a tableau. One phase delta per gate; the
41/// role-dependent half of conjugation, written **per type**.
42pub trait PhaseTrack {
43    /// `H` phase delta: flip the sign of a component with both `x` and `z` set
44    /// (`Y → −Y`).
45    fn flip_phase_where_xz(&mut self, q: usize);
46
47    /// `S` phase delta on `q`.
48    fn s_phase(&mut self, q: usize);
49
50    /// `CNOT` phase delta on `(ctrl, tgt)`.
51    fn cnot_phase(&mut self, ctrl: usize, tgt: usize);
52
53    /// `CZ` phase delta on `(a, b)`.
54    fn cz_phase(&mut self, a: usize, b: usize);
55
56    /// `X` phase delta on `q` (pure sign; no bit change).
57    fn x_phase(&mut self, q: usize);
58
59    /// `Y` phase delta on `q` (pure sign; no bit change).
60    fn y_phase(&mut self, q: usize);
61
62    /// `Z` phase delta on `q` (pure sign; no bit change).
63    fn z_phase(&mut self, q: usize);
64}
