4//! Binary bytecode (`.ssb`) round-trip for PPVM modules.
5//!
6//! Serializes a compiled [`PPVMModule`] to a little-endian container — header
7//! (magic + version + device info) → strings → function/label metadata → code —
8//! and reads it back. Each instruction is a `u16` opcode followed by its
9//! payload fields, as listed in the opcode table below.
10

25/// Generate the opcode enum and the instruction encoder/decoder from one table.
26///
27/// An entry's opcode is its position in the table, so the table is append-only:
28/// reordering or removing an entry changes the opcodes of every later one.
29macro_rules! opcode_table {
30    ($( $variant:ident = $($seg:ident)::+ $(( $($field:ident: $ty:ty),* ))? ),* $(,)?) => {
31        #[allow(non_camel_case_types, clippy::enum_variant_names)]
32        #[repr(u16)]
33        enum Opcode {
34            $( $variant, )*
35        }
36
37        fn write_instruction<W: Write>(inst: &Instruction, w: &mut W) -> eyre::Result<()> {
38            match inst {
39                $( Instruction::$variant($($seg)::+ $(( $($field),* ))?) => {
40                    w.write_all(&(Opcode::$variant as u16).to_le_bytes())?;
41                    $( $( $field.write_bytes(w)?; )* )?
42                } )*
43            }
44            Ok(())
45        }
46
47        fn read_instruction<R: Read>(r: &mut R) -> eyre::Result<Instruction> {
48            let opcode = read_u16(r)?;
49            $( if opcode == Opcode::$variant as u16 {
50                return Ok(Instruction::$variant($($seg)::+ $(( $( <$ty>::from_bytes(r)? ),* ))?));
51            } )*
52            Err(eyre::eyre!("unknown opcode {opcode}"))
53        }
54
55        /// One instance of every instruction, with distinct payload values.
56        #[cfg(test)]
57        fn every_instruction() -> Vec<Instruction> {
58            let mut next = 0u8;
59            let mut value = || {
60                next = next.wrapping_add(1);
61                next
62            };
63            vec![$( Instruction::$variant($($seg)::+ $(( $( <$ty>::from({ let _ = stringify!($field); value() }) ),* ))?), )*]
64        }
65    };
66}

278/// Current `.ssb` format version. The reader rejects any other version.
279pub const PPVM_BYTECODE_VERSION: u16 = 3;
280
281/// Byte length of the fixed portion of the header. The actual `header_size`
282/// in the stream may exceed this when the optional `observable` string is
283/// populated; the reader uses `header_size` to skip to the strings section.
284///
285/// Field widths (bytes): magic(4) + version(2) + header_size(4) + n_qubits(4)
286/// + coefficient_threshold(8) + backend(1) + max_pauli_weight_present(1)
287/// + max_pauli_weight(8) + observable_present(1) = 33.
288const FIXED_HEADER_SIZE: u32 = 4 + 2 + 4 + 4 + 8 + 1 + 1 + 8 + 1;
289
290/// Serialize a compiled module to the `.ssb` byte stream.
291pub fn write_module<W: Write>(module: &PPVMModule, w: &mut W) -> eyre::Result<()> {
292    if !module.constants.is_empty() || !module.source_symbols.is_empty() {
293        return Err(eyre::eyre!(
294            "bytecode cannot represent constants or source symbols"
295        ));
296    }
297
298    let info = &module.extra;
299    let n_qubits = u32::try_from(info.n_qubits)
300        .map_err(|_| eyre::eyre!("n_qubits {} does not fit in u32", info.n_qubits))?;
301
302    // The header is `FIXED_HEADER_SIZE` bytes plus, when an observable is
303    // present, a u32 length followed by its UTF-8 bytes.
304    let observable_bytes: &[u8] = info
305        .observable
306        .as_ref()
307        .map(String::as_bytes)
308        .unwrap_or(&[]);
309    let observable_present: u8 = u8::from(info.observable.is_some());
310    let observable_len = u32::try_from(observable_bytes.len()).map_err(|_| {
311        eyre::eyre!(
312            "observable length {} does not fit in u32",
313            observable_bytes.len()
314        )
315    })?;
316    let observable_trailer: u32 = if info.observable.is_some() {
317        4 + observable_len
318    } else {
319        0
320    };
321    let header_size = FIXED_HEADER_SIZE + observable_trailer;
322
323    // Header.
324    w.write_all(&PPVM_MAGIC.to_le_bytes())?;
325    w.write_all(&PPVM_BYTECODE_VERSION.to_le_bytes())?;
326    w.write_all(&header_size.to_le_bytes())?;
327    w.write_all(&n_qubits.to_le_bytes())?;
328    w.write_all(&info.coefficient_threshold.to_le_bytes())?;
329    w.write_all(&[backend_to_u8(info.backend)])?;
330    let (mpw_present, mpw_value) = match info.max_pauli_weight {
331        Some(w) => (
332            1u8,
333            u64::try_from(w)
334                .map_err(|_| eyre::eyre!("max_pauli_weight {} does not fit in u64", w))?,
335        ),
336        None => (0u8, 0u64),
337    };
338    w.write_all(&[mpw_present])?;
339    w.write_all(&mpw_value.to_le_bytes())?;
340    w.write_all(&[observable_present])?;
341    if info.observable.is_some() {
342        w.write_all(&observable_len.to_le_bytes())?;
343        w.write_all(observable_bytes)?;
344    }
345
346    // Strings section: count, then each entry as len-prefixed UTF-8.
347    let string_count =
348        u32::try_from(module.strings.len()).map_err(|_| eyre::eyre!("string count exceeds u32"))?;
349    w.write_all(&string_count.to_le_bytes())?;
350    for s in &module.strings {
351        let len = u32::try_from(s.len()).map_err(|_| eyre::eyre!("string length exceeds u32"))?;
352        w.write_all(&len.to_le_bytes())?;
353        w.write_all(s.as_bytes())?;
354    }
355
356    // Metadata section: function and label tables are required for calls and
357    // branches to retain their resolved targets after a bytecode round trip.
358    write_u32(w, u32::try_from(module.functions.len())?)?;
359    for function in &module.functions {
360        write_u32(w, function.name)?;
361        write_u32(w, function.local_count)?;
362        write_u32(w, function.start_address)?;
363        write_u32(w, function.end_address)?;
364        write_u32(w, function.file)?;
365        write_u32(w, u32::try_from(function.signature.params.len())?)?;
366        for parameter in &function.signature.params {
367            write_u32(w, parameter.name)?;
368            parameter.ty.write_bytes(w)?;
369        }
370        write_u32(w, u32::try_from(function.signature.ret.len())?)?;
371        for ty in &function.signature.ret {
372            ty.write_bytes(w)?;
373        }
374    }
375    write_u32(w, u32::try_from(module.labels.len())?)?;
376    for label in &module.labels {
377        write_u32(w, label.address)?;
378        write_u32(w, label.name)?;
379    }
380    write_u32(w, module.main_function.unwrap_or(u32::MAX))?;
381
382    // Code section: count, then each instruction's fixed-width frame.
383    let code_count =
384        u32::try_from(module.code.len()).map_err(|_| eyre::eyre!("code length exceeds u32"))?;
385    w.write_all(&code_count.to_le_bytes())?;
386    for inst in &module.code {
387        write_instruction(inst, w)?;
388    }
389
390    Ok(())
391}

393/// Reconstruct a module from a `.ssb` byte stream.
394pub fn read_module<R: Read>(r: &mut R) -> eyre::Result<PPVMModule> {
395    // Header.
396    let magic = read_u32(r)?;
397    if magic != PPVM_MAGIC {
398        return Err(eyre::eyre!(
399            "not a PPVM bytecode file (magic 0x{magic:08X})"
400        ));
401    }
402    let version = read_u16(r)?;
403    if version != PPVM_BYTECODE_VERSION {
404        return Err(eyre::eyre!("unsupported bytecode version {version}"));
405    }
406    let header_size = read_u32(r)?;
407    let n_qubits = read_u32(r)? as usize;
408    let coefficient_threshold = read_f64(r)?;
409    let backend = backend_from_u8(read_u8(r)?)?;
410    let mpw_present = read_u8(r)?;
411    let mpw_value = read_u64(r)?;
412    let max_pauli_weight = match mpw_present {
413        0 => None,
414        1 => Some(usize::try_from(mpw_value).map_err(|_| {
415            eyre::eyre!("max_pauli_weight {mpw_value} does not fit in usize on this platform")
416        })?),
417        other => {
418            return Err(eyre::eyre!(
419                "invalid max_pauli_weight presence byte {other}"
420            ));
421        }
422    };
423    let observable_present = read_u8(r)?;
424    let observable = match observable_present {
425        0 => None,
426        1 => {
427            let len = read_u32(r)? as usize;
428            let mut bytes = vec![0u8; len];
429            r.read_exact(&mut bytes)?;
430            Some(String::from_utf8(bytes)?)
431        }
432        other => {
433            return Err(eyre::eyre!("invalid observable presence byte {other}"));
434        }
435    };
436
437    // Sections begin at `header_size`; skip any header bytes beyond what this
438    // reader knows about (forward compat / self-description).
439    let consumed = FIXED_HEADER_SIZE
440        + if observable.is_some() {
441            4 + u32::try_from(observable.as_deref().unwrap().len())
442                .map_err(|_| eyre::eyre!("observable length does not fit in u32"))?
443        } else {
444            0
445        };
446    if header_size < consumed {
447        return Err(eyre::eyre!(
448            "header_size {header_size} smaller than the {consumed} bytes already consumed"
449        ));
450    }
451    skip_bytes(r, u64::from(header_size - consumed))?;
452
453    // Don't pre-allocate from an untrusted count; grow as entries are read.
454    let string_count = read_u32(r)?;
455    let mut strings = Vec::new();
456    for _ in 0..string_count {
457        let len = read_u32(r)? as usize;
458        let mut bytes = vec![0u8; len];
459        r.read_exact(&mut bytes)?;
460        strings.push(String::from_utf8(bytes)?);
461    }
462
463    let function_count = read_u32(r)?;
464    let mut functions = Vec::new();
465    for _ in 0..function_count {
466        let name = read_u32(r)?;
467        let local_count = read_u32(r)?;
468        let start_address = read_u32(r)?;
469        let end_address = read_u32(r)?;
470        let file = read_u32(r)?;
471        let parameter_count = read_u32(r)?;
472        let mut params = Vec::new();
473        for _ in 0..parameter_count {
474            params.push(Parameter {
475                name: read_u32(r)?,
476                ty: Type::from_bytes(r)?,
477            });
478        }
479        let return_count = read_u32(r)?;
480        let mut ret = Vec::new();
481        for _ in 0..return_count {
482            ret.push(Type::from_bytes(r)?);
483        }
484        functions.push(FunctionInfo {
485            name,
486            signature: Signature { params, ret },
487            local_count,
488            start_address,
489            end_address,
490            file,
491        });
492    }
493    let label_count = read_u32(r)?;
494    let mut labels = Vec::new();
495    for _ in 0..label_count {
496        labels.push(LabelInfo {
497            address: read_u32(r)?,
498            name: read_u32(r)?,
499        });
500    }
501    let main = read_u32(r)?;
502    let main_function = (main != u32::MAX).then_some(main);
503
504    let code_count = read_u32(r)?;
505    let mut code = Vec::new();
506    for _ in 0..code_count {
507        code.push(read_instruction(r)?);
508    }
509
510    Ok(PPVMModule {
511        extra: PPVMDeviceInfo {
512            magic,
513            n_qubits,
514            coefficient_threshold,
515            backend,
516            observable,
517            max_pauli_weight,
518        },
519        strings,
520        functions,
521        labels,
522        main_function,
523        code,
524        ..Default::default()
525    })
526}

528fn backend_to_u8(backend: BackendKind) -> u8 {
529    match backend {
530        BackendKind::Tableau => 0,
531        BackendKind::PauliSum => 1,
532        BackendKind::LossyPauliSum => 2,
533    }
534}
535
536fn backend_from_u8(byte: u8) -> eyre::Result<BackendKind> {
537    match byte {
538        0 => Ok(BackendKind::Tableau),
539        1 => Ok(BackendKind::PauliSum),
540        2 => Ok(BackendKind::LossyPauliSum),
541        other => Err(eyre::eyre!("invalid backend tag {other}")),
542    }
543}

545/// Serialize a module to an owned byte vector.
546pub fn module_to_bytes(module: &PPVMModule) -> eyre::Result<Vec<u8>> {
547    let mut buf = Vec::new();
548    write_module(module, &mut buf)?;
549    Ok(buf)
550}
551
552/// Reconstruct a module from a byte slice.
553pub fn module_from_bytes(bytes: &[u8]) -> eyre::Result<PPVMModule> {
554    read_module(&mut &bytes[..])
555}
556
557/// Cheap sniff: does this byte stream begin with the PPVM `.ssb` magic?
558///
559/// Reads the leading four bytes the same way [`read_module`] does — as a
560/// little-endian `u32` — so a positive result here means [`read_module`] will
561/// accept the magic. A stream shorter than the magic is not bytecode.
562pub fn is_bytecode(bytes: &[u8]) -> bool {
563    bytes.len() >= 4 && u32::from_le_bytes([bytes[0], bytes[1], bytes[2], bytes[3]]) == PPVM_MAGIC
564}
565
566/// "Dump": compile `.sst` source straight to the `.ssb` byte stream.
567pub fn compile_to_bytes(source: &str) -> eyre::Result<Vec<u8>> {
568    let module = crate::compile_program(source)?;
569    module_to_bytes(&module)
570}

572fn read_u8<R: Read>(r: &mut R) -> eyre::Result<u8> {
573    let mut b = [0u8; 1];
574    r.read_exact(&mut b)?;
575    Ok(b[0])
576}

578fn write_u32<W: Write>(w: &mut W, value: u32) -> eyre::Result<()> {
579    w.write_all(&value.to_le_bytes())?;
580    Ok(())
581}

606
607fn skip_bytes<R: Read>(r: &mut R, n: u64) -> eyre::Result<()> {
608    let skipped = std::io::copy(&mut r.take(n), &mut std::io::sink())?;
609    if skipped != n {
610        return Err(eyre::eyre!("unexpected EOF skipping {n} header bytes"));
611    }
612    Ok(())
613}
