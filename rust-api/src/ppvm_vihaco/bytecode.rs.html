4//! Binary bytecode (`.ssb`) round-trip for PPVM modules.
5//!
6//! Serializes a resolved [`Module`] to a little-endian container — header
7//! (magic + version + device info) → strings section → code section — and
8//! reads it back. The per-instruction codec is the existing `WriteBytes` /
9//! `FromBytes` generated for [`PPVMInstruction`]; this module only frames the
10//! container around it.
11

48/// Makes the boilerplate encode and decode functions for moving between
49/// {Type|Value} and Bytecode{Type|Value}.
50macro_rules! define_bytecode_codec {
51    (
52        $source:ident, $encoded:ident, $encode:ident, $decode:ident;
53        $($variant:ident $(($binding:ident))?),+ $(,)?
54    ) => {
55        fn $encode(value: $source) -> $encoded {
56            match value {
57                $(
58                    $source::$variant $(($binding))? =>
59                        $encoded::$variant $(($binding))?,
60                )+
61            }
62        }
63
64        fn $decode(value: $encoded) -> $source {
65            match value {
66                $(
67                    $encoded::$variant $(($binding))? =>
68                        $source::$variant $(($binding))?,
69                )+
70            }
71        }
72    };
73}

351/// Current `.ssb` format version. The reader rejects any other version.
352pub const PPVM_BYTECODE_VERSION: u16 = 2;
353
354/// Byte length of the fixed portion of the header. The actual `header_size`
355/// in the stream may exceed this when the optional `observable` string is
356/// populated; the reader uses `header_size` to skip to the strings section.
357///
358/// Field widths (bytes): magic(4) + version(2) + header_size(4) + n_qubits(4)
359/// + coefficient_threshold(8) + backend(1) + max_pauli_weight_present(1)
360/// + max_pauli_weight(8) + observable_present(1) = 33.
361const FIXED_HEADER_SIZE: u32 = 4 + 2 + 4 + 4 + 8 + 1 + 1 + 8 + 1;
362
363/// Serialize a resolved module to the v2 `.ssb` byte stream.
364pub fn write_module<W: Write>(module: &PPVMModule, w: &mut W) -> eyre::Result<()> {
365    if !module.constants.is_empty() || !module.source_symbols.is_empty() {
366        return Err(eyre::eyre!(
367            "bytecode v2 cannot represent constants or source symbols"
368        ));
369    }
370
371    let info = &module.extra;
372    let n_qubits = u32::try_from(info.n_qubits)
373        .map_err(|_| eyre::eyre!("n_qubits {} does not fit in u32", info.n_qubits))?;
374
375    // The header is `FIXED_HEADER_SIZE` bytes plus, when an observable is
376    // present, a u32 length followed by its UTF-8 bytes.
377    let observable_bytes: &[u8] = info
378        .observable
379        .as_ref()
380        .map(String::as_bytes)
381        .unwrap_or(&[]);
382    let observable_present: u8 = u8::from(info.observable.is_some());
383    let observable_len = u32::try_from(observable_bytes.len()).map_err(|_| {
384        eyre::eyre!(
385            "observable length {} does not fit in u32",
386            observable_bytes.len()
387        )
388    })?;
389    let observable_trailer: u32 = if info.observable.is_some() {
390        4 + observable_len
391    } else {
392        0
393    };
394    let header_size = FIXED_HEADER_SIZE + observable_trailer;
395
396    // Header.
397    w.write_all(&PPVM_MAGIC.to_le_bytes())?;
398    w.write_all(&PPVM_BYTECODE_VERSION.to_le_bytes())?;
399    w.write_all(&header_size.to_le_bytes())?;
400    w.write_all(&n_qubits.to_le_bytes())?;
401    w.write_all(&info.coefficient_threshold.to_le_bytes())?;
402    w.write_all(&[backend_to_u8(info.backend)])?;
403    let (mpw_present, mpw_value) = match info.max_pauli_weight {
404        Some(w) => (
405            1u8,
406            u64::try_from(w)
407                .map_err(|_| eyre::eyre!("max_pauli_weight {} does not fit in u64", w))?,
408        ),
409        None => (0u8, 0u64),
410    };
411    w.write_all(&[mpw_present])?;
412    w.write_all(&mpw_value.to_le_bytes())?;
413    w.write_all(&[observable_present])?;
414    if info.observable.is_some() {
415        w.write_all(&observable_len.to_le_bytes())?;
416        w.write_all(observable_bytes)?;
417    }
418
419    // Strings section: count, then each entry as len-prefixed UTF-8.
420    let string_count =
421        u32::try_from(module.strings.len()).map_err(|_| eyre::eyre!("string count exceeds u32"))?;
422    w.write_all(&string_count.to_le_bytes())?;
423    for s in &module.strings {
424        let len = u32::try_from(s.len()).map_err(|_| eyre::eyre!("string length exceeds u32"))?;
425        w.write_all(&len.to_le_bytes())?;
426        w.write_all(s.as_bytes())?;
427    }
428
429    // Metadata section: function and label tables are required for calls and
430    // branches to retain their resolved targets after a bytecode round trip.
431    write_u32(w, u32::try_from(module.functions.len())?)?;
432    for function in &module.functions {
433        write_u32(w, function.name)?;
434        write_u32(w, function.local_count)?;
435        write_u32(w, function.start_address)?;
436        write_u32(w, function.end_address)?;
437        write_u32(w, function.file)?;
438        write_u32(w, u32::try_from(function.signature.params.len())?)?;
439        for parameter in &function.signature.params {
440            write_u32(w, parameter.name)?;
441            parameter.ty.write_bytes(w)?;
442        }
443        write_u32(w, u32::try_from(function.signature.ret.len())?)?;
444        for ty in &function.signature.ret {
445            ty.write_bytes(w)?;
446        }
447    }
448    write_u32(w, u32::try_from(module.labels.len())?)?;
449    for label in &module.labels {
450        write_u32(w, label.address)?;
451        write_u32(w, label.name)?;
452    }
453    write_u32(w, module.main_function.unwrap_or(u32::MAX))?;
454
455    // Code section: count, then each instruction's fixed-width frame.
456    let code_count =
457        u32::try_from(module.code.len()).map_err(|_| eyre::eyre!("code length exceeds u32"))?;
458    w.write_all(&code_count.to_le_bytes())?;
459    for inst in &module.code {
460        encode_instruction(inst)?.write_bytes(w)?;
461    }
462
463    Ok(())
464}

466/// Reconstruct a module from a v2 `.ssb` byte stream.
467pub fn read_module<R: Read>(r: &mut R) -> eyre::Result<PPVMModule> {
468    // Header.
469    let magic = read_u32(r)?;
470    if magic != PPVM_MAGIC {
471        return Err(eyre::eyre!(
472            "not a PPVM bytecode file (magic 0x{magic:08X})"
473        ));
474    }
475    let version = read_u16(r)?;
476    if version != PPVM_BYTECODE_VERSION {
477        return Err(eyre::eyre!("unsupported bytecode version {version}"));
478    }
479    let header_size = read_u32(r)?;
480    let n_qubits = read_u32(r)? as usize;
481    let coefficient_threshold = read_f64(r)?;
482    let backend = backend_from_u8(read_u8(r)?)?;
483    let mpw_present = read_u8(r)?;
484    let mpw_value = read_u64(r)?;
485    let max_pauli_weight = match mpw_present {
486        0 => None,
487        1 => Some(usize::try_from(mpw_value).map_err(|_| {
488            eyre::eyre!("max_pauli_weight {mpw_value} does not fit in usize on this platform")
489        })?),
490        other => {
491            return Err(eyre::eyre!(
492                "invalid max_pauli_weight presence byte {other}"
493            ));
494        }
495    };
496    let observable_present = read_u8(r)?;
497    let observable = match observable_present {
498        0 => None,
499        1 => {
500            let len = read_u32(r)? as usize;
501            let mut bytes = vec![0u8; len];
502            r.read_exact(&mut bytes)?;
503            Some(String::from_utf8(bytes)?)
504        }
505        other => {
506            return Err(eyre::eyre!("invalid observable presence byte {other}"));
507        }
508    };
509
510    // Sections begin at `header_size`; skip any header bytes beyond what this
511    // reader knows about (forward compat / self-description).
512    let consumed = FIXED_HEADER_SIZE
513        + if observable.is_some() {
514            4 + u32::try_from(observable.as_deref().unwrap().len())
515                .map_err(|_| eyre::eyre!("observable length does not fit in u32"))?
516        } else {
517            0
518        };
519    if header_size < consumed {
520        return Err(eyre::eyre!(
521            "header_size {header_size} smaller than the {consumed} bytes already consumed"
522        ));
523    }
524    skip_bytes(r, u64::from(header_size - consumed))?;
525
526    // Don't pre-allocate from an untrusted count; grow as entries are read.
527    let string_count = read_u32(r)?;
528    let mut strings = Vec::new();
529    for _ in 0..string_count {
530        let len = read_u32(r)? as usize;
531        let mut bytes = vec![0u8; len];
532        r.read_exact(&mut bytes)?;
533        strings.push(String::from_utf8(bytes)?);
534    }
535
536    let function_count = read_u32(r)?;
537    let mut functions = Vec::new();
538    for _ in 0..function_count {
539        let name = read_u32(r)?;
540        let local_count = read_u32(r)?;
541        let start_address = read_u32(r)?;
542        let end_address = read_u32(r)?;
543        let file = read_u32(r)?;
544        let parameter_count = read_u32(r)?;
545        let mut params = Vec::new();
546        for _ in 0..parameter_count {
547            params.push(Parameter {
548                name: read_u32(r)?,
549                ty: Type::from_bytes(r)?,
550            });
551        }
552        let return_count = read_u32(r)?;
553        let mut ret = Vec::new();
554        for _ in 0..return_count {
555            ret.push(Type::from_bytes(r)?);
556        }
557        functions.push(FunctionInfo {
558            name,
559            signature: Signature { params, ret },
560            local_count,
561            start_address,
562            end_address,
563            file,
564        });
565    }
566    let label_count = read_u32(r)?;
567    let mut labels = Vec::new();
568    for _ in 0..label_count {
569        labels.push(LabelInfo {
570            address: read_u32(r)?,
571            name: read_u32(r)?,
572        });
573    }
574    let main = read_u32(r)?;
575    let main_function = (main != u32::MAX).then_some(main);
576
577    let code_count = read_u32(r)?;
578    let mut code = Vec::new();
579    for _ in 0..code_count {
580        code.push(decode_instruction(BytecodeInstruction::from_bytes(r)?));
581    }
582
583    Ok(PPVMModule {
584        extra: PPVMDeviceInfo {
585            magic,
586            n_qubits,
587            coefficient_threshold,
588            backend,
589            observable,
590            max_pauli_weight,
591        },
592        strings,
593        functions,
594        labels,
595        main_function,
596        code,
597        ..Default::default()
598    })
599}

601fn backend_to_u8(backend: BackendKind) -> u8 {
602    match backend {
603        BackendKind::Tableau => 0,
604        BackendKind::PauliSum => 1,
605        BackendKind::LossyPauliSum => 2,
606    }
607}
608
609fn backend_from_u8(byte: u8) -> eyre::Result<BackendKind> {
610    match byte {
611        0 => Ok(BackendKind::Tableau),
612        1 => Ok(BackendKind::PauliSum),
613        2 => Ok(BackendKind::LossyPauliSum),
614        other => Err(eyre::eyre!("invalid backend tag {other}")),
615    }
616}

618/// Serialize a module to an owned byte vector.
619pub fn module_to_bytes(module: &PPVMModule) -> eyre::Result<Vec<u8>> {
620    let mut buf = Vec::new();
621    write_module(module, &mut buf)?;
622    Ok(buf)
623}
624
625/// Reconstruct a module from a byte slice.
626pub fn module_from_bytes(bytes: &[u8]) -> eyre::Result<PPVMModule> {
627    read_module(&mut &bytes[..])
628}
629
630/// Cheap sniff: does this byte stream begin with the PPVM `.ssb` magic?
631///
632/// Reads the leading four bytes the same way [`read_module`] does — as a
633/// little-endian `u32` — so a positive result here means [`read_module`] will
634/// accept the magic. A stream shorter than the magic is not bytecode.
635pub fn is_bytecode(bytes: &[u8]) -> bool {
636    bytes.len() >= 4 && u32::from_le_bytes([bytes[0], bytes[1], bytes[2], bytes[3]]) == PPVM_MAGIC
637}
638
639/// "Dump": compile `.sst` source straight to the `.ssb` byte stream.
640pub fn compile_to_bytes(source: &str) -> eyre::Result<Vec<u8>> {
641    let module = crate::compile_program(source)?;
642    module_to_bytes(&module)
643}

645fn read_u8<R: Read>(r: &mut R) -> eyre::Result<u8> {
646    let mut b = [0u8; 1];
647    r.read_exact(&mut b)?;
648    Ok(b[0])
649}
650
651fn write_u32<W: Write>(w: &mut W, value: u32) -> eyre::Result<()> {
652    w.write_all(&value.to_le_bytes())?;
653    Ok(())
654}
655
656fn read_u16<R: Read>(r: &mut R) -> eyre::Result<u16> {
657    let mut b = [0u8; 2];
658    r.read_exact(&mut b)?;
659    Ok(u16::from_le_bytes(b))
660}
661
662fn read_u32<R: Read>(r: &mut R) -> eyre::Result<u32> {
663    let mut b = [0u8; 4];
664    r.read_exact(&mut b)?;
665    Ok(u32::from_le_bytes(b))
666}
667
668fn read_u64<R: Read>(r: &mut R) -> eyre::Result<u64> {
669    let mut b = [0u8; 8];
670    r.read_exact(&mut b)?;
671    Ok(u64::from_le_bytes(b))
672}
673
674fn read_f64<R: Read>(r: &mut R) -> eyre::Result<f64> {
675    let mut b = [0u8; 8];
676    r.read_exact(&mut b)?;
677    Ok(f64::from_le_bytes(b))
678}
679
680fn skip_bytes<R: Read>(r: &mut R, n: u64) -> eyre::Result<()> {
681    let skipped = std::io::copy(&mut r.take(n), &mut std::io::sink())?;
682    if skipped != n {
683        return Err(eyre::eyre!("unexpected EOF skipping {n} header bytes"));
684    }
685    Ok(())
686}
