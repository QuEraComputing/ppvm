14/// Largest qubit count any backend can simulate. The widest size bucket is
15/// backed by 2048-bit integers (`U2048` / `[u8; 256]`), so every constructor
16/// rejects `n_qubits > MAX_QUBITS` rather than panicking.
17pub const MAX_QUBITS: usize = 2048;
18
19/// Truncation strategy used by every `PauliSum` / `LossyPauliSum` size bucket.
20/// Coefficient-threshold pruning is always on; the Pauli-weight cap is set per
21/// run from the header (defaults to `usize::MAX` = no cap).
22pub(crate) type PauliSumStrategy = CombinedStrategy<CoefficientThreshold, MaxPauliWeight>;
23
24/// `PauliSum<T>`'s `T` for the lossless backend: `[u8; N]` storage, fx hash,
25/// f64 coefficients, the strategy above.
26pub(crate) type PauliSumConfig<const N: usize> = ByteFxHashF64<N, PauliSumStrategy>;
27
28/// Same as `PauliSumConfig` but with `LossyPauliWord` as the word type, so the
29/// loss-channel methods are dispatchable on the resulting `PauliSum<T>`.
30/// `LossyPauliWord`'s second type parameter (hasher) defaults to
31/// `fxhash::FxBuildHasher`, matching `ByteFxHashF64`'s internal hasher.
32pub(crate) type LossyPauliSumConfig<const N: usize> =
33    ByteFxHashF64<N, PauliSumStrategy, LossyPauliWord<[u8; N]>>;
34
35/// Build a `PauliSumStrategy` value from a `PPVMDeviceInfo`. Pulled out so the
36/// six size-bucket constructors don't each repeat the strategy spelling.
37pub(crate) fn paulisum_strategy(info: &PPVMDeviceInfo) -> PauliSumStrategy {
38    CombinedStrategy(
39        CoefficientThreshold(info.coefficient_threshold),
40        MaxPauliWeight(info.max_pauli_weight.unwrap_or(usize::MAX)),
41    )
42}

44/// Trait defining all methods required to implement the backend of the VM component
45pub trait CircuitBackend {
46    type Config: Config<Coeff = f64>;
47    type State: Clifford
48        + CliffordExtensions
49        + CliffordBatch
50        + CliffordExtensionsBatch
51        + Reset
52        + TGate<Self::Config>
53        + U3Gate<Self::Config>
54        + RotationOne<Self::Config>
55        + RotXY<Self::Config>
56        + RotationTwo<Self::Config>
57        + Depolarizing<Self::Config>
58        + Depolarizing2<Self::Config>
59        + PauliError<Self::Config>
60        + TwoQubitPauliError<Self::Config>
61        + LossChannel<Self::Config>
62        + CorrelatedLossChannel<Self::Config>;
63
64    fn state(&mut self) -> &mut Self::State;
65
66    /// Number of qubits on the device; every qubit operand must be below it.
67    fn n_qubits(&self) -> usize;
68
69    // NOTE: methods that differ between backends get dedicated methods
70
71    fn measure(&mut self, _addr0: usize) -> eyre::Result<MeasurementOutcome> {
72        Err(eyre!("measure is not supported on this backend"))
73    }
74
75    fn leakage(&mut self, _addr0: usize, _p0: f64, _p1: f64) -> eyre::Result<()> {
76        Err(eyre!("leakage is not supported on this backend"))
77    }
78
79    fn truncate(&mut self);
80
81    /// `Σ_{P matches pattern} ⟨P⟩` on the current state.
82    fn trace(&self, pattern: &PauliPattern) -> eyre::Result<f64>;
83
84    fn reset(&mut self);
85}

87impl<T, I, C> CircuitBackend for GeneralizedTableau<T, I, C>
88where
89    T: Config<Coeff = f64>,
90    <<T as Config>::Storage as BitView>::Store: PrimInt,
91    I: TableauIndex + Send + Sync + std::fmt::Debug,
92    C: SparseVector<Complex64, I> + std::fmt::Debug,
93{
94    type Config = T;
95    type State = GeneralizedTableau<T, I, C>;
96
97    fn state(&mut self) -> &mut Self::State {
98        self
99    }
100
101    fn n_qubits(&self) -> usize {
102        GeneralizedTableau::n_qubits(self)
103    }
104
105    fn measure(&mut self, q: usize) -> eyre::Result<MeasurementOutcome> {
106        let m = LossyMeasure::measure(self, q);
107        let e = match m {
108            Some(b) => {
109                if b {
110                    MeasurementOutcome::One
111                } else {
112                    MeasurementOutcome::Zero
113                }
114            }
115            None => MeasurementOutcome::Lost,
116        };
117        Ok(e)
118    }
119
120    fn truncate(&mut self) {}
121
122    fn trace(&self, pattern: &PauliPattern) -> eyre::Result<f64> {
123        Ok(GeneralizedTableau::trace(self, pattern))
124    }
125
126    fn reset(&mut self) {
127        self.reset_all();
128    }
129
130    fn leakage(&mut self, _addr0: usize, p0: f64, p1: f64) -> eyre::Result<()> {
131        self.leakage_channel(_addr0, p0, p1);
132        Ok(())
133    }
134}

136/// PauliSum that also stores the initial observable so we can reset the component
137pub struct PauliSumBackend<T: Config<Coeff = f64>> {
138    pub state: PauliSum<T>,
139    pub(crate) initial_state: PauliSum<T>,
140}
141
142impl<T: Config<Coeff = f64>> PauliSumBackend<T> {
143    pub fn new(observable: PauliSum<T>) -> Self {
144        Self {
145            state: observable.clone(),
146            initial_state: observable.clone(),
147        }
148    }
149}
150
151impl<T> CircuitBackend for PauliSumBackend<T>
152where
153    T: Config<Coeff = f64>,
154    T::PauliWordType: WordReset,
155    for<'a> PauliSum<T>: Trace<'a, PauliPattern, Output = f64>,
156{
157    type Config = T;
158    type State = PauliSum<T>;
159
160    fn state(&mut self) -> &mut Self::State {
161        &mut self.state
162    }
163
164    fn n_qubits(&self) -> usize {
165        self.state.n_qubits()
166    }
167
168    fn reset(&mut self) {
169        self.state = self.initial_state.clone();
170    }
171
172    fn truncate(&mut self) {
173        self.state.truncate();
174    }
175
176    fn trace(&self, pattern: &PauliPattern) -> eyre::Result<f64> {
177        Ok(Trace::trace(&self.state, pattern))
178    }
179}
