60// A breakpoint suspends the invocation via the debugger's `Break` effect.
61impl Step<field!(PPVM::debugger), debug::Breakpoint, Program> for PPVM {
62    fn step(&mut self, instruction: &debug::Breakpoint, module: &Program) -> eyre::Result<()> {
63        let message = <Self as BuildMessage<field!(PPVM::debugger), debug::Breakpoint, Program>>::build_message(
64            self, instruction, module,
65        )?;
66        let effect = self.debugger.execute(instruction, message)?;
67        self.execution.apply(effect);
68        Ok(())
69    }
70}

174// handling circuit effects
175impl vihaco::Apply<Effects<CircuitOutcomeEffect>> for PPVM {
176    fn apply(&mut self, effects: Effects<CircuitOutcomeEffect>) {
177        for effect in effects {
178            // Programs branch on outcomes: push each as u32 (0/1/2, 2 = Lost),
179            // and can consume trace values the same way.
180            match &effect {
181                CircuitOutcomeEffect::Measurement(m) => {
182                    for &outcome in &m.measurement_results {
183                        self.execution.apply(PushToStack((outcome as u32,)));
184                    }
185                }
186                CircuitOutcomeEffect::Trace(t) => {
187                    self.execution.apply(PushToStack((t.value,)));
188                }
189            }
190            self.record_outcome(effect);
191        }
192    }
193}

204impl Record<Effects<CircuitOutcomeEffect>> for PPVM {

212impl PPVM {
213    fn record_outcome(&mut self, effect: CircuitOutcomeEffect) {
214        match effect {
215            CircuitOutcomeEffect::Measurement(m) => {
216                self.measurement_record.record.push(m.measurement_results)
217            }
218            CircuitOutcomeEffect::Trace(t) => self.trace_record.record.push(t.value),
219        }
220    }

222    /// Execute one circuit instruction on a ready-made message, outside any
223    /// running program.
224    ///
225    /// The operand stack is not touched: outcomes land in the measurement and
226    /// trace records only. A qubit index outside the device is an error.
227    pub fn send<I>(&mut self, instruction: &I, message: I::Message) -> eyre::Result<()>
228    where
229        I: CircuitOp,
230        Self: Record<I::Effect>,
231    {
232        let effect = self.circuit.execute(instruction, message)?;
233        self.record(effect);
234        Ok(())
235    }

247    /// Enter `module`'s main without executing anything; `resume` runs it.
248    pub fn start(&mut self, module: &Program) -> eyre::Result<()> {
249        self.execution.start(module, &[])
250    }
251
252    /// Run the started program until it halts, returns, or hits a breakpoint.
253    pub fn resume(&mut self, module: &Program) -> eyre::Result<()> {
254        vihaco_stdlib::debugger::resume_with_breaks::<field!(PPVM::debugger), _, _, _, _, _>(
255            module,
256            self,
257            vihaco_stdlib::debugger::DebugAction::Continue,
258        )
259    }
260
261    /// Start `module` and run it to the end, continuing past every breakpoint.
262    pub fn run(&mut self, module: &Program) -> eyre::Result<()> {
263        self.start(module)?;
264        // Breakpoints only pause the interactive debugger; a batch run skips past them.
265        self.resume(module)?;
266        while self.is_paused() {
267            self.resume(module)?;
268        }
269        Ok(())
270    }
271
272    pub fn single_step(&mut self, module: &Program) -> eyre::Result<()> {
273        vihaco_stdlib::debugger::resume_with_breaks::<field!(PPVM::debugger), _, _, _, _, _>(
274            module,
275            self,
276            vihaco_stdlib::debugger::DebugAction::Step,
277        )
278    }
279
280    /// Index of the next instruction to execute.
281    pub fn pc(&self) -> u32 {
282        let pc = self.execution.pc();
283        match self.execution.run_state() {
284            // An explicit breakpoint leaves `pc` on itself; resuming skips past it.
285            // A gutter breakpoint resumes by running the instruction it displaced.
286            vihaco_stdlib::RunState::Other(DebuggerState::Breakpoint { resume_pc })
287                if !self.debugger.contains(&pc) =>
288            {
289                *resume_pc
290            }
291            _ => pc,
292        }
293    }
294
295    /// True while a breakpoint has suspended the running program.
296    pub fn is_paused(&self) -> bool {
297        matches!(
298            self.execution.run_state(),
299            vihaco_stdlib::RunState::Other(DebuggerState::Breakpoint { .. })
300        )
301    }
302
303    /// True once the program has finished (or before one was started).
304    pub fn is_halted(&self) -> bool {
305        self.execution.is_halted()
306    }
307}

343impl vihaco::Reset for PPVM {
344    fn reset(&mut self) {
345        self.execution.reset();
346        // Clears a paused breakpoint; installed gutter breakpoints live in the module.
347        self.debugger.reset();
348        self.cpu.reset();
349        self.circuit.reset();
350        self.measurement_record.record.clear();
351        self.trace_record.record.clear();
352    }
353}

364    /// Wrap header declarations and a straight-line `main` into SST v1 source.
365    /// `strings` are appended to the string table after `main` and the file name.
366    fn program(header: &str, strings: &[&str], body: &[&str]) -> String {
367        let strings: Vec<String> = ["main", "test.sst"]
368            .iter()
369            .chain(strings)
370            .map(|s| format!("{s:?}"))
371            .collect();
372        format!(
373            "sst v1
374.section(root):
375    .section(ppvm):
376    .header(ppvm):
377        {header}
378        metadata {{
379            strings: [{strings}],
380            functions: [
381                {{
382                    name: 0,
383                    signature: {{ params: [], ret: [] }},
384                    local_count: 0,
385                    start_address: 0,
386                    end_address: {end},
387                    file: 1,
388                }},
389            ],
390            labels: [],
391            constants: [],
392            main_function: 0,
393            source_symbols: [],
394            extra: {{}},
395        }};

397    .text(ppvm):
398        fn @main() {{
399            {body}
400        }}
401    .text(ppvm).
402    .section(ppvm).
403.section(root).
404",
405            strings = strings.join(", "),
406            end = body.len(),
