4//! Declarations in a PPVM program's `.header(...)` section, collected into the
5//! program metadata and a [`PPVMDeviceInfo`].
6
7use crate::device_info::{BackendKind, PPVMDeviceInfo};
8use vihaco::module::ProgramMetadata;
9use vihaco::{Parse, Type, Value};
10use vihaco_parser::Ident;
11
12pub type Metadata = ProgramMetadata<Value, Type>;
13
14#[derive(Debug, Clone, PartialEq, Parse)]
15#[syntax_class(value)]
16pub enum BackendKindSyntax {
17    #[pattern = "`tableau`"]
18    Tableau,
19    #[pattern = "`paulisum`"]
20    PauliSum,
21    #[pattern = "`lossy_paulisum`"]
22    LossyPauliSum,
23}
24
25impl From<BackendKindSyntax> for BackendKind {
26    fn from(value: BackendKindSyntax) -> Self {
27        match value {
28            BackendKindSyntax::Tableau => Self::Tableau,
29            BackendKindSyntax::PauliSum => Self::PauliSum,
30            BackendKindSyntax::LossyPauliSum => Self::LossyPauliSum,
31        }
32    }
33}
34
35/// One `;`-terminated declaration in a PPVM header section.
36#[derive(Debug, Parse)]
37#[syntax_class(metadata)]
38pub enum HeaderDeclaration {
39    #[pattern = "$0"]
40    Metadata(Metadata),
41    #[pattern = "circuit.n_qubits $0"]
42    NumQubits(usize),
43    #[pattern = "circuit.coefficient_threshold $0"]
44    CoefficientThreshold(f64),
45    #[pattern = "circuit.backend $0"]
46    Backend(BackendKindSyntax),
47    #[pattern = "circuit.observable $0"]
48    Observable(Ident),
49    #[pattern = "circuit.max_pauli_weight $0"]
50    MaxPauliWeight(usize),
51}
52
53/// Collect header declarations into the program metadata and device info.
54///
55/// Exactly one metadata block is required. A repeated `circuit.*` declaration
56/// overwrites the earlier one, as in the previous header syntax.
57pub fn collect_header(
58    declarations: impl IntoIterator<Item = HeaderDeclaration>,
59) -> eyre::Result<(Metadata, PPVMDeviceInfo)> {
60    let mut metadata = None;
61    let mut info = PPVMDeviceInfo::default();
62    for declaration in declarations {
63        match declaration {
64            HeaderDeclaration::Metadata(m) => {
65                eyre::ensure!(metadata.is_none(), "duplicate metadata declaration");
66                metadata = Some(m);
67            }
68            HeaderDeclaration::NumQubits(n) => info.n_qubits = n,
69            HeaderDeclaration::CoefficientThreshold(t) => info.coefficient_threshold = t,
70            HeaderDeclaration::Backend(b) => info.backend = b.into(),
71            HeaderDeclaration::Observable(s) => info.observable = Some(s.0),
72            HeaderDeclaration::MaxPauliWeight(w) => info.max_pauli_weight = Some(w),
73        }
74    }
75    let metadata = metadata.ok_or_else(|| eyre::eyre!("missing program metadata"))?;
76    Ok((metadata, info))
77}
