21/// Compile `.sst` source and write it to a `.ssb` bytecode file.
22pub fn dump_program(program: &str, output_path: &str) -> eyre::Result<()> {
23    std::fs::write(output_path, bytecode::compile_to_bytes(program)?)?;
24    Ok(())
25}
26
27/// Read a `.sst` file and dump it to a `.ssb` bytecode file.
28pub fn dump_file(input_path: &str, output_path: &str) -> eyre::Result<()> {
29    dump_program(&std::fs::read_to_string(input_path)?, output_path)
30}
