18#[derive(Debug, Clone, PartialEq, vihaco_parser_derive::Parse)]
19#[syntax_class(value)]
20pub enum BackendKindSyntax {
21    #[pattern = "`tableau`"]
22    Tableau,
23    #[pattern = "`paulisum`"]
24    PauliSum,
25    #[pattern = "`lossy_paulisum`"]
26    LossyPauliSum,
27}
28
29impl From<BackendKindSyntax> for BackendKind {
30    fn from(value: BackendKindSyntax) -> Self {
31        match value {
32            BackendKindSyntax::Tableau => Self::Tableau,
33            BackendKindSyntax::PauliSum => Self::PauliSum,
34            BackendKindSyntax::LossyPauliSum => Self::LossyPauliSum,
35        }
36    }
37}

39#[derive(Debug, Clone, PartialEq, vihaco_parser_derive::Parse)]
40#[syntax_class(metadata, head = "device")]
41pub enum PPVMHeader {
42    #[pattern = "circuit.n_qubits $0"]
43    NumQubits(usize),
44    #[pattern = "circuit.coefficient_threshold $0"]
45    CoefficientThreshold(f64),
46    #[pattern = "circuit.backend $0"]
47    Backend(BackendKindSyntax),
48    #[pattern = "circuit.observable $0"]
49    Observable(Ident),
50    #[pattern = "circuit.max_pauli_weight $0"]
51    MaxPauliWeight(usize),
52}

54#[derive(Debug, Default)]
55pub struct PPVMResolver {
56    strings: Vec<String>,
57}
58
59impl PPVMResolver {
60    pub fn new() -> Self {
61        Self::default()
62    }
63
64    fn apply_header(info: &mut PPVMDeviceInfo, header: PPVMHeader) -> eyre::Result<()> {
65        match header {
66            PPVMHeader::NumQubits(n) => info.n_qubits = n,
67            PPVMHeader::CoefficientThreshold(t) => info.coefficient_threshold = t,
68            PPVMHeader::Backend(b) => info.backend = b.into(),
69            PPVMHeader::Observable(s) => info.observable = Some(s.as_str().to_owned()),
70            PPVMHeader::MaxPauliWeight(w) => info.max_pauli_weight = Some(w),
71        }
72        Ok(())
73    }
74
75    fn intern(&mut self, value: &str) -> u32 {
76        if let Some(index) = self.strings.iter().position(|item| item == value) {
77            return index as u32;
78        }
79        let index = self.strings.len() as u32;
80        self.strings.push(value.to_owned());
81        index
82    }
83
84    fn runtime_type(ty: SurfaceType) -> Type {
85        match ty {
86            SurfaceType::Undefined => Type::Undefined,
87            SurfaceType::String => Type::String,
88            SurfaceType::Bool => Type::Bool,
89            SurfaceType::I64 => Type::I64,
90            SurfaceType::U32 => Type::U32,
91            SurfaceType::U64 => Type::U64,
92            SurfaceType::F64 => Type::F64,
93            SurfaceType::FunctionRef => Type::FunctionRef,
94            SurfaceType::HeapRef => Type::HeapRef,
95        }
96    }
97
98    fn lower_value(&mut self, ty: SurfaceType, value: SurfaceValue) -> eyre::Result<Value> {
99        let text = match value {
100            SurfaceValue::Quoted(QuotedString(value)) => {
101                return Ok(Value::String(self.intern(&value)));
102            }
103            SurfaceValue::Bare(BareToken(value)) => value,
104        };
105        Ok(match ty {
106            SurfaceType::Undefined => Value::Undefined,
107            SurfaceType::String => Value::String(self.intern(&text)),
108            SurfaceType::Bool => Value::Bool(text.parse()?),
109            SurfaceType::I64 => Value::I64(text.parse()?),
110            SurfaceType::U32 => Value::U32(text.parse()?),
111            SurfaceType::U64 => Value::U64(text.parse()?),
112            SurfaceType::F64 => Value::F64(text.parse()?),
113            SurfaceType::FunctionRef => Value::FunctionRef(text.parse()?),
114            SurfaceType::HeapRef => Value::HeapRef(text.parse()?),
115        })
116    }
117
118    fn lower_cpu(
119        &mut self,
120        instruction: CpuSurface,
121        labels: &HashMap<String, u32>,
122        functions: &HashMap<String, u32>,
123    ) -> eyre::Result<Option<CpuRuntime>> {
124        use CpuRuntime as R;
125        use CpuSurface as S;
126        let runtime = match instruction {
127            S::Span(a, b, c) => R::Span(a, b, c),
128            S::Label(_) => return Ok(None),
129            S::FunctionStart => R::FunctionStart,
130            S::FunctionEnd => R::FunctionEnd,
131            S::Breakpoint => R::Breakpoint,
132            S::Branch(name) => R::Branch(
133                *labels
134                    .get(name.as_str())
135                    .ok_or_else(|| eyre::eyre!("undefined label `@{}`", name.as_str()))?,
136            ),
137            S::ConditionalBranch(a, b) => R::ConditionalBranch(
138                *labels
139                    .get(a.as_str())
140                    .ok_or_else(|| eyre::eyre!("undefined label `@{}`", a.as_str()))?,
141                *labels
142                    .get(b.as_str())
143                    .ok_or_else(|| eyre::eyre!("undefined label `@{}`", b.as_str()))?,
144            ),
145            S::Return(n) => R::Return(n),
146            S::IndirectCall => R::IndirectCall,
147            S::Call(arity, name) => R::Call(
148                arity,
149                *functions
150                    .get(name.as_str())
151                    .ok_or_else(|| eyre::eyre!("undefined function `@{}`", name.as_str()))?,
152            ),
153            S::Halt => R::Halt,
154            S::Print => R::Print,
155            S::Load(ty, slot) => R::Load(Self::runtime_type(ty), slot),
156            S::Store(ty, slot) => R::Store(Self::runtime_type(ty), slot),
157            S::Dup => R::Dup,
158            S::HeapAlloc(n) => R::HeapAlloc(n),
159            S::GetItem => R::GetItem,
160            S::HeapDealloc => R::HeapDealloc,
161            S::Const(ty, value) => {
162                let value = self.lower_value(ty, value)?;
163                R::Const(Self::runtime_type(ty), value)
164            }
165            S::Add(ty) => R::Add(Self::runtime_type(ty)),
166            S::Sub(ty) => R::Sub(Self::runtime_type(ty)),
167            S::Mul(ty) => R::Mul(Self::runtime_type(ty)),
168            S::Div(ty) => R::Div(Self::runtime_type(ty)),
169            S::Rem(ty) => R::Rem(Self::runtime_type(ty)),
170            S::Neg(ty) => R::Neg(Self::runtime_type(ty)),
171            S::Shl(ty) => R::Shl(Self::runtime_type(ty)),
172            S::Shr(ty) => R::Shr(Self::runtime_type(ty)),
173            S::Rol(ty) => R::Rol(Self::runtime_type(ty)),
174            S::Ror(ty) => R::Ror(Self::runtime_type(ty)),
175            S::BitAnd(ty) => R::BitAnd(Self::runtime_type(ty)),
176            S::BitOr(ty) => R::BitOr(Self::runtime_type(ty)),
177            S::BitXor(ty) => R::BitXor(Self::runtime_type(ty)),
178            S::Not => R::Not,
179            S::And => R::And,
180            S::Or => R::Or,
181            S::Xor => R::Xor,
182            S::Eq(ty) => R::Eq(Self::runtime_type(ty)),
183            S::Ne(ty) => R::Ne(Self::runtime_type(ty)),
184            S::Lt(ty) => R::Lt(Self::runtime_type(ty)),
185            S::Gt(ty) => R::Gt(Self::runtime_type(ty)),
186            S::Le(ty) => R::Le(Self::runtime_type(ty)),
187            S::Ge(ty) => R::Ge(Self::runtime_type(ty)),
188        };
189        Ok(Some(runtime))
190    }
191
192    fn lower_circuit(instruction: CircuitSurfaceInstruction) -> CircuitInstruction {
193        use CircuitInstruction as R;
194        use CircuitSurfaceInstruction as S;
195        match instruction {
196            S::TwoQubitPauliError => R::TwoQubitPauliError,
197            S::Truncate => R::Truncate,
198            S::Trace => R::Trace,
199            S::X => R::X,
200            S::Y => R::Y,
201            S::Z => R::Z,
202            S::H => R::H,
203            S::SqrtXAdj => R::SqrtXAdj,
204            S::SqrtX => R::SqrtX,
205            S::SqrtYAdj => R::SqrtYAdj,
206            S::SqrtY => R::SqrtY,
207            S::SAdj => R::SAdj,
208            S::S => R::S,
209            S::CNOT => R::CNOT,
210            S::CZ => R::CZ,
211            S::TAdj => R::TAdj,
212            S::T => R::T,
213            S::RXX => R::RXX,
214            S::RYY => R::RYY,
215            S::RZZ => R::RZZ,
216            S::RX => R::RX,
217            S::RY => R::RY,
218            S::RZ => R::RZ,
219            S::U3 => R::U3,
220            S::Measure => R::Measure,
221            S::Reset => R::Reset,
222            S::R => R::R,
223            S::Loss => R::Loss,
224            S::CorrelatedLoss => R::CorrelatedLoss,
225            S::PauliError => R::PauliError,
226            S::Depolarize2 => R::Depolarize2,
227            S::Depolarize => R::Depolarize,
228        }
229    }
230}

232impl Resolve<ppvm::syntax::Instruction, SurfaceType, Vec<PPVMHeader>> for PPVMResolver {
233    type Module = LocalModule<PPVMInstruction, Value, Type, PPVMDeviceInfo>;
234
235    fn resolve_module(
236        &mut self,
237        parsed: ParsedModule<ppvm::syntax::Instruction, SurfaceType, Vec<PPVMHeader>>,
238    ) -> eyre::Result<Self::Module> {
239        let mut module = LocalModule::default();
240        for header in parsed.header {
241            Self::apply_header(&mut module.extra, header)?;
242        }
243
244        let mut functions = HashMap::new();
245        let mut function_address = 0u32;
246        for function in &parsed.functions {
247            functions.insert(function.name.as_str().to_owned(), function_address);
248            function_address += function
249                .body
250                .iter()
251                .filter(|instruction| {
252                    !matches!(
253                        instruction,
254                        ppvm::syntax::Instruction::Cpu(CpuSurface::Label(_))
255                    )
256                })
257                .count() as u32;
258        }
259
260        let mut labels = HashMap::new();
261        let mut address = 0u32;
262        for function in &parsed.functions {
263            for instruction in &function.body {
264                if let ppvm::syntax::Instruction::Cpu(CpuSurface::Label(name)) = instruction {
265                    labels.insert(name.as_str().to_owned(), address);
266                    module.labels.push(LabelInfo {
267                        address,
268                        name: self.intern(name.as_str()),
269                    });
270                } else {
271                    address += 1;
272                }
273            }
274        }
275
276        for function in parsed.functions {
277            let start_address = module.code.len() as u32;
278            for instruction in function.body {
279                let runtime = match instruction {
280                    ppvm::syntax::Instruction::Cpu(cpu) => self
281                        .lower_cpu(cpu, &labels, &functions)?
282                        .map(PPVMInstruction::Cpu),
283                    ppvm::syntax::Instruction::Circuit(circuit) => {
284                        Some(PPVMInstruction::Circuit(Self::lower_circuit(circuit)))
285                    }
286                };
287                if let Some(runtime) = runtime {
288                    module.code.push(runtime);
289                }
290            }
291            let end_address = module.code.len() as u32;
292            module.functions.push(FunctionInfo {
293                name: self.intern(function.name.as_str()),
294                signature: Signature {
295                    params: function
296                        .params
297                        .into_iter()
298                        .map(|Param { name, ty }| Parameter {
299                            name: self.intern(name.as_str()),
300                            ty: Self::runtime_type(ty),
301                        })
302                        .collect(),
303                    ret: function
304                        .return_ty
305                        .map(Self::runtime_type)
306                        .into_iter()
307                        .collect(),
308                },
309                local_count: 0,
310                start_address,
311                end_address,
312                file: 0,
313            });
314        }
315        module.main_function = functions.get("main").copied();
316        module.strings = std::mem::take(&mut self.strings);
317        Ok(module)
318    }
319}

321pub fn parse_headers(source: &str) -> eyre::Result<(Vec<PPVMHeader>, String)> {
322    let mut headers = Vec::new();
323    let mut body = String::new();
324    for line in source.lines() {
325        let trimmed = line.trim();
326        if trimmed.starts_with("device ") {
327            let header = trimmed.strip_suffix(';').unwrap_or(trimmed);
328            let parsed = PPVMHeader::parser()
329                .parse(header)
330                .into_result()
331                .map_err(|errors| eyre::eyre!("invalid device header `{header}`: {errors:?}"))?;
332            headers.push(parsed);
333        } else {
334            body.push_str(line);
335            body.push('\n');
336        }
337    }
338    Ok((headers, body))
339}

341pub fn parse_functions(
342    source: &str,
343) -> eyre::Result<Vec<ParsedFunction<ppvm::syntax::Instruction, SurfaceType>>> {
344    skip()
345        .ignore_then(
346            ParsedFunction::<ppvm::syntax::Instruction, SurfaceType>::parser()
347                .repeated()
348                .collect::<Vec<_>>(),
349        )
350        .then_ignore(skip())
351        .parse(source)
352        .into_result()
353        .map_err(|errors| eyre::eyre!("parsing functions failed: {errors:?}"))
354}
